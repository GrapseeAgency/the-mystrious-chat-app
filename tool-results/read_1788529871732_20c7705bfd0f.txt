   245→   245→/**
   246→   246→ * Snippet with the first match highlighted — clips a ≤64-char window
   247→   247→ * around the hit so long messages stay one tidy line.
   248→   248→ */
   249→   249→function SearchSnippet({ content, query }: { content: string; query: string }) {
   250→   250→  const lower = content.toLowerCase()
   251→   251→  const q = query.toLowerCase()
   252→   252→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   253→   253→  let from = 0
   254→   254→  let clippedHead = false
   255→   255→  if (idx > 28) {
   256→   256→    from = idx - 24
   257→   257→    clippedHead = true
   258→   258→  }
   259→   259→  const end = Math.min(content.length, idx + q.length + 28)
   260→   260→  const clippedTail = end < content.length
   261→   261→  const body = content.slice(from, end)
   262→   262→  const localIdx = idx - from
   263→   263→  return (
   264→   264→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   265→   265→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   266→   266→      {idx >= 0 ? (
   267→   267→        <>
   268→   268→          {body.slice(0, localIdx)}
   269→   269→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   270→   270→            {body.slice(localIdx, localIdx + q.length)}
   271→   271→          </mark>
   272→   272→          {body.slice(localIdx + q.length)}
   273→   273→        </>
   274→   274→      ) : (
   275→   275→        body
   276→   276→      )}
   277→   277→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   278→   278→    </p>
   279→   279→  )
   280→   280→}

   398→   398→      const groupName =
   399→   399→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   400→   400→      const displayName = conversationDisplayName(conv, me.id)

   411→   411→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   412→   412→          previewDeleted: previewInfo.deleted,
   413→   413→          draft: allDrafts[conv.id] ?? null,
   414→   414→          unreadCount: conv.unreadCount,
   415→   415→          dmName: other?.name ?? null,
   416→   416→          dmColor: other?.color ?? 'emerald',
   417→   417→          groupTitle: groupName,

   419→   419→          pinned: conv.pinnedAt !== null,

   422→   422→        },
   423→   423→      }
   424→   424→    })
   425→   425→  }, [conversations.data, me.id, onlineIds, typersIn, allDrafts])
   426→   426→
   427→   427→  const filteredRows = useMemo(() => {
   428→   428→    const q = searchQuery.trim().toLowerCase()
   429→   429→    if (!q) return rows
   430→   430→    return rows.filter(({ conv, props }) => {
   431→   431→      if (props.name.toLowerCase().includes(q)) return true
   432→   432→      if (props.draft?.toLowerCase().includes(q)) return true
   433→   433→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   434→   434→      return false
   435→   435→    })
