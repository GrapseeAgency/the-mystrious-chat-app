    77→    77→function ttlLabel(ttlSeconds: number): string {
    78→    78→  if (ttlSeconds <= 0) return 'Off'
    79→    79→  if (ttlSeconds === 86_400) return '24h'
    80→    80→  if (ttlSeconds === 604_800) return '7d'
    81→    81→  if (ttlSeconds === 2_592_000) return '30d'
    82→    82→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))}d`
    83→    83→}

   104→   104→function DisappearingSlider({
   105→   105→  ttlSeconds,
   106→   106→  disabled,
   107→   107→  reducedMotion,
   108→   108→  onCommit,
   109→   109→}: {
   110→   110→  ttlSeconds: number
   111→   111→  disabled: boolean
   112→   112→  reducedMotion: boolean
   113→   113→  onCommit: (ttlSeconds: number) => void
   114→   114→}) {
   115→   115→  const trackRef = useRef<HTMLDivElement | null>(null)
   116→   116→  const [draftIdx, setDraftIdx] = useState(() =>
   117→   117→    Math.max(0, TTL_STOPS.indexOf(ttlSeconds as (typeof TTL_STOPS)[number])),
   118→   118→  )
   119→   119→  const [dragging, setDragging] = useState(false)
   120→   120→
   121→   121→  // follow the committed server truth whenever a drag is not in flight
   122→   122→  const committedIdx = Math.max(
   123→   123→    0,
   124→   124→    TTL_STOPS.indexOf(ttlSeconds as (typeof TTL_STOPS)[number]),
   125→   125→  )
   126→   126→  const shownIdx = dragging ? draftIdx : committedIdx
   127→   127→
   128→   128→  const idxFromClientX = (clientX: number): number => {
   129→   129→    const el = trackRef.current
   130→   130→    if (!el) return committedIdx
   131→   131→    const rect = el.getBoundingClientRect()
   132→   132→    if (rect.width <= 0) return committedIdx
   133→   133→    const ratio = (clientX - rect.left) / rect.width
   134→   134→    return Math.min(TTL_STOPS.length - 1, Math.max(0, Math.round(ratio * (TTL_STOPS.length - 1))))
   135→   135→  }
   136→   136→
   137→   137→  const stopPct = (shownIdx / (TTL_STOPS.length - 1)) * 100
   138→   138→  // keep the riding label inside the card at the extreme stops
   139→   139→  const labelPct = Math.min(86, Math.max(14, stopPct))
   140→   140→  const slideTransition = reducedMotion ? { duration: 0.12 } : spring.snappy
   141→   141→
   142→   142→  return (
   143→   143→    <div className="mt-3 px-0.5 pb-0.5">
   144→   144→      {/* live label riding under the thumb */}
   145→   145→      <div className="relative mb-0.5 h-4" aria-hidden>
   146→   146→        <motion.span
   147→   147→          animate={{ left: `${labelPct}%` }}
   148→   148→          transition={slideTransition}
   149→   149→          style={{ x: '-50%' }}
   150→   150→          className="absolute top-0 whitespace-nowrap rounded-full bg-emerald-500/15 px-1.5 py-px text-[9.5px] font-bold text-emerald-600 ring-1 ring-inset ring-emerald-500/25 dark:text-emerald-400"
   151→   151→        >
   152→   152→          {TTL_STOP_LABELS[shownIdx]}
   153→   153→        </motion.span>
   154→   154→      </div>
   155→   155→
   156→   156→      <div
   157→   157→        ref={trackRef}
   158→   158→        role="slider"
   159→   159→        aria-label="Disappearing message timer"
   160→   160→        aria-valuemin={0}
   161→   161→        aria-valuemax={TTL_STOPS.length - 1}
   162→   162→        aria-valuenow={shownIdx}
   163→   163→        aria-valuetext={TTL_STOP_LABELS[shownIdx]}
   164→   164→        aria-disabled={disabled}
   165→   165→        onPointerDown={(event) => {
   166→   166→          if (disabled) return
   167→   167→          event.currentTarget.setPointerCapture(event.pointerId)
   168→   168→          setDragging(true)
   169→   169→          setDraftIdx(idxFromClientX(event.clientX))
   170→   170→          haptic(6)
   171→   171→        }}
   172→   172→        onPointerMove={(event) => {
   173→   173→          if (!dragging || disabled) return
   174→   174→          const idx = idxFromClientX(event.clientX)
   175→   175→          if (idx !== draftIdx) {
   176→   176→            setDraftIdx(idx)
   177→   177→            haptic(4)
   178→   178→          }
   179→   179→        }}
   180→   180→        onPointerUp={(event) => {
   181→   181→          if (!dragging) return
   182→   182→          setDragging(false)
   183→   183→          const idx = idxFromClientX(event.clientX)
   184→   184→          setDraftIdx(idx)
   185→   185→          haptic(10)
   186→   186→          const next = TTL_STOPS[idx]
   187→   187→          if (next !== ttlSeconds) onCommit(next)
   188→   188→        }}
   189→   189→        onPointerCancel={() => setDragging(false)}
   190→   190→        className={cn('relative h-7 touch-none select-none', disabled && 'pointer-events-none opacity-50')}
   191→   191→      >
   192→   192→        {/* resting track + emerald fill up to the thumb */}
   193→   193→        <div className="absolute top-1/2 h-1.5 w-full -translate-y-1/2 rounded-full bg-zinc-900/[0.07] dark:bg-white/10" />
   194→   194→        <div
   195→   195→          className="absolute top-1/2 h-1.5 -translate-y-1/2 rounded-full bg-gradient-to-r from-emerald-500/60 to-emerald-500 transition-[width] duration-150"
   196→   196→          style={{ width: `${stopPct}%` }}
   197→   197→        />
   198→   198→        {/* tick stops */}
   199→   199→        {TTL_STOPS.map((_, idx) => (
   200→   200→          <span
   201→   201→            key={idx}
   202→   202→            className={cn(
   203→   203→              'absolute top-1/2 size-2 -translate-x-1/2 -translate-y-1/2 rounded-full ring-2 ring-white dark:ring-zinc-900',
   204→   204→              idx <= shownIdx ? 'bg-emerald-500' : 'bg-zinc-300 dark:bg-zinc-600',
   205→   205→            )}
   206→   206→            style={{ left: `${(idx / (TTL_STOPS.length - 1)) * 100}%` }}
   207→   207→          />
   208→   208→        ))}
   209→   209→        {/* springy thumb — scales on grab for the haptic feel */}
   210→   210→        <motion.span
   211→   211→          aria-hidden
   212→   212→          animate={{ left: `${stopPct}%`, scale: dragging ? 1.18 : 1 }}
   213→   213→          transition={slideTransition}
   214→   214→          style={{ x: '-50%', y: '-50%' }}
   215→   215→          className="absolute top-1/2 z-10 flex size-5 items-center justify-center rounded-full bg-white shadow-[0_2px_10px_rgba(0,0,0,0.28)] ring-1 ring-black/10 dark:bg-zinc-100"
   216→   216→        >
   217→   217→          <span className="block size-2 rounded-full bg-emerald-500" />
   218→   218→        </motion.span>
   219→   219→      </div>
   220→   220→
   221→   221→      {/* the old preset pills — kept as tap-to-commit tick labels */}
   222→   222→      <div className="mt-1.5 grid grid-cols-4 gap-1" role="group" aria-label="Timer presets">
   223→   223→        {TTL_STOPS.map((t, idx) => (
   224→   224→          <button
   225→   225→            key={t}
   226→   226→            type="button"
   227→   227→            disabled={disabled}
   228→   228→            onClick={() => onCommit(t)}
   229→   229→            className={cn(
   230→   230→              'h-7 rounded-full text-[11px] font-bold outline-none transition-colors active:scale-95 disabled:opacity-50',
   231→   231→              idx === shownIdx
   232→   232→                ? 'bg-emerald-500/15 text-emerald-700 ring-1 ring-emerald-400 dark:text-emerald-300'
   233→   233→                : 'bg-zinc-900/[0.05] text-zinc-600 hover:bg-emerald-500/15 hover:text-emerald-700 dark:bg-white/[0.07] dark:text-zinc-300 dark:hover:text-emerald-400',
   234→   234→            )}
   235→   235→          >
   236→   236→            {TTL_STOP_LABELS[idx]}
   237→   237→          </button>
   238→   238→        ))}
   239→   239→      </div>
   240→   240→    </div>
   241→   241→  )
   242→   242→}

   374→   374→    onSuccess: () => {

   384→   384→  })

   386→   386→  // PATCH /members/{userId} { requesterId, action: 'promote' | 'demote' }

   388→   388→  const roleMutation = useMutation({

   391→   391→        `/api/conversations/${encodeURIComponent(conversationId)}/members/${encodeURIComponent(userId)}`,
   392→   392→        { method: 'PATCH', body: JSON.stringify({ requesterId: me.id, action }) },
   393→   393→      )
   394→   394→    },
   395→   395→    onSuccess: (_data, vars) => {
   396→   396→      // Surgical role patch — the response detail is built for the TARGET
   397→   397→      // viewer (its myMutedUntil is not ours), so never cache it wholesale.
   398→   398→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   399→   399→        old
   400→   400→          ? {
   401→   401→              ...old,
   402→   402→              members: old.members.map((m) =>
   403→   403→                m.id === vars.userId
   404→   404→                  ? { ...m, role: (vars.action === 'promote' ? 'admin' : 'member') as GroupRole }

   406→   406→              ),
   407→   407→            }
   408→   408→          : old,
   409→   409→      )
   410→   410→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   411→   411→      void queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   412→   412→      toast.success(vars.action === 'promote' ? 'Promoted to admin' : 'Role set to member')
   413→   413→      haptic(12)
   414→   414→    },
   415→   415→    onError: (error) =>
   416→   416→      toast.error(error instanceof Error ? error.message : 'Could not update the role'),
   417→   417→  })
   418→   418→
   419→   419→  const copyInvite = async (code: string) => {
   420→   420→    const link = `${window.location.origin}/?join=${code}`
   421→   421→    try {
   422→   422→      await navigator.clipboard.writeText(link)
