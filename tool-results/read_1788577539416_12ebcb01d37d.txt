   181→/** True when the query is empty or ANY of the haystacks matches it. */
   182→function hit(query: string, ...haystacks: Array<string | null | undefined>): boolean {
   183→  const q = query.trim().toLowerCase()
   184→  if (!q) return true
   185→  return haystacks
   186→    .filter((t): t is string => typeof t === 'string' && t.length > 0)
   187→    .some((t) => t.toLowerCase().includes(q))
   188→}

   371→  const setUiTheme = useUiThemeStore((s) => s.setTheme)
   372→  const navStyle = useNavStyleStore((s) => s.style)
   373→  const setNavStyle = useNavStyleStore((s) => s.setStyle)
   374→
   375→  const [query, setQuery] = useState('')
   376→
