    83→        method: 'POST',

    90→  })

    93→    mutationFn: (input: { toUsername: string; amount: number; note: string }) =>

    95→        method: 'POST',
    96→        headers: { 'Content-Type': 'application/json' },

    98→      }),

   110→
   111→  return (
   112→    <div className="flex flex-col gap-4">
   113→      {/* balance hero */}
   114→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-emerald-500 via-teal-500 to-cyan-600 p-5 text-white shadow-lg">
   115→        <div className="absolute -right-6 -top-8 size-28 rounded-full bg-white/10 blur-xl" />
   116→        <p className="text-[11px] font-medium uppercase tracking-wider text-white/70">Pulse balance</p>
   117→        <div className="mt-1 flex items-end gap-4">
   118→          <div>
   119→            <p className="text-3xl font-bold leading-none">{wallet ? wallet.coins.toLocaleString() : '—'}</p>
   120→            <p className="mt-1 text-[11px] font-medium text-white/80">PC · Pulse Coins</p>
   121→          </div>
   122→          <div className="border-l border-white/20 pl-4">
   123→            <p className="flex items-center gap-1 text-xl font-bold leading-none">
   124→              <Gem className="size-4" /> {wallet ? wallet.gems : '—'}
   125→            </p>
   126→            <p className="mt-1 text-[11px] font-medium text-white/80">Gems</p>
   127→          </div>
   128→        </div>
   129→        <Button
   130→          size="sm"
   131→          disabled={checkin.isPending || wallet?.checkedInToday}
   132→          onClick={() => checkin.mutate()}
   133→          className="mt-4 w-full bg-white font-semibold text-emerald-700 hover:bg-white/90 disabled:opacity-60"
   134→        >
   135→          {wallet?.checkedInToday ? (
   136→            <>Checked in today · streak {wallet.streak}</>

   138→            <>
   139→              <Gift className="mr-1.5 size-4" /> Daily check-in · +25 PC

   147→        <h3 className="flex items-center gap-1.5 text-sm font-semibold">
   148→          <Send className="size-4 text-emerald-600" /> Send coins by @handle
   149→        </h3>
   150→        <div className="mt-3 grid grid-cols-2 gap-2">
   151→          <Input

   208→// ── Tasks panel (3-column kanban) ────────────────────────────

   210→const COLUMNS: Array<{ id: HubTaskItem['status']; label: string }> = [
   211→  { id: 'todo', label: 'To do' },

   213→  { id: 'done', label: 'Done' },

   215→
   216→function TasksPanel({ me }: { me: AppUser }) {
   217→  const qc = useQueryClient()
   218→  const [title, setTitle] = useState('')
   219→
   220→  const tasksQ = useQuery({
   221→    queryKey: ['hub-tasks', me.id],
   222→    queryFn: async () => {
   223→      const res = await apiJson<{ tasks: HubTaskItem[] }>(`/api/hub/tasks?userId=${encodeURIComponent(me.id)}`)
   224→      return res.tasks
   225→    },
   226→  })
   227→  const invalidate = () => void qc.invalidateQueries({ queryKey: ['hub-tasks', me.id] })
   228→
   229→  const add = useMutation({
   230→    mutationFn: (t: string) =>
   231→      apiJson<{ task: HubTaskItem }>('/api/hub/tasks', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ userId: me.id, title: t }),
   235→      }),
   236→    onSuccess: () => {
   237→      setTitle('')
   238→      invalidate()
   239→    },
   240→    onError: (err: Error) => toast.error(err.message),
   241→  })
   242→
   243→  const move = useMutation({
   244→    mutationFn: (input: { id: string; status: HubTaskItem['status'] }) =>
   245→      apiJson(`/api/hub/tasks/${input.id}`, {
   246→        method: 'PATCH',
   247→        headers: { 'Content-Type': 'application/json' },
   248→        body: JSON.stringify({ userId: me.id, status: input.status }),
   249→      }),
   250→    onSettled: invalidate,
   251→    onError: (err: Error) => toast.error(err.message),
   252→  })
   253→
   254→  const remove = useMutation({
   255→    mutationFn: (id: string) => apiJson(`/api/hub/tasks/${id}?userId=${encodeURIComponent(me.id)}`, { method: 'DELETE' }),
   256→    onSettled: invalidate,
   257→    onError: (err: Error) => toast.error(err.message),
   258→  })
   259→
   260→  const byStatus = useMemo(() => {
   261→    const map: Record<HubTaskItem['status'], HubTaskItem[]> = { todo: [], doing: [], done: [] }
   262→    for (const t of tasksQ.data ?? []) map[t.status]?.push(t)
   263→    return map
   264→  }, [tasksQ.data])
   265→
   266→  return (
   267→    <div className="flex flex-col gap-3">
   268→      <form
   269→        className="flex gap-2"
   270→        onSubmit={(e) => {
   271→          e.preventDefault()
   272→          if (title.trim()) add.mutate(title.trim())
   273→        }}
   274→      >
   275→        <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="New task…" className="h-9" />
   276→        <Button type="submit" size="sm" disabled={!title.trim() || add.isPending} className="shrink-0">
   277→          <Check className="size-4" />
   278→        </Button>
   279→      </form>
   280→
   281→      <div className="grid grid-cols-3 gap-2">
   282→        {COLUMNS.map((col) => (
   283→          <div key={col.id} className="rounded-xl bg-zinc-100/70 p-2 dark:bg-white/5">
   284→            <p className="mb-2 px-1 text-[11px] font-bold uppercase tracking-wide text-zinc-500">
   285→              {col.label} · {byStatus[col.id].length}
   286→            </p>
   287→            <ul className="flex flex-col gap-1.5">
   288→              {byStatus[col.id].map((t) => (
   289→                <motion.li
   290→                  key={t.id}
   291→                  layout
   292→                  initial={{ opacity: 0, scale: 0.95 }}
   293→                  animate={{ opacity: 1, scale: 1 }}
   294→                  className="rounded-lg border border-zinc-200 bg-white p-2 shadow-sm dark:border-white/10 dark:bg-zinc-900/80"
   295→                >
   296→                  <p className={cn('text-[12px] font-medium leading-snug', t.status === 'done' && 'text-zinc-400 line-through')}>
   297→                    {t.title}
   298→                  </p>
   299→                  <div className="mt-1.5 flex items-center justify-between">
   300→                    <button
   301→                      type="button"
   302→                      aria-label="Advance status"
   303→                      className="text-[10px] font-semibold text-emerald-600 hover:underline"
   304→                      onClick={() =>
   305→                        move.mutate({ id: t.id, status: t.status === 'todo' ? 'doing' : t.status === 'doing' ? 'done' : 'todo' })
   306→                      }
   307→                    >
   308→                      {t.status === 'done' ? '↺ reopen' : '→ next'}
   309→                    </button>
   310→                    <button
   311→                      type="button"
   312→                      aria-label="Delete task"
   313→                      className="text-[10px] text-zinc-400 hover:text-rose-500"
   314→                      onClick={() => remove.mutate(t.id)}
   315→                    >
   316→                      <X className="size-4" aria-hidden />
   317→                    </button>
   318→                  </div>
   319→                </motion.li>
   320→              ))}
   321→              {byStatus[col.id].length === 0 ? (
   322→                <li className="rounded-lg border border-dashed border-zinc-300 px-2 py-3 text-center text-[10px] text-zinc-400 dark:border-white/15">
   323→                  empty
   324→                </li>
   325→              ) : null}
   326→            </ul>
   327→          </div>
   328→        ))}
   329→      </div>
   330→    </div>
   331→  )
   332→}

   336→function MarketPanel({ me }: { me: AppUser }) {
   337→  const qc = useQueryClient()
   338→  const [title, setTitle] = useState('')
   339→  const [price, setPrice] = useState('')
   340→  const [description, setDescription] = useState('')
   341→
   342→  const marketQ = useQuery({
   343→    queryKey: ['hub-market', me.id],
   344→    queryFn: async () => {
   345→      const res = await apiJson<{ listings: MarketListingItem[] }>(`/api/hub/market?userId=${encodeURIComponent(me.id)}`)
   346→      return res.listings
   347→    },
   348→    refetchInterval: 30_000,
   349→  })
   350→  const walletQ = useQuery({
   351→    queryKey: ['hub-wallet-mini', me.id],
   352→    queryFn: async () => {
   353→      const res = await apiJson<{ wallet: WalletState }>(`/api/hub/wallet?userId=${encodeURIComponent(me.id)}`)
   354→      return res.wallet
   355→    },
   356→  })
   357→
   358→  const invalidate = () => {
   359→    void qc.invalidateQueries({ queryKey: ['hub-market', me.id] })
   360→    void qc.invalidateQueries({ queryKey: ['hub-wallet-mini', me.id] })
   361→  }

   363→  const list = useMutation({
   364→    mutationFn: () =>
   365→      apiJson('/api/hub/market', {
   366→        method: 'POST',
   367→        headers: { 'Content-Type': 'application/json' },
   368→        body: JSON.stringify({ userId: me.id, title, description, price: Number(price) }),

   371→      toast.success('Listing posted to the market')
   372→      setTitle('')
   373→      setPrice('')
   374→      setDescription('')
   375→    },
   376→    onError: (err: Error) => toast.error(err.message),
   377→    onSettled: invalidate,
   378→  })

   380→  const buy = useMutation({
   381→    mutationFn: (id: string) =>
   382→      apiJson<{ ok: boolean }>(`/api/hub/market/${id}/buy`, {
   383→        method: 'POST',
   384→        headers: { 'Content-Type': 'application/json' },
   385→        body: JSON.stringify({ userId: me.id }),
   386→      }),
   387→    onSuccess: () => toast.success('Purchased — coins transferred to the seller'),
   388→    onError: (err: Error) => toast.error(err.message),
   389→    onSettled: invalidate,
   390→  })
   391→
   392→  return (
   393→    <div className="flex flex-col gap-3">
   394→      <section aria-label="Create listing" className="rounded-2xl border border-zinc-200/80 p-4 dark:border-white/10 dark:bg-zinc-900/60">
   395→        <h3 className="flex items-center gap-1.5 text-sm font-semibold">
   396→          <CircleDollarSign className="size-4 text-emerald-600" /> Sell something

   398→        </h3>
   399→        <form
   400→          className="mt-3 flex flex-col gap-2"
   401→          onSubmit={(e) => {
   402→            e.preventDefault()
   403→            if (title.trim() && price) list.mutate()
   404→          }}
   405→        >
   406→          <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Title (what are you selling?)" className="h-9" />
   407→          <Input value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Description (optional)" className="h-9" />
   408→          <div className="flex gap-2">
   409→            <Input value={price} onChange={(e) => setPrice(e.target.value.replace(/[^0-9]/g, ''))} placeholder="Price PC" inputMode="numeric" className="h-9" />
   410→            <Button type="submit" size="sm" disabled={!title.trim() || !price || list.isPending} className="shrink-0">
   411→              List
   412→            </Button>
   413→          </div>
   414→        </form>
   415→      </section>
   416→
   417→      <section aria-label="Listings" className="flex flex-col gap-2">
   418→        <h3 className="px-1 text-sm font-semibold">Open board</h3>
   419→        {(marketQ.data ?? []).map((l) => (
   420→          <motion.div
   421→            key={l.id}
   422→            layout
   423→            initial={{ opacity: 0, y: 4 }}
   424→            animate={{ opacity: 1, y: 0 }}
   425→            className={cn(
   426→              'rounded-xl border p-3',
   427→              l.status === 'sold'
   428→                ? 'border-zinc-200/80 bg-zinc-50/60 opacity-60 dark:border-white/10 dark:bg-transparent'
   429→                : 'border-zinc-200/80 bg-white dark:border-white/10 dark:bg-zinc-900/60',
   430→            )}
   431→          >
   432→            <div className="flex items-start justify-between gap-2">
   433→              <div className="min-w-0">
   434→                <p className="truncate text-sm font-semibold">{l.title}</p>
   435→                {l.description ? <p className="mt-0.5 line-clamp-2 text-xs text-zinc-500">{l.description}</p> : null}
   436→                <p className="mt-1 text-[11px] text-zinc-500">
   437→                  {l.status === 'sold'
   438→                    ? `sold to ${l.buyer?.username ? `@${l.buyer.username}` : l.buyer?.name ?? '—'}`
   439→                    : `by ${l.seller?.username ? `@${l.seller.username}` : l.seller?.name ?? '—'}`}
   440→                  {' · '}
   441→                  {new Date(l.createdAt).toLocaleDateString()}
   442→                </p>
   443→              </div>
   444→              <div className="flex shrink-0 flex-col items-end gap-1.5">
   445→                <p className="text-sm font-bold text-emerald-600">{l.price} PC</p>
   446→                {l.status === 'open' && !l.mine ? (
   447→                  <Button size="sm" variant="outline" className="h-7 px-2.5 text-xs" disabled={buy.isPending} onClick={() => buy.mutate(l.id)}>
   448→                    Buy
   449→                  </Button>
   450→                ) : null}
   451→                {l.mine && l.status === 'open' ? <span className="text-[10px] font-semibold text-zinc-400">yours</span> : null}
   452→              </div>
   453→            </div>
   454→          </motion.div>
   455→        ))}
   456→        {(marketQ.data?.length ?? 0) === 0 ? (
   457→          <p className="rounded-xl border border-dashed border-zinc-300 px-4 py-8 text-center text-[13px] text-zinc-500 dark:border-white/15">
   458→            The board is empty — be the first seller.
   459→          </p>
   460→        ) : null}
   461→      </section>
   462→    </div>
   463→  )
   464→}
