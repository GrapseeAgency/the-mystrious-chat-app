   158→/** Everything stateful lives here; unmounts with the sheet so nested layers never linger. */
   159→function GroupInfoInner({
   160→  conversationId,
   161→  meId,
   162→  onClose,
   163→}: {
   164→  conversationId: string
   165→  meId: string
   166→  onClose: () => void
   167→}) {
   168→  const queryClient = useQueryClient()
   169→  const [modal, setModal] = useState<SheetModalKind>(null)
   170→  const [profileUser, setProfileUser] = useState<AppUser | null>(null)
   171→
   172→  const detailQ = useQuery({
   173→    queryKey: ['conversation', conversationId],
   174→    queryFn: async (): Promise<ConversationDetail> => {
   175→      const res = await apiJson<ConversationResponse>(
   176→        `/api/conversations/${encodeURIComponent(conversationId)}?userId=${encodeURIComponent(meId)}`,
   177→      )
   178→      return res.conversation
   179→    },
   180→    staleTime: 10_000,
   181→    refetchInterval: 30_000,
   182→  })
   183→
   184→  const detail = detailQ.data ?? null
   185→  const isGroup = detail?.isGroup ?? false
   186→  const myRole: GroupRole = detail?.members.find((m) => m.id === meId)?.role ?? 'member'
   187→  const isAdmin = myRole === 'admin'
   188→  const adminCount = detail?.members.filter((m) => m.role === 'admin').length ?? 0
   189→
   190→  // Escape closes the topmost layer only (modal > profile drawer > sheet).
   191→  useEffect(() => {
   192→    const onKey = (event: KeyboardEvent) => {
   193→      if (event.key !== 'Escape') return
   194→      if (modal !== null || profileUser !== null) return // an inner layer owns this Escape
   195→      onClose()
   196→    }
   197→    window.addEventListener('keydown', onKey)
   198→    return () => window.removeEventListener('keydown', onKey)
   199→  }, [modal, profileUser, onClose])
   200→
   201→  const leaveGroup = useMutation({
   202→    mutationFn: async (): Promise<LeaveResponse> => {
   203→      return apiJson<LeaveResponse>(
   204→        `/api/conversations/${encodeURIComponent(conversationId)}/members`,
   205→        {
   206→          method: 'DELETE',
   207→          body: JSON.stringify({ requesterId: meId }),
   208→        },
   209→      )
   210→    },
   211→    onSuccess: (data) => {
   212→      const promoted = data.promotedUserId
   213→        ? (detail?.members.find((m) => m.id === data.promotedUserId)?.name ?? null)
   214→        : null
   215→      toast.success(
   216→        promoted ? `You left the group — ${promoted} is now an admin` : 'You left the group',
   217→      )
   218→      void queryClient.invalidateQueries({ queryKey: ['conversations', meId] })
   219→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   220→      setModal(null)
   221→      onClose()
   222→    },
   223→    onError: (error: Error) => {
   224→      toast.error(error.message || 'Could not leave the group')
   225→    },
   226→  })
   227→
   228→  return (
   229→    <>
   230→      {/* header */}
   231→      <header className="relative z-10 flex min-h-14 shrink-0 items-center gap-1 border-b border-zinc-200 bg-white/90 px-2 pt-[env(safe-area-inset-top)] backdrop-blur-md dark:border-zinc-800 dark:bg-zinc-900/90">
   232→        <Button
   233→          variant="ghost"
   234→          size="icon"
   235→          aria-label="Back to chat"
   236→          onClick={onClose}
   237→          className="size-10 shrink-0 rounded-full text-zinc-600 hover:bg-transparent hover:text-zinc-900 active:scale-95 dark:text-zinc-300 dark:hover:text-white"
   238→        >
   239→          <ChevronLeft className="size-6" aria-hidden />
   240→        </Button>
   241→        <p className="text-sm font-semibold tracking-tight text-zinc-900 dark:text-zinc-50">
   242→          {isGroup ? 'Group info' : 'Chat info'}
   243→        </p>
   244→      </header>
   245→
   246→      {/* body */}
   247→      {detailQ.isPending ? (
   248→        <InfoSkeleton isGroup={isGroup} />
   249→      ) : detailQ.isError || detail === null ? (
   250→        <div className="flex flex-1 flex-col items-center justify-center gap-3 px-8 text-center">
   251→          <UsersRound className="size-8 text-zinc-300 dark:text-zinc-600" aria-hidden />
   252→          <p className="text-sm font-medium text-zinc-500 dark:text-zinc-400">
   253→            Could not load the conversation info.
   254→          </p>
   255→          <Button
   256→            variant="outline"
   257→            size="sm"
   258→            onClick={() => void detailQ.refetch()}
   259→            className="h-9 rounded-full px-4 text-xs font-semibold"
   260→          >
   261→            Try again
   262→          </Button>
   263→        </div>
   264→      ) : isGroup ? (
   265→        <GroupInfoBody
   266→          detail={detail}
   267→          meId={meId}
   268→          isAdmin={isAdmin}
   269→          adminCount={adminCount}
   270→          onOpenModal={setModal}
   271→          onOpenProfile={setProfileUser}
   272→        />
   273→      ) : (
   274→        <DmInfoBody detail={detail} meId={meId} onOpenProfile={setProfileUser} />
   275→      )}
   276→
   277→      {/* nested modals — absolute inside the sheet so they layer above z-[70] */}
   278→      {detail !== null && isGroup ? (
   279→        <>
   280→          <RenameModal
   281→            open={modal === 'rename'}
   282→            onClose={() => setModal(null)}
   283→            conversationId={conversationId}
   284→            meId={meId}
   285→            currentName={detail.name ?? ''}
   286→          />
   287→          <AddMembersModal
   288→            open={modal === 'add'}
   289→            onClose={() => setModal(null)}
   290→            conversationId={conversationId}
   291→            meId={meId}
   292→            existingIds={detail.members.map((m) => m.id)}
   293→          />
   294→          <LeaveModal
   295→            open={modal === 'leave'}
   296→            onClose={() => setModal(null)}
   297→            groupName={detail.name?.trim() || 'this group'}
   298→            pending={leaveGroup.isPending}
   299→            onConfirm={() => leaveGroup.mutate()}
   300→          />
   301→        </>
   302→      ) : null}
   303→
   304→      {/* role/remove confirms + the member profile drawer (portaled) */}
   305→      <MemberActionHost
   306→        conversationId={conversationId}
   307→        meId={meId}
   308→        members={detail?.members ?? []}
   309→        modal={modal}
   310→        setModal={setModal}
   311→        profileUser={profileUser}
   312→        onCloseProfile={() => setProfileUser(null)}
   313→      />
   314→    </>
   315→  )
   316→}

   469→/**
   470→ * Management card for incoming webhooks. Everyone sees the list and can
   471→ * copy ingest URLs; creation + deletion are admin-only. Data is fully
   472→ * real: GET/POST /api/webhooks + DELETE /api/webhooks/[token].
   473→ */
   474→function WebhooksSection({
   475→  conversationId,
   476→  meId,
   477→  isAdmin,
   478→}: {
   479→  conversationId: string
   480→  meId: string
   481→  isAdmin: boolean
   482→}) {
   483→  const queryClient = useQueryClient()
   484→  const [createOpen, setCreateOpen] = useState(false)
   485→  const [deleteTarget, setDeleteTarget] = useState<WebhookItem | null>(null)
   486→
   487→  const webhooksQ = useQuery({
   488→    queryKey: ['webhooks', conversationId],
   489→    queryFn: async (): Promise<WebhookItem[]> => {
   490→      const res = await apiJson<WebhooksResponse>(
   491→        `/api/webhooks?conversationId=${encodeURIComponent(conversationId)}&requesterId=${encodeURIComponent(meId)}`,
   492→      )
   493→      return res.webhooks
   494→    },
   495→    staleTime: 5_000,
   496→  })
   497→
   498→  const copyUrl = async (webhook: WebhookItem) => {
   499→    try {
   500→      await navigator.clipboard.writeText(`${location.origin}/api/webhooks/${webhook.token}`)
   501→      toast.success('Webhook URL copied')
   502→    } catch {
   503→      toast.error('Could not copy the URL')
   504→    }
   505→  }
   506→
   507→  const deleteWebhook = useMutation({
   508→    mutationFn: async (token: string) => {
   509→      return apiJson<{ ok: boolean }>(
   510→        `/api/webhooks/${encodeURIComponent(token)}?requesterId=${encodeURIComponent(meId)}`,
   511→        { method: 'DELETE' },
   512→      )
   513→    },
   514→    onSuccess: () => {
   515→      void queryClient.invalidateQueries({ queryKey: ['webhooks', conversationId] })
   516→      toast.success('Webhook deleted')
   517→      setDeleteTarget(null)
   518→    },
   519→    onError: (error: Error) => {
   520→      toast.error(error.message || 'Could not delete the webhook')
   521→    },
   522→  })
   523→
   524→  const webhooks = webhooksQ.data ?? []
   525→
   526→  return (
   527→    <div className="px-4 pb-1">
   528→      <div className="mb-2 flex items-center justify-between px-1">
   529→        <h3 className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wider text-zinc-500 dark:text-zinc-400">
   530→          <Webhook className="size-3.5" aria-hidden />
   531→          Webhooks
   532→        </h3>
   533→        <span className="text-[11px] font-medium text-zinc-400 dark:text-zinc-500">
   534→          {webhooks.length}
   535→        </span>
   536→      </div>
   537→
   538→      {webhooksQ.isPending ? (
   539→        <div
   540→          className="space-y-2 rounded-2xl border border-zinc-200 bg-zinc-50/60 p-3 dark:border-zinc-800 dark:bg-zinc-800/40"
   541→          aria-busy="true"
   542→          aria-label="Loading webhooks"
   543→        >
   544→          {Array.from({ length: 2 }).map((_, i) => (
   545→            <div key={i} className="flex items-center gap-3">
   546→              <Skeleton className="size-8 rounded-full" />
   547→              <div className="flex-1 space-y-1.5">
   548→                <Skeleton className="h-3.5 w-1/2 rounded-md" />
   549→                <Skeleton className="h-3 w-1/3 rounded-md" />
   550→              </div>
   551→            </div>
   552→          ))}
   553→        </div>
   554→      ) : webhooksQ.isError ? (

   682→function WebhookCreateModal({
   683→  onClose,

   685→  meId,
   686→}: {
   687→  onClose: () => void
   688→  conversationId: string
   689→  meId: string
   690→}) {
   691→  const queryClient = useQueryClient()

   693→  const trimmed = name.trim()
   694→  const valid = trimmed.length > 0 && trimmed.length <= WEBHOOK_NAME_MAX
   695→
   696→  const create = useMutation({
   697→    mutationFn: async (): Promise<WebhookItem> => {
   698→      return apiJson<WebhookItem>(
   699→        '/api/webhooks',
   700→        jsonBody({ conversationId, name: trimmed, requesterId: meId }),

  1057→function SheetModal({
  1058→  open,
  1059→  onClose,
  1060→  label,
  1061→  children,
  1062→}: {
  1063→  open: boolean
  1064→  onClose: () => void
  1065→  label: string
  1066→  children: ReactNode
  1067→}) {
  1068→  useEffect(() => {
  1069→    if (!open) return
  1070→    const onKey = (event: KeyboardEvent) => {
  1071→      if (event.key === 'Escape') onClose()
  1072→    }
  1073→    window.addEventListener('keydown', onKey)
  1074→    return () => window.removeEventListener('keydown', onKey)
  1075→  }, [open, onClose])
  1076→
  1077→  return (
  1078→    <AnimatePresence>
  1079→      {open ? (
  1080→        <motion.div
  1081→          initial={{ opacity: 0 }}
  1082→          animate={{ opacity: 1 }}
  1083→          exit={{ opacity: 0 }}
  1084→          transition={{ duration: 0.16 }}
  1085→          className="absolute inset-0 z-20 flex items-center justify-center bg-black/50 p-5 backdrop-blur-[2px]"
  1086→          onClick={onClose}
