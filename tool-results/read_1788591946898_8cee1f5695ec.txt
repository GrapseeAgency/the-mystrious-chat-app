    66→/** Disappearing-message TTL → short label (mirrors the room header menu). */
    67→function ttlLabel(ttlSeconds: number): string {
    68→  if (ttlSeconds <= 0) return 'Off'
    69→  if (ttlSeconds === 86_400) return '24h'
    70→  if (ttlSeconds === 604_800) return '7d'
    71→  if (ttlSeconds === 2_592_000) return '30d'
    72→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))}d`
    73→}

    75→export function RoomInfoPage({

    89→  const isGroup = detail?.isGroup ?? false

   108→  }, [loadedMessages])
   109→
   110→  const onlineCount = useMemo(
   111→    () => (detail?.members ?? []).filter((m) => onlineIds.has(m.id)).length,

   113→  )
   114→
   115→  /** admins first, then name — stable view order */

   121→      return a.name.localeCompare(b.name)
   122→    })
   123→    return list
   124→  }, [detail])
   125→
   126→  const adminCount = useMemo(
   127→    () => members.filter((m) => m.role === 'admin').length,
   128→    [members],
   129→  )
   130→
   131→  /** directory ids already in the group — the add-members view subtracts them */
   132→  const existingIds = useMemo(
   133→    () => new Set((detail?.members ?? []).map((m) => m.id)),
   134→    [detail],
   135→  )
   136→
   137→  // ── member-list search (only surfaced past 8 members) ─────
   138→
   139→  const [memberFilter, setMemberFilter] = useState('')
   140→  const memberQuery = memberFilter.trim().toLowerCase()
   141→  const visibleMembers = useMemo(() => {
   142→    if (members.length <= 8 || memberQuery.length === 0) return members

   144→      (m) =>
   145→        m.name.toLowerCase().includes(memberQuery) ||

   147→    )
   148→  }, [members, memberQuery])
   149→
   150→  // ── real actions ─────────────────────────────────────────
   151→
   152→  const muteMutation = useMutation({
   153→    mutationFn: async (until: '8h' | '1w' | 'always' | null) => {
   154→      return apiJson<{ ok: boolean; mutedUntil: string | null }>(
   155→        `/api/conversations/${encodeURIComponent(conversationId)}/mute`,
   156→        jsonBody({ userId: me.id, until }),
   157→      )
   158→    },
   159→    onSuccess: (data) => {
   160→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   161→        old ? { ...old, myMutedUntil: data.mutedUntil } : old,
   162→      )

   164→      haptic(12)
   165→    },
   166→    onError: () => toast.error('Could not update the mute'),
   167→  })
   168→
   169→  const inviteMutation = useMutation({
   170→    mutationFn: async () => {
   171→      return apiJson<{ inviteCode: string }>(
   172→        `/api/conversations/${encodeURIComponent(conversationId)}/invite`,
   173→        jsonBody({ requesterId: me.id, regenerate: false }),
   174→      )
   175→    },
   176→    onSuccess: (res) => {
   177→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   178→        old ? { ...old, inviteCode: res.inviteCode } : old,
   179→      )
   180→      void copyInvite(res.inviteCode)
   181→    },
   182→    onError: (error) =>
   183→      toast.error(error instanceof Error ? error.message : 'Could not create the invite link'),
   184→  })
   185→
   186→  // POST /members/{userId} is DELETE-only (PATCH = role change) — the R27
   187→  // draft sent method:POST via jsonBody and got a silent 405. Fixed here.
   188→  const removeMutation = useMutation({
   189→    mutationFn: async (userId: string) => {
   190→      return apiJson<{ ok: boolean }>(
   191→        `/api/conversations/${encodeURIComponent(conversationId)}/members/${encodeURIComponent(userId)}`,
   192→        { method: 'DELETE', body: JSON.stringify({ requesterId: me.id }) },
   193→      )
   194→    },
   195→    onMutate: async (userId: string) => {

   198→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   199→        old ? { ...old, members: old.members.filter((m) => m.id !== userId) } : old,
   200→      )
   201→      return { prev }
   202→    },
   203→    onSuccess: () => {
   204→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   205→      void queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   206→      toast.success('Removed from the group')
   207→      haptic(14)
   208→    },
   209→    onError: (error, _userId, ctx) => {
   210→      if (ctx?.prev) queryClient.setQueryData(['conversation', conversationId], ctx.prev)
   211→      toast.error(error instanceof Error ? error.message : 'Could not remove the member')
   212→    },
   213→  })
   214→
   215→  // PATCH /members/{userId} { requesterId, action: 'promote' | 'demote' }
   216→  // Admin-only server-side; demote 400s when the target is the last admin.

   218→    mutationFn: async ({ userId, action }: { userId: string; action: 'promote' | 'demote' }) => {

   220→        `/api/conversations/${encodeURIComponent(conversationId)}/members/${encodeURIComponent(userId)}`,

   228→        old
   229→          ? {
   230→              ...old,
   231→              members: old.members.map((m) =>
   232→                m.id === vars.userId
   233→                  ? { ...m, role: (vars.action === 'promote' ? 'admin' : 'member') as GroupRole }
   234→                  : m,
   235→              ),
   236→            }
   237→          : old,
   238→      )
   239→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   240→      void queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })

   242→      haptic(12)
   243→    },
   244→    onError: (error) =>
   245→      toast.error(error instanceof Error ? error.message : 'Could not update the role'),
   246→  })
   247→
   248→  const copyInvite = async (code: string) => {
   249→    const link = `${window.location.origin}/?join=${code}`
   250→    try {
   251→      await navigator.clipboard.writeText(link)
   252→      toast.success('Invite link copied to clipboard')
   253→    } catch {
   254→      toast.error('Could not copy the link')
   255→    }
   256→  }
   257→
   258→  // two-tap confirm for destructive row actions (remove / dismiss-admin)
   259→  type ArmedConfirm = { id: string; kind: 'remove' | 'demote' }

   264→    setConfirm({ id: userId, kind })
   265→    haptic(8)
   266→    confirmTimerRef.current = setTimeout(() => {
   267→      confirmTimerRef.current = null
   268→      setConfirm(null)
   269→    }, 2600)
   270→  }
   271→  const tapConfirm = (kind: ArmedConfirm['kind'], userId: string, run: () => void) => {
   272→    if (confirm?.id === userId && confirm.kind === kind) {
   273→      if (confirmTimerRef.current !== null) {
   274→        clearTimeout(confirmTimerRef.current)
   275→        confirmTimerRef.current = null
   276→      }
   277→      setConfirm(null)
   278→      run()
   279→    } else {
   280→      armConfirm(kind, userId)
   281→    }
   282→  }
   283→  // remove keeps the R27 two-tap shape — armRemove / tapRemove
   284→  const armRemove = (userId: string) => armConfirm('remove', userId)
   285→  const tapRemove = (userId: string) =>
   286→    tapConfirm('remove', userId, () => removeMutation.mutate(userId))
   287→  const tapDemote = (userId: string) =>
   288→    tapConfirm('demote', userId, () => roleMutation.mutate({ userId, action: 'demote' }))
   289→
   290→  // R28-b: full-screen add-members sub-view INSIDE the info page
   291→  const [addOpen, setAddOpen] = useState(false)
   292→
   293→  const anim = {
   294→    initial: reducedMotion ? false : { opacity: 0, y: 10 },
   295→    animate: { opacity: 1, y: 0 },
   296→  }
   297→
   298→  return (
   299→    <motion.div
   300→      initial={reducedMotion ? false : { y: '100%' }}
   301→      animate={{ y: 0 }}
   302→      exit={{ y: '100%' }}
   303→      transition={spring.soft}
   304→      role="dialog"
   305→      aria-label={isGroup ? `About ${title}` : `About ${title}`}
   306→      className="absolute inset-0 z-[55] flex flex-col bg-zinc-100/85 backdrop-blur-2xl dark:bg-black/70"
   307→    >
   308→      {/* header */}
   309→      <div className="flex min-h-14 shrink-0 items-center gap-1.5 px-2 pt-[env(safe-area-inset-top)]">
   310→        <button
   311→          type="button"
   312→          aria-label="Back to conversation"
   313→          onClick={() => {
   314→            haptic(6)
   315→            onClose()
   316→          }}
   317→          className="glass-pill flex size-10 shrink-0 items-center justify-center text-zinc-600 outline-none transition-transform active:scale-90 dark:text-zinc-300"
   318→        >
   319→          <ChevronLeft className="size-6" aria-hidden />
   320→        </button>
   321→        <p className="text-sm font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   322→          {isGroup ? 'Group info' : 'Chat info'}
   323→        </p>
   324→      </div>
   325→
   326→      <div className="pulse-scroll min-h-0 flex-1 overflow-y-auto overscroll-contain px-3 pt-1 pb-8">
   327→        {/* hero */}
   328→        <motion.div
   329→          {...anim}
   330→          transition={spring.soft}
   331→          className="glass-deep glass-sheen relative overflow-hidden rounded-3xl p-5"
   332→        >
   333→          <div
   334→            aria-hidden
   335→            className="pointer-events-none absolute -top-16 -right-14 size-44 rounded-full opacity-25 blur-2xl"
   336→            style={{
   337→              backgroundImage: isGroup
   338→                ? groupGradientFor(conversationId)
   339→                : gradientFor(other?.color ?? 'emerald'),
   340→            }}
   341→          />
   342→          <div className="flex items-center gap-4">
   343→            {isGroup ? (
   344→              <GroupAvatar title={title} id={conversationId} size={68} />
   345→            ) : (
   346→              <UserAvatar
   347→                name={other?.name ?? title}
   348→                color={other?.color}
   349→                avatar={other?.avatar}
   350→                size={68}
   351→                showPresence
   352→                online={other ? onlineIds.has(other.id) : false}
   353→              />
   354→            )}
   355→            <div className="min-w-0 flex-1">
   356→              <div className="truncate text-lg font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   357→                {detail === undefined ? <Skeleton className="h-6 w-36 rounded-lg" /> : title}
   358→              </div>
   359→              {detail === undefined ? (
   360→                <Skeleton className="mt-1.5 h-4 w-24 rounded-md" />
   361→              ) : isGroup ? (
   362→                <p className="mt-0.5 text-xs font-medium text-zinc-500 dark:text-zinc-400">
   363→                  {members.length} {members.length === 1 ? 'member' : 'members'} · {onlineCount} online
   364→                </p>
   365→              ) : (
   366→                <p className="mt-0.5 text-xs font-medium text-zinc-500 dark:text-zinc-400">
   367→                  {other ? (onlineIds.has(other.id) ? 'Online now' : 'Offline') : 'Direct chat'}
   368→                </p>
   369→              )}
   370→              <div className="mt-2 flex flex-wrap gap-1.5">
   371→                {isGroup && (detail?.broadcastMode ?? false) ? (
   372→                  <span className="flex items-center gap-1 rounded-full bg-amber-500/10 px-2 py-0.5 text-[10px] font-bold text-amber-600 dark:text-amber-400">
   373→                    <Megaphone className="size-3" aria-hidden />

   387→
   388→        {/* quick stats — real counts over the loaded window */}
   389→        <motion.div {...anim} transition={{ ...spring.soft, delay: stagger(1) }} className="mt-3 grid grid-cols-3 gap-2">
   390→          {[
   391→            { icon: ImageIcon, label: 'Photos', value: stats.media },
   392→            { icon: Link2, label: 'Links', value: stats.links },
   393→            { icon: Pin, label: 'Pinned', value: pinnedCount },

   396→              key={tile.label}
   397→              className="glass-deep glass-sheen flex flex-col items-center gap-0.5 rounded-2xl px-2 py-3"

   401→                {tile.value}
   402→              </span>
   403→              <span className="text-[10px] font-semibold tracking-wide text-zinc-400 uppercase dark:text-zinc-500">
   404→                {tile.label}
   405→              </span>
   406→            </div>
   407→          ))}
   408→        </motion.div>
   409→        {historyPartial ? (
   410→          <p className="mt-1.5 px-1 text-center text-[10px] text-zinc-400 dark:text-zinc-500">
   411→            Counted from loaded history — older messages exist.
   412→          </p>
   413→        ) : null}
   414→
   415→        {/* actions */}
   416→        <motion.div
   417→          {...anim}
   418→          transition={{ ...spring.soft, delay: stagger(2) }}

   420→        >
   421→          {/* mute — real per-user watermark API */}
   422→          <div className="glass-row-hover flex items-center gap-3 rounded-2xl px-3 py-2.5">

   431→              </p>
   432→              {isMuted && detail?.myMutedUntil ? (
   433→                <p className="text-[11px] text-zinc-400 dark:text-zinc-500">
   434→                  until {formatListStamp(detail.myMutedUntil)}
   435→                </p>
   436→              ) : null}
   437→            </div>
   438→            {isMuted ? (
   439→              <button
   440→                type="button"
   441→                disabled={muteMutation.isPending}
   442→                onClick={() => muteMutation.mutate(null)}
   443→                className="glass-pill h-8 shrink-0 px-3 text-xs font-bold text-emerald-600 outline-none transition-transform active:scale-95 disabled:opacity-50 dark:text-emerald-400"
   444→              >
   445→                Unmute
   446→              </button>

   449→                {(['8h', '1w', 'always'] as const).map((preset) => (
   450→                  <button
   451→                    key={preset}
   452→                    type="button"
   453→                    disabled={muteMutation.isPending}
   454→                    onClick={() => muteMutation.mutate(preset)}
   455→                    className="h-8 rounded-full bg-zinc-900/[0.05] px-2.5 text-[11px] font-bold text-zinc-600 outline-none transition-transform hover:bg-emerald-500/15 hover:text-emerald-700 active:scale-95 disabled:opacity-50 dark:bg-white/[0.07] dark:text-zinc-300 dark:hover:text-emerald-400"
   456→                  >
   457→                    {preset === 'always' ? 'Always' : preset}
   458→                  </button>
   459→                ))}
   460→              </div>
   461→            )}
   462→          </div>
   463→
   464→          {/* disappearing TTL — display-only (changing lives in the room menu) */}
   465→          <div className="glass-row-hover flex items-center gap-3 rounded-2xl px-3 py-2.5">
   466→            <Timer className="size-4 shrink-0 text-zinc-400" aria-hidden />
   467→            <p className="min-w-0 flex-1 truncate text-sm font-medium text-zinc-800 dark:text-zinc-100">
   468→              Disappearing messages
   469→            </p>
   470→            <span
   471→              className={cn(
   472→                'shrink-0 rounded-full px-2 py-0.5 text-[10px] font-bold',
   473→                (detail?.ttlSeconds ?? 0) > 0
   474→                  ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   475→                  : 'bg-zinc-900/[0.05] text-zinc-500 dark:bg-white/[0.07] dark:text-zinc-400',
   476→              )}
   477→            >
   478→              {detail === undefined ? '…' : ttlLabel(detail.ttlSeconds)}
   479→            </span>
   480→          </div>
   481→
   482→          {/* invite link — real /invite API, groups + admin only (hidden honestly) */}
   483→          {isGroup && isAdmin ? (
   484→            <div className="glass-row-hover flex items-center gap-3 rounded-2xl px-3 py-2.5">
   485→              <Users className="size-4 shrink-0 text-zinc-400" aria-hidden />
   486→              <div className="min-w-0 flex-1">

   488→                  Invite link
   489→                </p>
   490→                <p className="truncate text-[11px] text-zinc-400 dark:text-zinc-500">
   491→                  {detail?.inviteCode ? `Code ${detail.inviteCode}` : 'No active link yet'}
   492→                </p>

   494→              <button
