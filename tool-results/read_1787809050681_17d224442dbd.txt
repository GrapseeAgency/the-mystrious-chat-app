   232→/**
   233→ * Snippet with the first match highlighted — clips a ≤64-char window
   234→ * around the hit so long messages stay one tidy line.
   235→ */
   236→function SearchSnippet({ content, query }: { content: string; query: string }) {
   237→  const lower = content.toLowerCase()
   238→  const q = query.toLowerCase()
   239→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   240→  let from = 0
   241→  let clippedHead = false
   242→  if (idx > 28) {
   243→    from = idx - 24
   244→    clippedHead = true
   245→  }
   246→  const end = Math.min(content.length, idx + q.length + 28)
   247→  const clippedTail = end < content.length
   248→  const body = content.slice(from, end)
   249→  const localIdx = idx - from
   250→  return (
   251→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   252→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   253→      {idx >= 0 ? (
   254→        <>
   255→          {body.slice(0, localIdx)}
   256→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   257→            {body.slice(localIdx, localIdx + q.length)}
   258→          </mark>
   259→          {body.slice(localIdx + q.length)}
   260→        </>
   261→      ) : (
   262→        body
   263→      )}
   264→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   265→    </p>
   266→  )
   267→}

   345→export function ChatsTab({
   346→  me,
   347→  onOpenConversation,
   348→  onOpenContacts,
   349→  onRequestNewChat,
   350→  onGoProfile,
   351→}: {
   352→  me: AppUser
   353→  onOpenConversation: (conversationId: string, unreadAnchorMs: number | null, jumpMessageId?: string) => void
   354→  onOpenContacts: () => void
   355→  onRequestNewChat: () => void
   356→  onGoProfile: () => void
   357→}) {
   358→  const realtime = usePulseRealtime()
   359→  const typersIn = realtime.typersIn
   360→  const onlineIds = realtime.onlineIds
   361→  const [searching, setSearching] = useState(false)
   362→  const [searchQuery, setSearchQuery] = useState('')
   363→  /** debounced mirror of searchQuery feeding the server message search */
   364→  const [deferredQuery, setDeferredQuery] = useState('')
   365→  const deferredTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   366→
   367→  const conversations = useQuery({
   368→    queryKey: ['conversations', me.id],
   369→    queryFn: async (): Promise<ConversationSummary[]> => {
   370→      const res = await apiJson<ConversationsResponse>(
   371→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
   372→      )
   373→      return res.conversations
   374→    },
   375→    refetchInterval: 6_000,
   376→  })
   377→
   378→  const rows = useMemo(() => {
   379→    return (conversations.data ?? []).map((conv) => {
   380→      const previewInfo = conversationPreview(conv, me.id)
   381→      const other = conv.isGroup ? null : otherMemberOf(conv, me.id)
   382→      const groupName =
   383→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   384→      const displayName = conversationDisplayName(conv, me.id)
   385→      return {
   386→        conv,
   387→        props: {
   388→          id: conv.id,
   389→          isGroup: conv.isGroup,
   390→          name: displayName,
   391→          time: conv.lastMessage
   392→            ? formatListStamp(conv.lastMessage.createdAt)
   393→            : formatListStamp(conv.updatedAt),
   394→          preview: previewInfo.text,
   395→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   396→          previewDeleted: previewInfo.deleted,
   397→          unreadCount: conv.unreadCount,
   398→          dmName: other?.name ?? null,
   399→          dmColor: other?.color ?? 'emerald',
   400→          groupTitle: groupName,
   401→          online: !conv.isGroup && other !== null && onlineIds.has(other.id),
   402→          pinned: conv.pinnedAt !== null,
   403→          muted: conv.mutedUntil !== null && Date.parse(conv.mutedUntil) > Date.now(),
   404→          typing: typersIn(conv.id, me.id).length > 0,
   405→        },
   406→      }
   407→    })
   408→  }, [conversations.data, me.id, onlineIds, typersIn])
   409→
   410→  const filteredRows = useMemo(() => {
   411→    const q = searchQuery.trim().toLowerCase()
   412→    if (!q) return rows
   413→    return rows.filter(({ conv, props }) => {
   414→      if (props.name.toLowerCase().includes(q)) return true
   415→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   416→      return false
   417→    })
   418→  }, [rows, searchQuery])
   419→
   420→  /** Freeze "where was I" from the list summary AT TAP TIME (pre-read watermark). */
   421→  const handlePress = useCallback(
   422→    (conv: ConversationSummary) => {
   423→      const mine = conv.members.find((m) => m.id === me.id) as
   424→        | (AppUser & { lastReadAt?: string })
   425→        | undefined
   426→      const wm = mine?.lastReadAt ? Date.parse(mine.lastReadAt) : Number.NaN
   427→      const anchor = conv.unreadCount > 0 && !Number.isNaN(wm) ? wm : null
   428→      onOpenConversation(conv.id, anchor)
   429→    },
   430→    [onOpenConversation, me.id],
   431→  )
   432→
   433→  const closeSearch = useCallback(() => {
   434→    if (deferredTimerRef.current !== null) {
   435→      clearTimeout(deferredTimerRef.current)
   436→      deferredTimerRef.current = null
   437→    }
   438→    setSearching(false)
   439→    setSearchQuery('')
   440→    setDeferredQuery('')
   441→  }, [])
   442→
   443→  /** typing in the search field: instant local filter + 250ms-debounced server search */
   444→  const handleSearchInput = useCallback((value: string) => {
   445→    setSearchQuery(value)
   446→    if (deferredTimerRef.current !== null) clearTimeout(deferredTimerRef.current)
   447→    const v = value.trim()
   448→    deferredTimerRef.current = setTimeout(() => {
   449→      deferredTimerRef.current = null
   450→      setDeferredQuery(v)
   451→    }, 250)
   452→  }, [])
   453→
   454→  const serverSearch = useQuery({
   455→    queryKey: ['global-search', me.id, deferredQuery],
   456→    enabled: searching && deferredQuery.length >= 2,
   457→    staleTime: 15_000,
   458→    queryFn: async (): Promise<SearchResponse> =>
   459→      apiJson<SearchResponse>(
   460→        `/api/search?userId=${encodeURIComponent(me.id)}&q=${encodeURIComponent(deferredQuery)}`,
   461→      ),
   462→  })
   463→  const serverHits = searching && deferredQuery.length >= 2 ? (serverSearch.data?.messages ?? []) : []
   464→
   465→  // ── row long-press action sheet (pin/unpin) ───────────────
   466→  const [sheetConv, setSheetConv] = useState<ConversationSummary | null>(null)
   467→  const queryClient = useQueryClient()
   468→
   469→  const togglePin = useMutation({
   470→    mutationFn: async (conv: ConversationSummary) => {
   471→      return apiJson<{ ok: boolean; pinned: boolean }>(
   472→        `/api/conversations/${encodeURIComponent(conv.id)}/pin`,
   473→        {
   474→          method: 'PATCH',
   475→          headers: { 'Content-Type': 'application/json' },
   476→          body: JSON.stringify({ userId: me.id }),
   477→        },
   478→      )
   479→    },
   480→    onSuccess: (data) => {
   481→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   482→      toast.success(data.pinned ? 'Pinned to top' : 'Unpinned')
   483→      setSheetConv(null)
   484→    },
   485→    onError: () => {
   486→      toast.error('Could not update the pin')
   487→    },
   488→  })
   489→
   490→  /** per-user notification mute — '8h' | '1w' | 'always' | null */
   491→  const toggleMute = useMutation({
   492→    mutationFn: async ({ conv, until }: { conv: ConversationSummary; until: '8h' | '1w' | 'always' | null }) => {
   493→      return apiJson<{ ok: boolean; mutedUntil: string | null }>(
   494→        `/api/conversations/${encodeURIComponent(conv.id)}/mute`,
   495→        {
   496→          method: 'PATCH',
   497→          headers: { 'Content-Type': 'application/json' },
   498→          body: JSON.stringify({ userId: me.id, until }),
   499→        },
   500→      )
   501→    },
   502→    onSuccess: (data) => {
   503→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   504→      toast.success(
   505→        data.mutedUntil === null
   506→          ? 'Notifications unmuted'
   507→          : data.mutedUntil !== null && Date.parse(data.mutedUntil) - Date.now() > 20 * 365 * 24 * 3600 * 1000
   508→            ? 'Muted — always'
   509→            : `Muted until ${formatListStamp(data.mutedUntil as string)}`,
   510→      )
   511→      setSheetConv(null)
   512→    },
   513→    onError: () => {
   514→      toast.error('Could not update the mute')
   515→    },
   516→  })
   517→
   518→  const openSheetFor = useCallback((conv: ConversationSummary) => setSheetConv(conv), [])
   519→
   520→  const sheetMuted =
   521→    sheetConv !== null &&
   522→    sheetConv.mutedUntil !== null &&
   523→    Date.parse(sheetConv.mutedUntil) > Date.now()
   524→
   525→  const data = conversations.data ?? []
   526→
   527→  return (
   528→    <div className="absolute inset-0 flex flex-col bg-white dark:bg-zinc-900">
   529→      {/* header */}

   735→              <button

   737→                role="menuitem"
   738→                onClick={() => setSheetConv(null)}
   739→                className="flex w-full items-center justify-center rounded-2xl px-3 py-3 text-left text-sm font-medium text-zinc-500 outline-none transition-colors hover:bg-zinc-100 dark:text-zinc-400 dark:hover:bg-zinc-800"

   742→              </button>

   744→          ) : null}
   745→        </DrawerContent>
