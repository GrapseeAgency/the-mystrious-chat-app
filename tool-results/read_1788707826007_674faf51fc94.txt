   496→/**
   497→ * Glass-pill segmented picker (the locked .glass-pill chip row) with a
   498→ * sliding active fill. Every onChange writes straight to a real store.
   499→ */
   500→function PillPicker<T extends string>({
   501→  layoutId,
   502→  value,
   503→  onChange,
   504→  options,
   505→  ariaLabel,
   506→}: {
   507→  layoutId: string
   508→  value: T
   509→  onChange: (v: T) => void
   510→  options: Array<{ value: T; label: string; Icon?: LucideIcon }>
   511→  ariaLabel: string
   512→}) {
   513→  const reduced = useReducedMotion()
   514→  return (
   515→    <div className="glass-pill flex w-full items-center gap-1 p-1" role="radiogroup" aria-label={ariaLabel}>
   516→      {options.map((o) => {
   517→        const selected = value === o.value
   518→        return (
   519→          <button
   520→            key={o.value}
   521→            type="button"
   522→            role="radio"
   523→            aria-checked={selected}
   524→            onClick={() => {
   525→              haptic(8)
   526→              onChange(o.value)
   527→            }}
   528→            className={cn(
   529→              'relative flex h-9 min-w-0 flex-1 items-center justify-center rounded-full px-2 text-[12.5px] font-semibold outline-none transition-colors duration-150 focus-visible:ring-2 focus-visible:ring-emerald-500/60',
   530→              selected
   531→                ? 'text-zinc-900 dark:text-white'
   532→                : 'text-zinc-500 hover:text-zinc-800 dark:text-zinc-400 dark:hover:text-zinc-100',
   533→            )}
   534→          >
   535→            {selected ? (
   536→              <motion.span
   537→                layoutId={layoutId}
   538→                transition={reduced ? { duration: 0 } : spring.snappy}
   539→                className="absolute inset-0 rounded-full bg-white shadow-sm dark:bg-zinc-600/70"
   540→                aria-hidden
   541→              />
   542→            ) : null}
   543→            <span className="relative z-10 flex min-w-0 items-center gap-1.5">
   544→              {o.Icon ? <o.Icon className="size-3.5 shrink-0" aria-hidden /> : null}
   545→              <span className="truncate">{o.label}</span>
   546→            </span>
   547→          </button>
   548→        )
   549→      })}
   550→    </div>
   551→  )
   552→}
