   219→const COLUMNS: Array<{ id: HubTaskItem['status']; label: string }> = [

   222→  { id: 'done', label: 'Done' },

   225→function TasksPanel({ me }: { me: AppUser }) {
   226→  const qc = useQueryClient()
   227→  const [title, setTitle] = useState('')
   228→
   229→  const tasksQ = useQuery({
   230→    queryKey: ['hub-tasks', me.id],
   231→    queryFn: async () => {
   232→      const res = await apiJson<{ tasks: HubTaskItem[] }>(`/api/hub/tasks?userId=${encodeURIComponent(me.id)}`)
   233→      return res.tasks
   234→    },
   235→  })
   236→  const invalidate = () => void qc.invalidateQueries({ queryKey: ['hub-tasks', me.id] })
   237→
   238→  const add = useMutation({
   239→    mutationFn: (t: string) =>
   240→      apiJson<{ task: HubTaskItem }>('/api/hub/tasks', {
   241→        method: 'POST',
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify({ userId: me.id, title: t }),
   244→      }),
   245→    onSuccess: () => {
   246→      setTitle('')
   247→      invalidate()
   248→    },
   249→    onError: (err: Error) => toast.error(err.message),
   250→  })
   251→
   252→  const move = useMutation({
   253→    mutationFn: (input: { id: string; status: HubTaskItem['status'] }) =>
   254→      apiJson(`/api/hub/tasks/${input.id}`, {
   255→        method: 'PATCH',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({ userId: me.id, status: input.status }),
   258→      }),
   259→    onSettled: invalidate,
   260→    onError: (err: Error) => toast.error(err.message),
   261→  })
   262→
   263→  const remove = useMutation({
   264→    mutationFn: (id: string) => apiJson(`/api/hub/tasks/${id}?userId=${encodeURIComponent(me.id)}`, { method: 'DELETE' }),
   265→    onSettled: invalidate,
   266→    onError: (err: Error) => toast.error(err.message),
   267→  })
   268→
   269→  const byStatus = useMemo(() => {
   270→    const map: Record<HubTaskItem['status'], HubTaskItem[]> = { todo: [], doing: [], done: [] }
   271→    for (const t of tasksQ.data ?? []) map[t.status]?.push(t)
   272→    return map
   273→  }, [tasksQ.data])
   274→
   275→  return (
   276→    <div className="flex flex-col gap-3">
   277→      <form
   278→        className="flex gap-2"
   279→        onSubmit={(e) => {
   280→          e.preventDefault()
   281→          if (title.trim()) add.mutate(title.trim())
   282→        }}
   283→      >
   284→        <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="New task…" className="h-9" />
   285→        <Button type="submit" size="sm" disabled={!title.trim() || add.isPending} className="shrink-0">
   286→          <Check className="size-4" />
   287→        </Button>
   288→      </form>
   289→
   290→      <div className="grid grid-cols-3 gap-2">
   291→        {COLUMNS.map((col) => (
   292→          <div key={col.id} className="rounded-xl bg-zinc-100/70 p-2 dark:bg-white/5">
   293→            <p className="mb-2 px-1 text-[11px] font-bold uppercase tracking-wide text-zinc-500">
   294→              {col.label} · {byStatus[col.id].length}
   295→            </p>
   296→            <ul className="flex flex-col gap-1.5">
   297→              {byStatus[col.id].map((t) => (
   298→                <motion.li
   299→                  key={t.id}
   300→                  layout
   301→                  initial={{ opacity: 0, scale: 0.95 }}
   302→                  animate={{ opacity: 1, scale: 1 }}
   303→                  className="rounded-lg border border-zinc-200 bg-white p-2 shadow-sm dark:border-white/10 dark:bg-zinc-900/80"
   304→                >
   305→                  <p className={cn('text-[12px] font-medium leading-snug', t.status === 'done' && 'text-zinc-400 line-through')}>
   306→                    {t.title}
   307→                  </p>
   308→                  <div className="mt-1.5 flex items-center justify-between">
   309→                    <button
   310→                      type="button"
   311→                      aria-label="Advance status"
   312→                      className="text-[10px] font-semibold text-emerald-600 hover:underline"
   313→                      onClick={() =>
   314→                        move.mutate({ id: t.id, status: t.status === 'todo' ? 'doing' : t.status === 'doing' ? 'done' : 'todo' })
   315→                      }
   316→                    >
   317→                      {t.status === 'done' ? '↺ reopen' : '→ next'}
   318→                    </button>
   319→                    <button
   320→                      type="button"
   321→                      aria-label="Delete task"
   322→                      className="text-[10px] text-zinc-400 hover:text-rose-500"
   323→                      onClick={() => remove.mutate(t.id)}
   324→                    >
   325→                      <X className="size-4" aria-hidden />
   326→                    </button>
   327→                  </div>
   328→                </motion.li>
   329→              ))}
   330→              {byStatus[col.id].length === 0 ? (
   331→                <li className="rounded-lg border border-dashed border-zinc-300 px-2 py-3 text-center text-[10px] text-zinc-400 dark:border-white/15">
   332→                  empty
   333→                </li>
   334→              ) : null}
   335→            </ul>
   336→          </div>
   337→        ))}
   338→      </div>
   339→    </div>
   340→  )
   341→}

   392→        method: 'POST',
   393→        headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({ userId: me.id }),

   409→          className="mt-3 flex flex-col gap-2"
   410→          onSubmit={(e) => {
   411→            e.preventDefault()
   412→            if (title.trim() && price) list.mutate()
   413→          }}
   414→        >
   415→          <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Title (what are you selling?)" className="h-9" />
   416→          <Input value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Description (optional)" className="h-9" />
   417→          <div className="flex gap-2">
   418→            <Input value={price} onChange={(e) => setPrice(e.target.value.replace(/[^0-9]/g, ''))} placeholder="Price PC" inputMode="numeric" className="h-9" />
   419→            <Button type="submit" size="sm" disabled={!title.trim() || !price || list.isPending} className="shrink-0">
   420→              List
   421→            </Button>
   422→          </div>
   423→        </form>
   424→      </section>
   425→
   426→      <section aria-label="Listings" className="flex flex-col gap-2">

   436→              l.status === 'sold'

   439→            )}
   440→          >
   441→            <div className="flex items-start justify-between gap-2">

   443→                <p className="truncate text-sm font-semibold">{l.title}</p>
   444→                {l.description ? <p className="mt-0.5 line-clamp-2 text-xs text-zinc-500">{l.description}</p> : null}
   445→                <p className="mt-1 text-[11px] text-zinc-500">
   446→                  {l.status === 'sold'
   447→                    ? `sold to ${l.buyer?.username ? `@${l.buyer.username}` : l.buyer?.name ?? '—'}`
   448→                    : `by ${l.seller?.username ? `@${l.seller.username}` : l.seller?.name ?? '—'}`}
   449→                  {' · '}
   450→                  {new Date(l.createdAt).toLocaleDateString()}
   451→                </p>
   452→              </div>
   453→              <div className="flex shrink-0 flex-col items-end gap-1.5">
   454→                <p className="text-sm font-bold text-emerald-600">{l.price} PC</p>
   455→                {l.status === 'open' && !l.mine ? (
   456→                  <Button size="sm" variant="outline" className="h-7 px-2.5 text-xs" disabled={buy.isPending} onClick={() => buy.mutate(l.id)}>
   457→                    Buy
   458→                  </Button>
   459→                ) : null}
   460→                {l.mine && l.status === 'open' ? <span className="text-[10px] font-semibold text-zinc-400">yours</span> : null}
   461→              </div>
   462→            </div>
   463→          </motion.div>
   464→        ))}
   465→        {(marketQ.data?.length ?? 0) === 0 ? (
   466→          <p className="rounded-xl border border-dashed border-zinc-300 px-4 py-8 text-center text-[13px] text-zinc-500 dark:border-white/15">
   467→            The board is empty — be the first seller.
   468→          </p>
   469→        ) : null}
   470→      </section>
   471→    </div>
   472→  )
   473→}
