     1→// ─────────────────────────────────────────────────────────────
     2→// Pulse — 1:1 voice + video calls (R33-a, WhatsApp/Signal-style).
     3→//
     4→// Two exports wire the whole feature into chat-room:
     5→//   1. useCallSession({...}) — the call state machine: WebRTC
     6→//      (getUserMedia + RTCPeerConnection, Google STUN), signaling over the
     7→//      shared socket channel (call:offer/answer/ice/reject/cancel/hangup),
     8→//      ring timeout handling and the SINGLE-WRITER call log rule
     9→//      (the CALLER's client POSTs every terminal row to /api/calls).
    10→//   2. <CallOverlay session={...} /> — the full-screen glass call
    11→//      experience: outgoing ring, incoming ring, active call (remote video
    12→//      full-bleed or big avatar, live duration, mic/camera toggles) and a
    13→//      brief ended summary card. Honest permission-error card when the mic
    14→//      is denied; graceful voice fallback when no camera exists.
    15→//
    16→// Mounted once per DM room — incoming calls surface while that room is open
    17→// (this wave: overlay lives at chat-room level, not app level).

    96→/** 65 → "1:05", 3675 → "1:01:15". */
    97→export function formatCallDuration(totalSec: number): string {
    98→  const sec = Math.max(0, Math.floor(totalSec))
    99→  const h = Math.floor(sec / 3600)
   100→  const m = Math.floor((sec % 3600) / 60)
   101→  const s = sec % 60
   102→  const pad = (v: number) => v.toString().padStart(2, '0')
   103→  return h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${m}:${pad(s)}`
   104→}

   114→
   115→export function useCallSession(options: CallSessionOptions): CallSessionControls {
   116→  const { meId, meName, meColor = 'emerald', meAvatar = null, conversationId, peer } = options
   117→  const { subscribeCallEvents, emitCallEvent, callChannelConnected } = usePulseCallChannel()
   118→
   119→  const [state, setState] = useState<CallUiState>('idle')
   120→  const [kind, setKind] = useState<CallKind>('voice')
   121→  const [direction, setDirection] = useState<'outgoing' | 'incoming' | null>(null)
   122→  const [activePeer, setActivePeer] = useState<CallPeer | null>(null)
   123→  const [error, setError] = useState<string | null>(null)
   124→  const [summary, setSummary] = useState<string | null>(null)
   125→  const [durationSec, setDurationSec] = useState(0)
   126→  const [micEnabled, setMicEnabled] = useState(true)
   127→  const [cameraEnabled, setCameraEnabled] = useState(true)
   128→  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null)
   129→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)
   130→
   131→  // Mutable mirrors so the (stable) socket listener always sees fresh values.
   132→  // (Written inside callbacks/effects only — never during render.)
   133→  const stateRef = useRef<CallUiState>('idle')
   134→  const callIdRef = useRef<string | null>(null)
   135→  const kindRef = useRef<CallKind>('voice')
   136→  const directionRef = useRef<'outgoing' | 'incoming' | null>(null)
   137→  const answeredRef = useRef(false)
   138→  const connectedAtRef = useRef(0)
   139→  const pcRef = useRef<RTCPeerConnection | null>(null)
   140→  const localStreamRef = useRef<MediaStream | null>(null)
   141→  const pendingIceRef = useRef<RTCIceCandidateInit[]>([])
   142→  const offerRef = useRef<IncomingOfferMeta | null>(null)
   143→  const peerRef = useRef<CallPeer | null>(peer)
   144→  const optionsRef = useRef({ meId, meName, meColor, meAvatar, conversationId })
   145→
   146→  // Keep option/peer mirrors in sync after each commit.
   147→  useEffect(() => {
   148→    peerRef.current = peer
   149→  }, [peer])
   150→  useEffect(() => {
   151→    optionsRef.current = { meId, meName, meColor, meAvatar, conversationId }
   152→  }, [meId, meName, meColor, meAvatar, conversationId])
   153→  useEffect(() => {
   154→    stateRef.current = state
   155→  }, [state])
   156→
   157→  /** Fire-and-forget CallLog row — the CALLER is the single writer. */
   158→  const writeLog = useCallback(async (status: 'completed' | 'missed' | 'declined', duration = 0, callKind?: CallKind) => {
   159→    const opts = optionsRef.current
   160→    const target = peerRef.current
   161→    if (!target || target.id === opts.meId) return
   162→    try {
   163→      await apiJson('/api/calls', {
   164→        method: 'POST',
   165→        body: JSON.stringify({
   166→          userId: opts.meId,
   167→          conversationId: opts.conversationId,
   168→          peerId: target.id,
   169→          kind: callKind ?? kindRef.current,
   170→          status,
   171→          durationSec: Math.max(0, Math.round(duration)),
   172→        }),
   173→      })
   174→    } catch {
   175→      toast.error('Could not save this call to history')
   176→    }
   177→  }, [])
   178→
   179→  /** Stop every track, close the peer connection, clear queued ICE. */
   180→  const teardownMedia = useCallback(() => {
   181→    if (pcRef.current) {
   182→      pcRef.current.onicecandidate = null
   183→      pcRef.current.ontrack = null
   184→      pcRef.current.onconnectionstatechange = null
   185→      try {
   186→        pcRef.current.close()
   187→      } catch {
   188→        // already closed
   189→      }
   190→      pcRef.current = null
   191→    }
   192→    for (const track of localStreamRef.current?.getTracks() ?? []) track.stop()
   193→    localStreamRef.current = null
   194→    pendingIceRef.current = []
   195→    offerRef.current = null
   196→    answeredRef.current = false
   197→    connectedAtRef.current = 0
   198→    setLocalStream(null)
   199→    setRemoteStream(null)
   200→    setDurationSec(0)
   201→    setMicEnabled(true)
   202→    setCameraEnabled(true)
   203→  }, [])
   204→
   205→  const showEnded = useCallback((text: string) => {
   206→    setSummary(text)
   207→    setState('ended')
   208→    stateRef.current = 'ended'
   209→  }, [])
   210→
   211→  const resetIdle = useCallback(() => {
   212→    teardownMedia()
   213→    callIdRef.current = null
   214→    directionRef.current = null
   215→    setState('idle')
   216→    stateRef.current = 'idle'
   217→    setSummary(null)
   218→    setError(null)
   219→    setActivePeer(null)
   220→    setDirection(null)
   221→  }, [teardownMedia])
   222→
   223→  const createPeerConnection = useCallback((): RTCPeerConnection => {
   224→    const pc = new RTCPeerConnection(STUN_SERVERS)
   225→    pc.onicecandidate = (event) => {
   226→      if (!event.candidate || !callIdRef.current) return
   227→      const me = optionsRef.current.meId
   228→      const target = directionRef.current === 'outgoing' ? peerRef.current?.id : offerRef.current?.from
   229→      if (!target) return
   230→      emitCallEvent('call:ice', {
   231→        callId: callIdRef.current,
   232→        conversationId: optionsRef.current.conversationId,
   233→        from: me,
   234→        to: target,
   235→        kind: kindRef.current,
   236→        candidate: event.candidate.candidate,
   237→        sdpMid: event.candidate.sdpMid,
   238→        sdpMLineIndex: event.candidate.sdpMLineIndex,
   239→      })
   240→    }
   241→    pc.ontrack = (event) => {
   242→      const [stream] = event.streams
   243→      if (stream) setRemoteStream(stream)
   244→    }
   245→    pc.onconnectionstatechange = () => {
   246→      // Quality hint chip derives from connectionState in the overlay.
   247→    }
   248→    // Keep the mic/camera toggles honest after renegotiations.
   249→    const stream = localStreamRef.current
   250→    if (stream) for (const track of stream.getTracks()) pc.addTrack(track, stream)
   251→    pcRef.current = pc
   252→    return pc
   253→  }, [emitCallEvent])
   254→
   255→  /**
   256→   * getUserMedia with honest fallbacks: no camera (or camera constraint
   257→   * rejected) → retry as a voice call with a toast; mic denied → throw so the
   258→   * caller can show the glass error card.
   259→   */
   260→  const acquireMedia = useCallback(async (wanted: CallKind): Promise<CallKind> => {
   261→    try {
   262→      const stream = await navigator.mediaDevices.getUserMedia({
   263→        audio: true,
   264→        video: wanted === 'video' ? { width: { ideal: 1280 }, height: { ideal: 720 } } : false,
   265→      })
   266→      localStreamRef.current = stream
   267→      setLocalStream(stream)
   268→      setCameraEnabled(wanted === 'video')
   269→      return wanted
   270→    } catch (err) {
   271→      const name = err instanceof DOMException ? err.name : ''
   272→      if (wanted === 'video' && (name === 'NotFoundError' || name === 'OverconstrainedError' || name === 'NotReadableError')) {
   273→        // No usable camera → fall back to voice, never fake it.
   274→        toast('Camera unavailable — starting a voice call')
   275→        setKind('voice')
   276→        kindRef.current = 'voice'
   277→        const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   278→        localStreamRef.current = stream
   279→        setLocalStream(stream)
   280→        setCameraEnabled(false)
   281→        return 'voice'
   282→      }
   283→      throw err
   284→    }
   285→  }, [])
   286→
   287→  const startCall = useCallback(
   288→    (wanted: CallKind) => {
   289→      if (stateRef.current !== 'idle') return
   290→      const target = peerRef.current
   291→      if (!target || target.id === optionsRef.current.meId) return
   292→      if (!callChannelConnected) {
   293→        toast.error('You are offline — calls need a connection')
   294→        return
   295→      }
   296→
   297→      const callId = `call-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`
   298→      callIdRef.current = callId
   299→      directionRef.current = 'outgoing'
   300→      kindRef.current = wanted
   301→      setKind(wanted)
   302→      setDirection('outgoing')
   303→      setActivePeer(target)
   304→      setSummary(null)
   305→      setError(null)
   306→      setState('calling')
   307→      stateRef.current = 'calling'
   308→
   309→      void (async () => {
   310→        try {
   311→          const actualKind = await acquireMedia(wanted)
   312→          if (callIdRef.current !== callId) return // canceled while prompting
   313→          const pc = createPeerConnection()
   314→          const offer = await pc.createOffer({
   315→            offerToReceiveAudio: true,
   316→            offerToReceiveVideo: actualKind === 'video',
   317→          })
   318→          await pc.setLocalDescription(offer)
   319→          emitCallEvent('call:offer', {
   320→            callId,
   321→            conversationId: optionsRef.current.conversationId,
   322→            from: optionsRef.current.meId,
   323→            to: target.id,
   324→            kind: actualKind,
   325→            sdp: offer.sdp ?? '',
   326→            callerName: optionsRef.current.meName,
   327→            callerColor: optionsRef.current.meColor,
   328→            callerAvatar: optionsRef.current.meAvatar,
   329→          })
   330→        } catch (err) {
   331→          // Mic denied (or unavailable) — honest error card, no fake UI.
   332→          console.error('[call] getUserMedia failed:', err instanceof Error ? err.message : err)
   333→          setError('Microphone access is needed for calls. Check the browser permissions and try again.')
   334→        }
   335→      })()
   336→    },
   337→    [acquireMedia, callChannelConnected, createPeerConnection, emitCallEvent],
   338→  )
   339→
   340→  const acceptCall = useCallback(() => {
   341→    const offer = offerRef.current
   342→    if (!offer || stateRef.current !== 'incoming') return
   343→    haptic(16)
   344→
   345→    void (async () => {
   346→      try {
   347→        const actualKind = await acquireMedia(offer.kind)
   348→        const pc = createPeerConnection()
   349→        await pc.setRemoteDescription({ type: 'offer', sdp: offer.sdp })
   350→        const answer = await pc.createAnswer()
   351→        await pc.setLocalDescription(answer)
   352→        emitCallEvent('call:answer', {
   353→          callId: offer.callId,
   354→          conversationId: offer.conversationId,
   355→          from: optionsRef.current.meId,
   356→          to: offer.from,
   357→          kind: actualKind,
   358→          sdp: answer.sdp ?? '',
   359→        })
   360→        // Flush candidates that arrived while we were ringing.
   361→        for (const candidate of pendingIceRef.current.splice(0)) {
   362→          try {
   363→            await pc.addIceCandidate(candidate)
   364→          } catch {
   365→            // stale candidate — ICE will recover
   366→          }
   367→        }
   368→        answeredRef.current = true
   369→        connectedAtRef.current = Date.now()
   370→        setDurationSec(0)
   371→        setState('active')
   372→        stateRef.current = 'active'
   373→        haptic(10)
   374→      } catch (err) {
   375→        console.error('[call] accept failed:', err instanceof Error ? err.message : err)
   376→        // Callee could not join → reject politely; the caller logs the row.
   377→        emitCallEvent('call:reject', {
   378→          callId: offer.callId,
   379→          conversationId: offer.conversationId,
   380→          from: optionsRef.current.meId,
   381→          to: offer.from,
   382→          kind: offer.kind,
   383→        })
   384→        setError('Microphone access is needed for calls. Check the browser permissions and try again.')
   385→        offerRef.current = offer
   386→        callIdRef.current = offer.callId
   387→        directionRef.current = 'incoming'
   388→      }
   389→    })()
   390→  }, [acquireMedia, createPeerConnection, emitCallEvent])
   391→
   392→  const declineCall = useCallback(() => {
   393→    const offer = offerRef.current
   394→    if (!offer || stateRef.current !== 'incoming') return
   395→    haptic(12)
   396→    // Callee NEVER writes the row — the caller (who got call:reject) logs 'declined'.
   397→    emitCallEvent('call:reject', {
   398→      callId: offer.callId,
   399→      conversationId: offer.conversationId,
   400→      from: optionsRef.current.meId,
   401→      to: offer.from,
   402→      kind: offer.kind,
   403→    })
   404→    teardownMedia()
   405→    callIdRef.current = null
   406→    showEnded('Declined')
   407→  }, [emitCallEvent, showEnded, teardownMedia])
   408→
   409→  const endCall = useCallback(() => {
   410→    const callId = callIdRef.current
   411→    const current = stateRef.current
   412→    if (current === 'idle') return
   413→    haptic(14)
   414→
   415→    if (current === 'incoming') {
   416→      declineCall()
   417→      return
   418→    }
   419→
   420→    if (current === 'calling' && directionRef.current === 'outgoing') {
   421→      if (callId) {
   422→        emitCallEvent('call:cancel', {
   423→          callId,
   424→          conversationId: optionsRef.current.conversationId,
   425→          from: optionsRef.current.meId,
   426→          to: peerRef.current?.id ?? '',
   427→          kind: kindRef.current,
   428→        })
   429→      }
   430→      // Caller cancels an unanswered ring → 'missed' (single-writer rule).
   431→      void writeLog('missed', 0)
   432→      showEnded('Call canceled')
   433→      return
   434→    }
   435→
   436→    if (current === 'active') {
   437→      const elapsed = connectedAtRef.current > 0 ? (Date.now() - connectedAtRef.current) / 1000 : 0
   438→      if (callId) {
   439→        emitCallEvent('call:hangup', {
   440→          callId,
   441→          conversationId: optionsRef.current.conversationId,
   442→          from: optionsRef.current.meId,
   443→          to:
   444→            directionRef.current === 'outgoing'
   445→              ? peerRef.current?.id ?? ''
   446→              : offerRef.current?.from ?? '',
   447→          kind: kindRef.current,
   448→          durationSec: Math.round(elapsed),
   449→        })
   450→      }
   451→      if (directionRef.current === 'outgoing') {
   452→        void writeLog('completed', elapsed)
   453→      }
   454→      showEnded(elapsed >= 1 ? `Call ended · ${formatCallDuration(elapsed)}` : 'Call ended')
   455→      return
   456→    }
   457→  }, [declineCall, emitCallEvent, showEnded, writeLog])
   458→
   459→  const dismissError = useCallback(() => {
   460→    // A failed attempt (e.g. mic denied) still counts as an unanswered call.
   461→    if (directionRef.current === 'outgoing') void writeLog('missed', 0)
   462→    resetIdle()
   463→  }, [resetIdle, writeLog])
   464→
   465→  const toggleMic = useCallback(() => {
   466→    const stream = localStreamRef.current
   467→    if (!stream) return
   468→    const next = !stream.getAudioTracks().every((t) => t.enabled)
   469→    for (const track of stream.getAudioTracks()) track.enabled = next
   470→    setMicEnabled(next)
   471→    haptic(6)
   472→  }, [])
   473→
   474→  const toggleCamera = useCallback(() => {
   475→    const stream = localStreamRef.current
   476→    if (!stream || kindRef.current !== 'video') return
   477→    const videoTracks = stream.getVideoTracks()
   478→    if (videoTracks.length === 0) return
   479→    const next = !videoTracks.every((t) => t.enabled)
   480→    for (const track of videoTracks) track.enabled = next
   481→    setCameraEnabled(next)
   482→    haptic(6)
   483→  }, [])
   484→
   485→  // ── signaling listener (stable; everything via refs) ────────
   486→  const handleCallEvent = useCallback(
   487→    (event: string, raw: unknown) => {
   488→      if (raw === null || typeof raw !== 'object') return
   489→      const payload = raw as Record<string, unknown>
   490→      const callId = typeof payload.callId === 'string' ? payload.callId : ''
   491→      const me = optionsRef.current.meId
   492→
   493→      if (event === 'call:offer') {
   494→        const from = typeof payload.from === 'string' ? payload.from : ''
   495→        if (!callId || from === me) return
   496→        // Already busy → reject immediately (the caller logs 'declined').
   497→        if (stateRef.current !== 'idle') {
   498→          emitCallEvent('call:reject', {
   499→            callId,
   500→            conversationId: optionsRef.current.conversationId,
   501→            from: me,
   502→            to: from,
   503→            kind: payload.kind === 'video' ? 'video' : 'voice',
   504→          })
   505→          return
   506→        }
   507→        const kind = payload.kind === 'video' ? 'video' : 'voice'
   508→        offerRef.current = {
   509→          callId,
   510→          conversationId: typeof payload.conversationId === 'string' ? payload.conversationId : '',
   511→          from,
   512→          kind,
   513→          sdp: typeof payload.sdp === 'string' ? payload.sdp : '',
   514→          peer: {
   515→            id: from,
   516→            name: typeof payload.callerName === 'string' && payload.callerName ? payload.callerName : 'Someone',
   517→            color: typeof payload.callerColor === 'string' && payload.callerColor ? payload.callerColor : 'emerald',
   518→            avatar: typeof payload.callerAvatar === 'string' ? payload.callerAvatar : null,
   519→          },
   520→        }
   521→        callIdRef.current = callId
   522→        directionRef.current = 'incoming'
   523→        kindRef.current = kind
   524→        setActivePeer(offerRef.current.peer)
   525→        setKind(kind)
   526→        setDirection('incoming')
   527→        setSummary(null)
   528→        setError(null)
   529→        setState('incoming')
   530→        stateRef.current = 'incoming'
   531→        haptic(30)
   532→        return
   533→      }
   534→
   535→      // Everything below targets the call we are part of.
   536→      if (!callId || callId !== callIdRef.current) return
   537→
   538→      if (event === 'call:answer' && directionRef.current === 'outgoing') {
   539→        const sdp = typeof payload.sdp === 'string' ? payload.sdp : ''
   540→        if (!sdp || !pcRef.current) return
   541→        void (async () => {
   542→          try {
   543→            await pcRef.current?.setRemoteDescription({ type: 'answer', sdp })
   544→            for (const candidate of pendingIceRef.current.splice(0)) {
   545→              try {
   546→                await pcRef.current?.addIceCandidate(candidate)
   547→              } catch {
   548→                // stale candidate — ICE recovers on its own
   549→              }
   550→            }
   551→            answeredRef.current = true
   552→            connectedAtRef.current = Date.now()
   553→            setDurationSec(0)
   554→            setState('active')
   555→            stateRef.current = 'active'
   556→            haptic(10)
   557→          } catch (err) {
   558→            console.error('[call] answer apply failed:', err instanceof Error ? err.message : err)
   559→            showEnded('Call failed')
   560→          }
   561→        })()
   562→        return
   563→      }
   564→
   565→      if (event === 'call:ice') {
   566→        const candidateString = typeof payload.candidate === 'string' ? payload.candidate : ''
   567→        if (!candidateString) return
   568→        const candidate: RTCIceCandidateInit = {
   569→          candidate: candidateString,
   570→          sdpMid: typeof payload.sdpMid === 'string' ? payload.sdpMid : null,
   571→          sdpMLineIndex:
   572→            typeof payload.sdpMLineIndex === 'number' && Number.isFinite(payload.sdpMLineIndex)
   573→              ? Math.floor(payload.sdpMLineIndex)
   574→              : null,
   575→        }
   576→        if (pcRef.current && pcRef.current.remoteDescription) {
   577→          void pcRef.current.addIceCandidate(candidate).catch(() => {
   578→            // stale candidate — ignore
   579→          })
   580→        } else {
   581→          pendingIceRef.current.push(candidate)
   582→        }
   583→        return
   584→      }
   585→
   586→      if (event === 'call:reject' && directionRef.current === 'outgoing') {
   587→        // Caller got declined → logs 'declined' (single-writer rule).
   588→        teardownMedia()
   589→        callIdRef.current = null
   590→        void writeLog('declined', 0)
   591→        showEnded('Declined')
   592→        return
   593→      }
   594→
   595→      if (event === 'call:cancel') {
   596→        const reason: CallCancelReason =
   597→          payload.reason === 'timeout' || payload.reason === 'busy' || payload.reason === 'offline'
   598→            ? payload.reason
   599→            : 'cancel'
   600→        const wasOutgoing = directionRef.current === 'outgoing'
   601→        const wasAnswered = answeredRef.current
   602→        teardownMedia()
   603→        callIdRef.current = null
   604→        if (wasOutgoing && !wasAnswered) {
   605→          // Caller's ring died (timeout / self-cancel) → 'missed'.
   606→          void writeLog('missed', 0)
   607→          showEnded(CANCEL_SUMMARY[reason])
   608→        } else if (!wasOutgoing && !wasAnswered) {
   609→          showEnded('Missed call')
   610→        } else if (wasAnswered) {
   611→          showEnded('Call ended')
   612→        } else {
   613→          showEnded(CANCEL_SUMMARY[reason])
   614→        }
   615→        return
   616→      }
   617→
   618→      if (event === 'call:hangup') {
   619→        const elapsed = connectedAtRef.current > 0 ? (Date.now() - connectedAtRef.current) / 1000 : 0
   620→        const wasOutgoing = directionRef.current === 'outgoing'
   621→        teardownMedia()
   622→        callIdRef.current = null
   623→        if (wasOutgoing) void writeLog('completed', elapsed)
   624→        showEnded(elapsed >= 1 ? `Call ended · ${formatCallDuration(elapsed)}` : 'Call ended')
   625→        return
   626→      }
   627→    },
   628→    [emitCallEvent, showEnded, teardownMedia, writeLog],
   629→  )
   630→
   631→  useEffect(() => subscribeCallEvents(handleCallEvent), [subscribeCallEvents, handleCallEvent])
   632→
   633→  // Live duration ticker while connected.
   634→  useEffect(() => {
   635→    if (state !== 'active') return undefined
   636→    const interval = setInterval(() => {
   637→      if (connectedAtRef.current > 0) {
   638→        setDurationSec(Math.floor((Date.now() - connectedAtRef.current) / 1000))
   639→      }
   640→    }, 1000)
   641→    return () => clearInterval(interval)
   642→  }, [state])
   643→
   644→  // Ended card auto-dismisses after ~2.5s.
   645→  useEffect(() => {
   646→    if (state !== 'ended') return undefined
   647→    const timer = setTimeout(() => resetIdle(), 2500)
   648→    return () => clearTimeout(timer)
   649→  }, [state, resetIdle])
   650→
   651→  return {
   652→    state,
   653→    kind,
   654→    direction,
   655→    peer: activePeer,
   656→    error,
   657→    summary,
   658→    durationSec,
   659→    micEnabled,
   660→    cameraEnabled,
   661→    remoteStream,
   662→    localStream,
   663→    startCall,
   664→    acceptCall,
   665→    declineCall,
   666→    endCall,
   667→    dismissError,
   668→    toggleMic,
   669→    toggleCamera,
   670→  }
   671→}
