   127→function tallyOf(rsvps: EventRsvpWire[]): EventCounts {
   128→  const counts: EventCounts = { going: 0, maybe: 0, no: 0 }
   129→  for (const r of rsvps) counts[r.status] += 1
   130→  return counts
   131→}

   133→/** 'in 2h 15m' · 'Starting now' under a minute · future only. */
   134→function countdownLabel(startsAtMs: number, nowMs: number): string {
   135→  const diff = startsAtMs - nowMs
   136→  if (diff < 60_000) return 'Starting now'
   137→  const mins = Math.floor(diff / 60_000)
   138→  if (mins < 60) return `in ${mins}m`
   139→  const hours = Math.floor(mins / 60)
   140→  if (hours < 24) return mins % 60 > 0 ? `in ${hours}h ${mins % 60}m` : `in ${hours}h`
   141→  const days = Math.floor(hours / 24)
   142→  return hours % 24 > 0 ? `in ${days}d ${hours % 24}h` : `in ${days}d`
   143→}
   144→
   145→/** '2d ago' relative chip for the collapsed Past section. */
   146→function pastLabel(startsAtMs: number, nowMs: number): string {
   147→  const mins = Math.max(0, Math.floor((nowMs - startsAtMs) / 60_000))
   148→  if (mins < 1) return 'just now'
   149→  if (mins < 60) return `${mins}m ago`
   150→  const hours = Math.floor(mins / 60)
   151→  if (hours < 24) return `${hours}h ago`
   152→  return `${Math.floor(hours / 24)}d ago`
   153→}

   155→/** Local `YYYY-MM-DDTHH:mm` for tomorrow 18:00 — datetime-local default. */
   156→function defaultStartLocal(): string {
   157→  const d = new Date()
   158→  d.setDate(d.getDate() + 1)
   159→  d.setHours(18, 0, 0, 0)
   160→  const pad = (n: number) => String(n).padStart(2, '0')
   161→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   162→}

   196→/** Initials avatar stack of the going members (up to 4 + '+n'). */
   197→function GoingStack({
   198→  rsvps,
   199→  memberById,
   200→}: {
   201→  rsvps: EventRsvpWire[]
   202→  memberById: Map<string, EventMember>
   203→}) {
   204→  const going = rsvps.filter((r) => r.status === 'going')
   205→  if (going.length === 0) return null
   206→  const shown = going.slice(0, 4)
   207→  const extra = going.length - shown.length
   208→  return (
   209→    <div className="flex items-center" role="img" aria-label={`${going.length} going`}>
   210→      {shown.map((r, i) => (
   211→        <span
   212→          key={r.userId}
   213→          title={r.name}
   214→          className={cn(
   215→            'flex size-[22px] items-center justify-center rounded-full bg-gradient-to-br text-[9px] font-bold text-white ring-2 ring-zinc-950',
   216→            gradientFor(memberById.get(r.userId)?.color ?? 'emerald'),
   217→          )}
   218→          style={{ marginLeft: i === 0 ? 0 : -6 }}
   219→        >
   220→          {initialsOf(r.name)}
   221→        </span>
   222→      ))}
   223→      {extra > 0 ? (
   224→        <span
   225→          className="flex size-[22px] items-center justify-center rounded-full bg-zinc-700 text-[9px] font-bold text-zinc-200 ring-2 ring-zinc-950"
   226→          style={{ marginLeft: -6 }}
   227→        >
   228→          +{extra}
   229→        </span>
   230→      ) : null}
   231→    </div>
   232→  )
   233→}

   237→export function EventsSheet({
   238→  open,
   239→  onClose,
   240→  conversationId,
   241→  me,
   242→  members,
   243→  myRole = 'member',
   244→}: {
   245→  open: boolean
   246→  onClose: () => void
   247→  conversationId: string
   248→  me: { id: string }
   249→  members: EventMember[]
   250→  myRole?: GroupRole
   251→}) {
   252→  const reduce = useReducedMotion()
   253→  const queryClient = useQueryClient()
   254→
   255→  // form state
   256→  const [title, setTitle] = useState('')
   257→  const [startsAtLocal, setStartsAtLocal] = useState(defaultStartLocal)
   258→  const [location, setLocation] = useState('')
   259→  // two-tap delete confirm (row id currently armed)
   260→  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null)
   261→  const deleteTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   262→  const [pastOpen, setPastOpen] = useState(false)
   263→
   264→  // live clock for countdown chips — recomputed every 30s while open
   265→  const [nowMs, setNowMs] = useState(() => Date.now())
   266→  useEffect(() => {
   267→    if (!open) return
   268→    // async first tick (fresh clock on open) then steady 30s cadence
   269→    let timer: ReturnType<typeof setTimeout>
   270→    const tick = () => {
   271→      setNowMs(Date.now())
   272→      timer = setTimeout(tick, TICK_MS)
   273→    }
   274→    timer = setTimeout(tick, 0)
   275→    return () => clearTimeout(timer)
   276→  }, [open])
   277→
   278→  useEffect(() => {
   279→    return () => {
   280→      if (deleteTimerRef.current) clearTimeout(deleteTimerRef.current)
   281→    }
   282→  }, [])
   283→
   284→  const memberById = useMemo(() => new Map(members.map((m) => [m.id, m])), [members])
   285→
   286→  // ── data (5s poll, paused when document.hidden) ──────────────
   287→
   288→  const eventsQuery = useQuery({
   289→    queryKey: ['events', conversationId],
   290→    enabled: open && conversationId.length > 0 && me.id.length > 0,
   291→    refetchInterval: POLL_MS,
   292→    refetchIntervalInBackground: false, // tab hidden → poll pauses
   293→    staleTime: 0,
   294→    retry: 1,
   295→    queryFn: async (): Promise<EventsGetResponse> =>
   296→      apiJson<EventsGetResponse>(
   297→        `/api/conversations/${encodeURIComponent(conversationId)}/events?userId=${encodeURIComponent(me.id)}`,
   298→      ),
   299→  })
   300→
   301→  const rsvpMutation = useMutation({
   302→    mutationFn: ({ eventId, status }: { eventId: string; status: EventRsvpStatus }) =>
   303→      apiJson<RsvpResponse>(`/api/events/${encodeURIComponent(eventId)}/rsvp`, {
   304→        method: 'POST',
   305→        body: JSON.stringify({ userId: me.id, status }),
   306→      }),
   307→    // optimistic: move my pill + tally immediately, the poll confirms
   308→    onMutate: async ({ eventId, status }) => {
   309→      await queryClient.cancelQueries({ queryKey: ['events', conversationId] })
   310→      const prev = queryClient.getQueryData<EventsGetResponse>(['events', conversationId])
   311→      queryClient.setQueryData<EventsGetResponse>(['events', conversationId], (old) => {
   312→        if (!old) return old
   313→        return {
   314→          events: old.events.map((e) => {
   315→            if (e.id !== eventId) return e
   316→            const others = e.rsvps.filter((r) => r.userId !== me.id)
   317→            const rsvps: EventRsvpWire[] = [
   318→              ...others,
   319→              { userId: me.id, name: memberById.get(me.id)?.name ?? 'You', status },
   320→            ]
   321→            return { ...e, rsvps, counts: tallyOf(rsvps), myStatus: status }
   322→          }),
   323→        }
   324→      })
   325→      return { prev }
   326→    },
   327→    onError: (error, _vars, ctx) => {
   328→      if (ctx?.prev) queryClient.setQueryData(['events', conversationId], ctx.prev)
   329→      toast.error(error instanceof Error ? error.message : 'RSVP failed — try again.')
   330→    },
   331→    onSettled: () => {
   332→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   333→    },
   334→  })
   335→
   336→  const createMutation = useMutation({
   337→    mutationFn: (payload: { title: string; startsAt: string; location: string }) =>
   338→      apiJson<EventsPostResponse>(
   339→        `/api/conversations/${encodeURIComponent(conversationId)}/events`,
   340→        {
   341→          method: 'POST',
   342→          body: JSON.stringify({ userId: me.id, ...payload }),
   343→        },
   344→      ),
   345→    onSuccess: (res) => {
   346→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   347→      setTitle('')
   348→      setLocation('')
   349→      setStartsAtLocal(defaultStartLocal())
   350→      haptic(14)
   351→      toast.success(`“${res.event.title}” is on the calendar.`)
   352→    },
   353→    onError: (error) => {
   354→      toast.error(error instanceof Error ? error.message : 'Could not schedule the event.')
   355→    },
   356→  })
   357→
   358→  const deleteMutation = useMutation({
   359→    mutationFn: (eventId: string) =>
   360→      apiJson<DeleteResponse>(`/api/events/${encodeURIComponent(eventId)}?userId=${encodeURIComponent(me.id)}`, {
   361→        method: 'DELETE',
   362→      }),
   363→    onSuccess: () => {
   364→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   365→      haptic(14)
   366→      toast.success('Event deleted.')
   367→    },
   368→    onError: (error) => {
   369→      toast.error(error instanceof Error ? error.message : 'Could not delete the event.')
   370→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   371→    },
   372→  })
   373→
   374→  // ── derived lists (server pre-sorts: upcoming asc, past desc) ─
   375→
   376→  const events = eventsQuery.data?.events ?? []
   377→  const upcoming = useMemo(
   378→    () => events.filter((e) => new Date(e.startsAt).getTime() >= nowMs),
   379→    [events, nowMs],
   380→  )
   381→  const past = useMemo(
   382→    () => events.filter((e) => new Date(e.startsAt).getTime() < nowMs),
   383→    [events, nowMs],
   384→  )
   385→
   386→  const offline = eventsQuery.isError
   387→
   388→  const submitCreate = (e: React.FormEvent<HTMLFormElement>) => {
   389→    e.preventDefault()
   390→    const trimmed = title.trim()
   391→    const ms = startsAtLocal.length > 0 ? Date.parse(startsAtLocal) : Number.NaN
   392→    if (trimmed.length === 0 || Number.isNaN(ms)) return
   393→    createMutation.mutate({
   394→      title: trimmed,
   395→      startsAt: new Date(ms).toISOString(),
   396→      location: location.trim(),
   397→    })
   398→  }
   399→
   400→  const handleDeleteTap = (eventId: string) => {
   401→    if (confirmDeleteId !== eventId) {
   402→      setConfirmDeleteId(eventId)
   403→      haptic(12)
   404→      if (deleteTimerRef.current) clearTimeout(deleteTimerRef.current)
   405→      deleteTimerRef.current = setTimeout(() => setConfirmDeleteId(null), DELETE_CONFIRM_MS)
   406→      return
   407→    }
   408→    if (deleteTimerRef.current) {
   409→      clearTimeout(deleteTimerRef.current)
   410→      deleteTimerRef.current = null
   411→    }
   412→    setConfirmDeleteId(null)
   413→    deleteMutation.mutate(eventId)
   414→  }
   415→
   416→  const statusLine = offline ? 'Reconnecting…' : `${upcoming.length} upcoming · live`
   417→
   418→  // ── row renderer (shared by upcoming + past) ─────────────────
   419→
   420→  const renderRow = (e: GroupEventWire, kind: 'upcoming' | 'past') => {
   421→    const startsAtMs = new Date(e.startsAt).getTime()
   422→    const canDelete = me.id === e.createdById || myRole === 'admin'
   423→    const creatorName = (e.createdById ? memberById.get(e.createdById)?.name : undefined) ?? e.createdByName
   424→    return (
   425→      <motion.div
   426→        key={e.id}
   427→        layout={!reduce}
   428→        variants={rowVariants}
   429→        initial={reduce ? false : 'hidden'}
   430→        animate="show"
   431→        exit={reduce ? { opacity: 0 } : { opacity: 0, scale: 0.96, y: 6 }}
   432→        transition={spring.soft}
   433→        style={{ willChange: 'transform' }}
   434→        className="rounded-2xl border border-white/10 bg-white/[0.04] p-3.5"
   435→      >
   436→        <div className="flex gap-3">
   437→          <DateTile startsAtMs={startsAtMs} />
   438→          <div className="min-w-0 flex-1">
   439→            <div className="flex items-start justify-between gap-2">
   440→              <p className="min-w-0 truncate text-[14px] font-bold leading-tight text-zinc-50">
   441→                {e.title}
   442→              </p>
   443→              {canDelete ? (
   444→                <motion.button
   445→                  type="button"
   446→                  whileTap={reduce ? undefined : pressTap}
   447→                  disabled={deleteMutation.isPending}
   448→                  aria-label={
   449→                    confirmDeleteId === e.id
   450→                      ? `Tap again to delete event ${e.title}`
   451→                      : `Delete event ${e.title}`
   452→                  }
   453→                  onClick={() => handleDeleteTap(e.id)}
   454→                  className={cn(
   455→                    'flex size-7 shrink-0 items-center justify-center rounded-full outline-none ring-emerald-400/60 transition-colors focus-visible:ring-2 disabled:opacity-50',
   456→                    confirmDeleteId === e.id
   457→                      ? 'bg-rose-500 text-white'
   458→                      : 'text-zinc-500 hover:bg-rose-500/10 hover:text-rose-400',
   459→                  )}
   460→                >
   461→                  <Trash2 className="size-3.5" aria-hidden />
   462→                </motion.button>
   463→              ) : null}
   464→            </div>
   465→            <div className="mt-1 flex flex-wrap items-center gap-x-2.5 gap-y-0.5">
   466→              <span
   467→                className={cn(
   468→                  'rounded-full px-2 py-0.5 text-[10.5px] font-bold tabular-nums',
   469→                  kind === 'upcoming'
   470→                    ? startsAtMs - nowMs < 60_000
   471→                      ? 'bg-emerald-500 text-white'
   472→                      : 'bg-emerald-500/15 text-emerald-400'
   473→                    : 'bg-white/5 text-zinc-500',
   474→                )}
   475→              >
   476→                {kind === 'upcoming' ? countdownLabel(startsAtMs, nowMs) : pastLabel(startsAtMs, nowMs)}
   477→              </span>
   478→              {e.location ? (
   479→                <span className="flex min-w-0 items-center gap-1 text-[11px] font-medium text-zinc-400">
   480→                  <MapPin className="size-3 shrink-0 text-zinc-500" aria-hidden />
   481→                  <span className="truncate">{e.location}</span>
   482→                </span>
   483→              ) : null}
   484→            </div>
   485→            {creatorName ? (
   486→              <p className="mt-1 text-[11px] font-medium text-zinc-500">by {creatorName}</p>
   487→            ) : null}
   488→          </div>
   489→        </div>
   490→
   491→        <div className="mt-2.5 flex items-center justify-between gap-2">
   492→          <div role="group" aria-label={`RSVP for ${e.title}`} className="flex gap-1.5">
   493→            {RSVP_CHOICES.map(({ status, label }) => {
   494→              const mine = e.myStatus === status
   495→              return (
   496→                <motion.button
   497→                  key={status}
   498→                  type="button"
   499→                  whileTap={reduce ? undefined : pressTap}
   500→                  aria-pressed={mine}
   501→                  aria-label={`RSVP ${status} ${e.title}`}
   502→                  disabled={rsvpMutation.isPending}
   503→                  onClick={() => {
   504→                    if (e.myStatus !== status) {
   505→                      haptic(8)
   506→                      rsvpMutation.mutate({ eventId: e.id, status })
   507→                    }
   508→                  }}
   509→                  className={cn(
   510→                    'flex h-8 items-center gap-1 rounded-full px-2.5 text-[11px] font-bold outline-none ring-emerald-400/60 transition-colors duration-150 focus-visible:ring-2 disabled:opacity-60',
   511→                    mine
   512→                      ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-600/25'
   513→                      : 'bg-white/5 text-zinc-300 hover:bg-white/10',
   514→                  )}
   515→                >
   516→                  {label}
   517→                  <span className={cn('tabular-nums', mine ? 'text-white/90' : 'text-zinc-500')}>
   518→                    {e.counts[status]}
   519→                  </span>
   520→                </motion.button>
   521→              )
   522→            })}
   523→          </div>
   524→          <GoingStack rsvps={e.rsvps} memberById={memberById} />
   525→        </div>
   526→      </motion.div>
   527→    )
   528→  }
   529→
   530→  return (
   531→    <Drawer
   532→      open={open}
   533→      handleOnly
   534→      onOpenChange={(next) => {
   535→        if (!next) onClose()
   536→      }}
   537→    >
   538→      <DrawerContent
   539→        aria-label="Events"
   540→        className="mx-auto h-[92dvh] max-h-[92dvh] max-w-[420px] rounded-t-3xl border-white/10 bg-zinc-950 px-0 pb-0 [&>div:first-child]:bg-white/20 dark:border-white/10 dark:bg-zinc-950"
   541→      >
   542→        <DrawerTitle className="sr-only">Events</DrawerTitle>
   543→        <DrawerDescription className="sr-only">
   544→          Schedule group events and RSVP with everyone in this chat
   545→        </DrawerDescription>
   546→
   547→        <motion.div
   548→          initial={reduce ? false : { opacity: 0, y: 28, scale: 0.985 }}
   549→          animate={{ opacity: 1, y: 0, scale: 1 }}

   732→                  </AnimatePresence>
   733→                </motion.section>

   735→            </AnimatePresence>
   736→          </div>
   737→        </motion.div>
   738→      </DrawerContent>
   739→    </Drawer>
   740→  )

   743→/**
   744→ * Drop-in wiring for chat-room (lead, 2 lines):
   745→ *   const events = useEventsSheet(conversationId, me.id, members, myRole?)
   746→ *   ... {events.node}
   747→ * Listens for the `pulse:open-events` CustomEvent dispatched by
   748→ * the /events slash-palette entry.
   749→ */
   750→export function useEventsSheet(
   751→  conversationId: string,
   752→  meId: string,
   753→  members: EventMember[],
   754→  myRole: GroupRole = 'member',
   755→) {
   756→  const [open, setOpen] = useState(false)
   757→
   758→  useEffect(() => {
   759→    const handler = () => setOpen(true)
   760→    window.addEventListener(EVENTS_OPEN_EVENT, handler)
   761→    return () => window.removeEventListener(EVENTS_OPEN_EVENT, handler)
   762→  }, [])
   763→
   764→  const node = (
   765→    <EventsSheet
   766→      open={open}
   767→      onClose={() => setOpen(false)}
   768→      conversationId={conversationId}
   769→      me={{ id: meId }}
   770→      members={members}
   771→      myRole={myRole}
   772→    />
   773→  )
   774→
   775→  return { open, setOpen, node }
   776→}
