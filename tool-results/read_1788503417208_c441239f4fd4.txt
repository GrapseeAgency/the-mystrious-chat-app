   209→const COLUMNS: Array<{ id: HubTaskItem['status']; label: string }> = [

   215→function TasksPanel({ me }: { me: AppUser }) {
   216→  const qc = useQueryClient()
   217→  const [title, setTitle] = useState('')
   218→
   219→  const tasksQ = useQuery({
   220→    queryKey: ['hub-tasks', me.id],
   221→    queryFn: async () => {
   222→      const res = await apiJson<{ tasks: HubTaskItem[] }>(`/api/hub/tasks?userId=${encodeURIComponent(me.id)}`)
   223→      return res.tasks
   224→    },
   225→  })
   226→  const invalidate = () => void qc.invalidateQueries({ queryKey: ['hub-tasks', me.id] })
   227→
   228→  const add = useMutation({
   229→    mutationFn: (t: string) =>
   230→      apiJson<{ task: HubTaskItem }>('/api/hub/tasks', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({ userId: me.id, title: t }),
   234→      }),
   235→    onSuccess: () => {
   236→      setTitle('')
   237→      invalidate()
   238→    },
   239→    onError: (err: Error) => toast.error(err.message),
   240→  })
   241→
   242→  const move = useMutation({
   243→    mutationFn: (input: { id: string; status: HubTaskItem['status'] }) =>
   244→      apiJson(`/api/hub/tasks/${input.id}`, {
   245→        method: 'PATCH',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({ userId: me.id, status: input.status }),
   248→      }),
   249→    onSettled: invalidate,
   250→    onError: (err: Error) => toast.error(err.message),
   251→  })
   252→
   253→  const remove = useMutation({
   254→    mutationFn: (id: string) => apiJson(`/api/hub/tasks/${id}?userId=${encodeURIComponent(me.id)}`, { method: 'DELETE' }),
   255→    onSettled: invalidate,
   256→    onError: (err: Error) => toast.error(err.message),
   257→  })
   258→
   259→  const byStatus = useMemo(() => {
   260→    const map: Record<HubTaskItem['status'], HubTaskItem[]> = { todo: [], doing: [], done: [] }
   261→    for (const t of tasksQ.data ?? []) map[t.status]?.push(t)
   262→    return map
   263→  }, [tasksQ.data])
   264→
   265→  return (
   266→    <div className="flex flex-col gap-3">
   267→      <form
   268→        className="flex gap-2"
   269→        onSubmit={(e) => {
   270→          e.preventDefault()
   271→          if (title.trim()) add.mutate(title.trim())
   272→        }}
   273→      >
   274→        <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="New task…" className="h-9" />
   275→        <Button type="submit" size="sm" disabled={!title.trim() || add.isPending} className="shrink-0">
   276→          <Check className="size-4" />
   277→        </Button>
   278→      </form>
   279→
   280→      <div className="grid grid-cols-3 gap-2">
   281→        {COLUMNS.map((col) => (
   282→          <div key={col.id} className="rounded-xl bg-zinc-100/70 p-2 dark:bg-white/5">
   283→            <p className="mb-2 px-1 text-[11px] font-bold uppercase tracking-wide text-zinc-500">
   284→              {col.label} · {byStatus[col.id].length}
   285→            </p>
   286→            <ul className="flex flex-col gap-1.5">
   287→              {byStatus[col.id].map((t) => (
   288→                <motion.li
   289→                  key={t.id}
   290→                  layout
   291→                  initial={{ opacity: 0, scale: 0.95 }}
   292→                  animate={{ opacity: 1, scale: 1 }}
   293→                  className="rounded-lg border border-zinc-200 bg-white p-2 shadow-sm dark:border-white/10 dark:bg-zinc-900/80"
   294→                >
   295→                  <p className={cn('text-[12px] font-medium leading-snug', t.status === 'done' && 'text-zinc-400 line-through')}>
   296→                    {t.title}
   297→                  </p>
   298→                  <div className="mt-1.5 flex items-center justify-between">
   299→                    <button
   300→                      type="button"
   301→                      aria-label="Advance status"
   302→                      className="text-[10px] font-semibold text-emerald-600 hover:underline"
   303→                      onClick={() =>
   304→                        move.mutate({ id: t.id, status: t.status === 'todo' ? 'doing' : t.status === 'doing' ? 'done' : 'todo' })
   305→                      }
   306→                    >
   307→                      {t.status === 'done' ? '↺ reopen' : '→ next'}
   308→                    </button>
   309→                    <button
   310→                      type="button"
   311→                      aria-label="Delete task"
   312→                      className="text-[10px] text-zinc-400 hover:text-rose-500"
   313→                      onClick={() => remove.mutate(t.id)}
   314→                    >
   315→                      ✕
   316→                    </button>
   317→                  </div>
   318→                </motion.li>
   319→              ))}
   320→              {byStatus[col.id].length === 0 ? (
   321→                <li className="rounded-lg border border-dashed border-zinc-300 px-2 py-3 text-center text-[10px] text-zinc-400 dark:border-white/15">
   322→                  empty
   323→                </li>
   324→              ) : null}
   325→            </ul>
   326→          </div>
   327→        ))}
   328→      </div>
   329→    </div>
   330→  )
   331→}

   439→                  {' · '}
   440→                  {new Date(l.createdAt).toLocaleDateString()}

   447→                    Buy
   448→                  </Button>
   449→                ) : null}
   450→                {l.mine && l.status === 'open' ? <span className="text-[10px] font-semibold text-zinc-400">yours</span> : null}

   452→            </div>
   453→          </motion.div>

   457→            The board is empty — be the first seller.
   458→          </p>
   459→        ) : null}
   460→      </section>
   461→    </div>
   462→  )
   463→}

   982→      </AnimatePresence>
   983→    </div>
