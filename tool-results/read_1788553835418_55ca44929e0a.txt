   852→/**
   853→ * CountUp-style spring counter — the motion value renders straight to the DOM
   854→ * (no per-frame React state), spring-popping on mount and retargeting on change.
   855→ */
   856→function CountUp({ value, cap = false }: { value: number; cap?: boolean }) {
   857→  const reducedMotion = useReducedMotion()
   858→  const count = useMotionValue(0)
   859→  const text = useTransform(count, (v) => {
   860→    const n = Math.round(v)
   861→    return cap && n > 99 ? '99+' : n.toLocaleString('en-US')
   862→  })
   863→
   864→  useEffect(() => {
   865→    if (reducedMotion) {
   866→      count.set(value)
   867→      return
   868→    }
   869→    const controls = animate(count, value, { type: 'spring', stiffness: 140, damping: 26 })
   870→    return () => controls.stop()
   871→  }, [value, reducedMotion, count])
   872→
   873→  return <motion.span>{text}</motion.span>
   874→}

   961→/** Mirrors normalizeUsername on the server: 3–20 chars, a-z0-9_ */

   963→  return value.length >= HANDLE_MIN && value.length <= HANDLE_MAX && HANDLE_RE.test(value)
   964→}
   965→
   966→/** Keep only characters the server would accept. */
   967→function sanitizeHandleInput(value: string): string {
   968→  return value
   969→    .toLowerCase()
   970→    .replace(/[^a-z0-9_]/g, '')
   971→    .slice(0, HANDLE_MAX)
   972→}

   974→function HandleEditorDialog({
   975→  open,
   976→  onOpenChange,
   977→  me,
   978→  onSaved,
   979→}: {
   980→  open: boolean
   981→  onOpenChange: (v: boolean) => void
   982→  me: AppUser
   983→  onSaved: (user: AppUser) => void
   984→}) {
   985→  // body mounts only while open → its state resets naturally on every open
   986→  if (!open) return null
   987→  return (
   988→    <Dialog open onOpenChange={onOpenChange}>
   989→      <HandleEditorBody me={me} onSaved={onSaved} onClose={() => onOpenChange(false)} />
   990→    </Dialog>
   991→  )
   992→}
   993→
   994→function HandleEditorBody({
   995→  me,
   996→  onSaved,
   997→  onClose,
   998→}: {
   999→  me: AppUser
  1000→  onSaved: (user: AppUser) => void
  1001→  onClose: () => void
  1002→}) {
  1003→  const [value, setValue] = useState(me.username ?? '')
  1004→  const [debounced, setDebounced] = useState(me.username ?? '')
  1005→  /** suggestion captured from a 409 username_taken response */
  1006→  const [clashSuggestion, setClashSuggestion] = useState<string | null>(null)
  1007→
  1008→  const trimmed = value.trim()
  1009→  const changed = trimmed !== (me.username ?? '')
  1010→  const valid = isValidHandle(trimmed)
  1011→
  1012→  // live availability — debounced, skipped while re-typing the current handle
  1013→  const checkEnabled = valid && debounced !== (me.username ?? '')
  1014→  const checkQ = useQuery({
  1015→    queryKey: ['username-check', debounced],
  1016→    queryFn: async (): Promise<{ available: boolean; suggestion: string | null }> => {
  1017→      return apiJson<{ available: boolean; suggestion: string | null }>(
  1018→        `/api/users/check-username?username=${encodeURIComponent(debounced)}`,
  1019→      )
  1020→    },
  1021→    enabled: checkEnabled,
  1022→    staleTime: 5_000,
  1023→  })
  1024→
  1025→  const checkStale = !checkEnabled || debounced !== trimmed
  1026→  const checking = valid && changed && (checkStale || checkQ.isFetching)
  1027→  const available = valid && changed && !checkStale && checkQ.data?.available === true
  1028→  const taken = valid && changed && !checkStale && checkQ.data?.available === false
  1029→
  1030→  useEffect(() => {
  1031→    const timer = setTimeout(() => setDebounced(trimmed), HANDLE_DEBOUNCE_MS)
  1032→    return () => clearTimeout(timer)
  1033→  }, [trimmed])
  1034→
  1035→  const save = useMutation({
  1036→    mutationFn: async (): Promise<AppUser> => {
  1037→      const res = await apiJson<{ user: AppUser }>(
  1038→        `/api/users/${encodeURIComponent(me.id)}`,
  1039→        {
  1040→          method: 'PATCH',
  1041→          body: JSON.stringify({ username: trimmed }),
  1042→        },
  1043→      )
  1044→      return res.user
  1045→    },
  1046→    onSuccess: (user) => {
  1047→      onSaved(user)
  1048→      toast.success(user.username ? `@${user.username} is yours now` : 'Handle saved')
  1049→      onClose()
  1050→    },
  1051→    onError: async (error: Error) => {
  1052→      toast.error(error.message || 'Could not save the handle')
  1053→      if (error instanceof ApiError && error.status === 409) {
  1054→        // the PATCH route's 409 carries a suggestion but ApiError drops it —
  1055→        // re-ask the availability endpoint for the nearest free variant
  1056→        try {
  1057→          const res = await apiJson<{ available: boolean; suggestion: string | null }>(
  1058→            `/api/users/check-username?username=${encodeURIComponent(trimmed)}`,
  1059→          )
  1060→          setClashSuggestion(res.suggestion)
  1061→        } catch {
  1062→          setClashSuggestion(null)
  1063→        }
  1064→      }
  1065→    },
  1066→  })
  1067→
  1068→  return (
  1069→    <DialogContent className="max-w-[340px] gap-3 rounded-2xl border-zinc-200 bg-white p-5 dark:border-zinc-800 dark:bg-zinc-900">
  1070→        <DialogHeader className="text-left">
  1071→          <DialogTitle className="text-base tracking-tight">Your @handle</DialogTitle>
  1072→          <DialogDescription className="text-xs leading-relaxed">
  1073→            3–20 characters: lowercase letters, digits, underscore. Friends can find you by it.
  1074→          </DialogDescription>
  1075→        </DialogHeader>
  1076→
  1077→        <div className="space-y-1.5">
  1078→          <div className="relative">
  1079→            <span
  1080→              aria-hidden
  1081→              className="pointer-events-none absolute left-3.5 top-1/2 -translate-y-1/2 text-[15px] font-semibold text-zinc-400 dark:text-zinc-500"
  1082→            >
  1083→              @
  1084→            </span>
  1085→            <Input
  1086→              value={value}
  1087→              onChange={(e) => {
  1088→                setClashSuggestion(null)
  1089→                setValue(sanitizeHandleInput(e.target.value))
  1090→              }}
  1091→              placeholder="e.g. alice_chen"
  1092→              maxLength={HANDLE_MAX}
  1093→              autoComplete="off"
  1094→              autoCorrect="off"
  1095→              autoCapitalize="none"
  1096→              spellCheck={false}
  1097→              enterKeyHint="done"
  1098→              aria-label="Your handle"
  1099→              aria-invalid={(taken || clashSuggestion !== null) || undefined}
  1100→              aria-describedby="handle-editor-availability"
  1101→              className={cn(
  1102→                'h-11 rounded-xl border-zinc-200 bg-zinc-50 pl-8 text-[15px] focus-visible:ring-emerald-500/60 dark:border-zinc-700 dark:bg-zinc-800',
  1103→                (taken || clashSuggestion !== null) &&
  1104→                  'border-amber-400 focus-visible:ring-amber-500/50 dark:border-amber-500/60',
  1105→              )}
  1106→            />
  1107→          </div>
  1108→
  1109→          <p
  1110→            id="handle-editor-availability"
  1111→            role="status"
  1112→            aria-live="polite"
  1113→            className="flex min-h-[18px] flex-wrap items-center gap-1.5 text-xs font-medium"
  1114→          >
  1115→            {trimmed.length === 0 ? (
  1116→              <span className="text-zinc-400 dark:text-zinc-500">Type a handle, or leave empty.</span>
  1117→            ) : !valid ? (
  1118→              <span className="text-zinc-500 dark:text-zinc-400">
  1119→                {HANDLE_MIN}–{HANDLE_MAX} characters: a-z, 0-9, underscore.
  1120→              </span>
  1121→            ) : checking ? (
  1122→              <span className="flex items-center gap-1.5 text-zinc-500 dark:text-zinc-400">
  1123→                <LoaderCircle className="size-3.5 animate-spin" aria-hidden />
  1124→                Checking @{trimmed}…
  1125→              </span>
  1126→            ) : available ? (
  1127→              <span className="flex items-center gap-1.5 text-emerald-600 dark:text-emerald-400">
  1128→                <Check className="size-3.5" strokeWidth={3} aria-hidden />
  1129→                @{trimmed} is free!
  1130→              </span>
  1131→            ) : trimmed === (me.username ?? '') ? (
  1132→              <span className="flex items-center gap-1.5 text-zinc-500 dark:text-zinc-400">
  1133→                <Check className="size-3.5" strokeWidth={3} aria-hidden />
  1134→                That&apos;s your current handle
  1135→              </span>
  1136→            ) : taken ? (
  1137→              <span className="flex flex-wrap items-center gap-1.5 text-amber-600 dark:text-amber-400">
  1138→                <X className="size-3.5" strokeWidth={3} aria-hidden />
  1139→                @{trimmed} is taken
  1140→                {checkQ.data?.suggestion ? (
  1141→                  <button
  1142→                    type="button"
  1143→                    onClick={() => setValue(sanitizeHandleInput(checkQ.data?.suggestion ?? ''))}
  1144→                    className="ml-1 rounded-full bg-amber-500/15 px-2 py-0.5 text-[11px] font-bold text-amber-700 outline-none transition-colors hover:bg-amber-500/25 active:scale-95 dark:text-amber-300"
  1145→                  >
  1146→                    Use @{checkQ.data.suggestion}
  1147→                  </button>
  1148→                ) : null}
  1149→              </span>
  1150→            ) : clashSuggestion ? (
  1151→              <span className="flex flex-wrap items-center gap-1.5 text-amber-600 dark:text-amber-400">
  1152→                <X className="size-3.5" strokeWidth={3} aria-hidden />
  1153→                @{trimmed} was just taken
  1154→                <button
  1155→                  type="button"
  1156→                  onClick={() => setValue(sanitizeHandleInput(clashSuggestion))}
  1157→                  className="ml-1 rounded-full bg-amber-500/15 px-2 py-0.5 text-[11px] font-bold text-amber-700 outline-none transition-colors hover:bg-amber-500/25 active:scale-95 dark:text-amber-300"
  1158→                >
  1159→                  Use @{clashSuggestion}
  1160→                </button>
  1161→              </span>
  1162→            ) : null}
  1163→          </p>
  1164→        </div>
  1165→
  1166→        <div className="flex gap-2">
  1167→          <Button
  1168→            variant="outline"
  1169→            onClick={onClose}
  1170→            className="h-10 flex-1 rounded-xl text-sm font-semibold"
  1171→          >
  1172→            Cancel
  1173→          </Button>
  1174→          <Button
  1175→            onClick={() => save.mutate()}
  1176→            disabled={!valid || checking || taken || !changed || save.isPending}
  1177→            className="h-10 flex-1 rounded-xl bg-emerald-600 text-sm font-semibold text-white hover:bg-emerald-500 active:scale-[0.98]"
  1178→          >
  1179→            {save.isPending ? (
  1180→              <LoaderCircle className="size-4 animate-spin" aria-hidden />
  1181→            ) : (
  1182→              'Save handle'
  1183→            )}
  1184→          </Button>
  1185→        </div>
  1186→      </DialogContent>
  1187→  )
  1188→}
