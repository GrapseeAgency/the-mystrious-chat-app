    79→/** Modal identity — parameterized variants carry the target member id. */
    80→type SheetModalKind = 'rename' | 'add' | 'leave' | `promote:${string}` | `demote:${string}` | `remove:${string}` | null
    81→
    82→function modalTarget(kind: SheetModalKind): { kind: string; userId: string | null } {
    83→  if (kind === null || !kind.includes(':')) return { kind: kind ?? 'none', userId: null }
    84→  const [k, id] = kind.split(':')
    85→  return { kind: k, userId: id }
    86→}

   124→/** Everything stateful lives here; unmounts with the sheet so nested layers never linger. */
   125→function GroupInfoInner({
   126→  conversationId,
   127→  meId,
   128→  onClose,
   129→}: {
   130→  conversationId: string
   131→  meId: string
   132→  onClose: () => void
   133→}) {
   134→  const queryClient = useQueryClient()
   135→  const [modal, setModal] = useState<SheetModalKind>(null)
   136→  const [profileUser, setProfileUser] = useState<AppUser | null>(null)
   137→
   138→  const detailQ = useQuery({
   139→    queryKey: ['conversation', conversationId],
   140→    queryFn: async (): Promise<ConversationDetail> => {
   141→      const res = await apiJson<ConversationResponse>(
   142→        `/api/conversations/${encodeURIComponent(conversationId)}?userId=${encodeURIComponent(meId)}`,
   143→      )
   144→      return res.conversation
   145→    },
   146→    staleTime: 10_000,
   147→    refetchInterval: 30_000,
   148→  })
   149→
   150→  const detail = detailQ.data ?? null
   151→  const isGroup = detail?.isGroup ?? false
   152→  const myRole: GroupRole = detail?.members.find((m) => m.id === meId)?.role ?? 'member'
   153→  const isAdmin = myRole === 'admin'
   154→  const adminCount = detail?.members.filter((m) => m.role === 'admin').length ?? 0
   155→
   156→  // Escape closes the topmost layer only (modal > profile drawer > sheet).
   157→  useEffect(() => {
   158→    const onKey = (event: KeyboardEvent) => {
   159→      if (event.key !== 'Escape') return
   160→      if (modal !== null || profileUser !== null) return // an inner layer owns this Escape
   161→      onClose()
   162→    }
   163→    window.addEventListener('keydown', onKey)
   164→    return () => window.removeEventListener('keydown', onKey)
   165→  }, [modal, profileUser, onClose])
   166→
   167→  const leaveGroup = useMutation({
   168→    mutationFn: async (): Promise<LeaveResponse> => {
   169→      return apiJson<LeaveResponse>(
   170→        `/api/conversations/${encodeURIComponent(conversationId)}/members`,
   171→        {
   172→          method: 'DELETE',
   173→          body: JSON.stringify({ requesterId: meId }),
   174→        },
   175→      )
   176→    },
   177→    onSuccess: (data) => {
   178→      const promoted = data.promotedUserId
   179→        ? (detail?.members.find((m) => m.id === data.promotedUserId)?.name ?? null)
   180→        : null
   181→      toast.success(
   182→        promoted ? `You left the group — ${promoted} is now an admin` : 'You left the group',
   183→      )
   184→      void queryClient.invalidateQueries({ queryKey: ['conversations', meId] })
   185→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   186→      setModal(null)
   187→      onClose()
   188→    },
   189→    onError: (error: Error) => {
   190→      toast.error(error.message || 'Could not leave the group')
   191→    },
   192→  })
   193→
   194→  return (
   195→    <>
   196→      {/* header */}
   197→      <header className="relative z-10 flex min-h-14 shrink-0 items-center gap-1 border-b border-zinc-200 bg-white/90 px-2 pt-[env(safe-area-inset-top)] backdrop-blur-md dark:border-zinc-800 dark:bg-zinc-900/90">
   198→        <Button
   199→          variant="ghost"
   200→          size="icon"
   201→          aria-label="Back to chat"
   202→          onClick={onClose}
   203→          className="size-10 shrink-0 rounded-full text-zinc-600 hover:bg-transparent hover:text-zinc-900 active:scale-95 dark:text-zinc-300 dark:hover:text-white"
   204→        >
   205→          <ChevronLeft className="size-6" aria-hidden />
   206→        </Button>
   207→        <p className="text-sm font-semibold tracking-tight text-zinc-900 dark:text-zinc-50">
   208→          {isGroup ? 'Group info' : 'Chat info'}
   209→        </p>
   210→      </header>
   211→
   212→      {/* body */}
   213→      {detailQ.isPending ? (
   214→        <InfoSkeleton isGroup={isGroup} />
   215→      ) : detailQ.isError || detail === null ? (
   216→        <div className="flex flex-1 flex-col items-center justify-center gap-3 px-8 text-center">
   217→          <UsersRound className="size-8 text-zinc-300 dark:text-zinc-600" aria-hidden />
   218→          <p className="text-sm font-medium text-zinc-500 dark:text-zinc-400">
   219→            Could not load the conversation info.
   220→          </p>
   221→          <Button
   222→            variant="outline"
   223→            size="sm"
   224→            onClick={() => void detailQ.refetch()}
   225→            className="h-9 rounded-full px-4 text-xs font-semibold"
   226→          >
   227→            Try again
   228→          </Button>
   229→        </div>
   230→      ) : isGroup ? (
   231→        <GroupInfoBody
   232→          detail={detail}
   233→          meId={meId}
   234→          isAdmin={isAdmin}
   235→          adminCount={adminCount}
   236→          onOpenModal={setModal}
   237→          onOpenProfile={setProfileUser}
   238→        />
   239→      ) : (
   240→        <DmInfoBody detail={detail} meId={meId} onOpenProfile={setProfileUser} />
   241→      )}
   242→
   243→      {/* nested modals — absolute inside the sheet so they layer above z-[70] */}
   244→      {detail !== null && isGroup ? (
   245→        <>
   246→          <RenameModal
   247→            open={modal === 'rename'}
   248→            onClose={() => setModal(null)}
   249→            conversationId={conversationId}
   250→            meId={meId}
   251→            currentName={detail.name ?? ''}
   252→          />
   253→          <AddMembersModal
   254→            open={modal === 'add'}
   255→            onClose={() => setModal(null)}
   256→            conversationId={conversationId}
   257→            meId={meId}
   258→            existingIds={detail.members.map((m) => m.id)}
   259→          />
   260→          <LeaveModal
   261→            open={modal === 'leave'}
   262→            onClose={() => setModal(null)}
   263→            groupName={detail.name?.trim() || 'this group'}
   264→            pending={leaveGroup.isPending}
   265→            onConfirm={() => leaveGroup.mutate()}
   266→          />
   267→        </>
   268→      ) : null}
   269→
   270→      {/* role/remove confirms + the member profile drawer (portaled) */}
   271→      <MemberActionHost
   272→        conversationId={conversationId}
   273→        meId={meId}
   274→        members={detail?.members ?? []}
   275→        modal={modal}
   276→        setModal={setModal}
   277→        profileUser={profileUser}
   278→        onCloseProfile={() => setProfileUser(null)}
   279→      />
   280→    </>
   281→  )
   282→}

   517→/**

   523→function PromoteDemoteItems({
   524→  member,
   525→  onOpenModal,
   526→}: {
   527→  member: ConversationDetail['members'][number]
   528→  onOpenModal: (m: SheetModalKind) => void
   529→}) {

   531→    <>

   535→          className="gap-2.5 rounded-lg py-2.5 text-sm"

   538→          Promote to admin

   540→      ) : (
   541→        <DropdownMenuItem
   542→          onSelect={() => onOpenModal(`demote:${member.id}`)}
   543→          className="gap-2.5 rounded-lg py-2.5 text-sm"

   545→          <UsersRound className="size-4 text-zinc-500" aria-hidden />
   546→          Demote to member

   548→      )}

   672→function SheetModal({
   673→  open,
   674→  onClose,
   675→  label,
   676→  children,
   677→}: {
   678→  open: boolean
   679→  onClose: () => void
   680→  label: string
   681→  children: ReactNode
   682→}) {
   683→  useEffect(() => {
   684→    if (!open) return
   685→    const onKey = (event: KeyboardEvent) => {
   686→      if (event.key === 'Escape') onClose()
   687→    }
   688→    window.addEventListener('keydown', onKey)
   689→    return () => window.removeEventListener('keydown', onKey)
   690→  }, [open, onClose])
   691→
   692→  return (
   693→    <AnimatePresence>
   694→      {open ? (
   695→        <motion.div
   696→          initial={{ opacity: 0 }}
   697→          animate={{ opacity: 1 }}
   698→          exit={{ opacity: 0 }}
   699→          transition={{ duration: 0.16 }}
   700→          className="absolute inset-0 z-20 flex items-center justify-center bg-black/50 p-5 backdrop-blur-[2px]"
   701→          onClick={onClose}
   702→          role="dialog"
   703→          aria-modal="true"
   704→          aria-label={label}
   705→        >
   706→          <motion.div
   707→            initial={{ scale: 0.9, y: 16, opacity: 0 }}
   708→            animate={{ scale: 1, y: 0, opacity: 1 }}
   709→            exit={{ scale: 0.95, y: 8, opacity: 0 }}
   710→            transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   711→            className="w-full max-w-[340px] rounded-2xl border border-zinc-200 bg-white p-5 shadow-2xl dark:border-zinc-800 dark:bg-zinc-900"
   712→            onClick={(event) => event.stopPropagation()}
   713→          >
   714→            {children}
   715→          </motion.div>
   716→        </motion.div>
   717→      ) : null}
   718→    </AnimatePresence>
   719→  )
   720→}

   724→function RenameModal({
   725→  open,
   726→  onClose,
   727→  conversationId,
   728→  meId,
   729→  currentName,
   730→}: {
   731→  open: boolean
   732→  onClose: () => void
   733→  conversationId: string
   734→  meId: string
   735→  currentName: string
   736→}) {
   737→  const queryClient = useQueryClient()
   738→  const [name, setName] = useState(currentName)
   739→
   740→  const trimmed = name.trim()
   741→  const valid = trimmed.length > 0 && trimmed.length <= GROUP_NAME_MAX && trimmed !== currentName
   742→
   743→  const rename = useMutation({
   744→    mutationFn: async (): Promise<ConversationResponse> => {
   745→      return apiJson<ConversationResponse>(
   746→        `/api/conversations/${encodeURIComponent(conversationId)}`,
   747→        { method: 'PATCH', body: JSON.stringify({ requesterId: meId, name: trimmed }) },
   748→      )
   749→    },
   750→    onMutate: async () => {
   751→      await queryClient.cancelQueries({ queryKey: ['conversation', conversationId] })
   752→      const prev = queryClient.getQueryData<ConversationDetail>(['conversation', conversationId])
   753→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   754→        old ? { ...old, name: trimmed } : old,
   755→      )
   756→      return { prev }
   757→    },
   758→    onSuccess: (data) => {
   759→      // route returns the detail built for the requester (me) — safe to cache directly
   760→      queryClient.setQueryData(['conversation', conversationId], data.conversation)
   761→      void queryClient.invalidateQueries({ queryKey: ['conversations', meId] })
   762→      toast.success('Group renamed')
   763→      onClose()
   764→    },
   765→    onError: (error: Error, _vars, ctx) => {
   766→      if (ctx?.prev) queryClient.setQueryData(['conversation', conversationId], ctx.prev)
   767→      toast.error(error.message || 'Could not rename the group')
   768→    },
   769→  })
   770→
   771→  return (
   772→    <SheetModal open={open} onClose={onClose} label="Rename group">
   773→      <form
   774→        onSubmit={(event) => {
   775→          event.preventDefault()
   776→          if (valid && !rename.isPending) rename.mutate()
   777→        }}
   778→      >
   779→        <h3 className="text-base font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   780→          Rename group
   781→        </h3>
   782→        <p className="mt-1 text-xs leading-relaxed text-zinc-500 dark:text-zinc-400">
   783→          Visible to every member. 1–{GROUP_NAME_MAX} characters.
   784→        </p>
   785→        <div className="mt-3 space-y-1.5">
   786→          <Label htmlFor="group-rename-input" className="sr-only">
   787→            Group name
   788→          </Label>
   789→          <Input
   790→            id="group-rename-input"
   791→            value={name}
   792→            maxLength={GROUP_NAME_MAX}
   793→            onChange={(event) => setName(event.target.value.slice(0, GROUP_NAME_MAX))}
   794→            autoFocus
   795→            autoComplete="off"
   796→            enterKeyHint="done"
   797→            aria-invalid={trimmed.length === 0 || undefined}
   798→            className="h-11 rounded-xl border-zinc-200 bg-zinc-50 text-[15px] focus-visible:ring-emerald-500/60 dark:border-zinc-700 dark:bg-zinc-800"
   799→          />
   800→          <p className="text-right text-[10px] font-medium text-zinc-400 dark:text-zinc-500">
   801→            {trimmed.length}/{GROUP_NAME_MAX}
   802→          </p>
   803→        </div>
   804→        <div className="mt-3 flex gap-2">
   805→          <Button
   806→            type="button"
   807→            variant="outline"
   808→            onClick={onClose}
   809→            className="h-10 flex-1 rounded-xl text-sm font-semibold"
   810→          >
   811→            Cancel
   812→          </Button>
   813→          <Button
   814→            type="submit"
   815→            disabled={!valid || rename.isPending}
   816→            className="h-10 flex-1 rounded-xl bg-emerald-600 text-sm font-semibold text-white hover:bg-emerald-500 active:scale-[0.98]"
   817→          >
   818→            {rename.isPending ? <LoaderCircle className="size-4 animate-spin" aria-hidden /> : 'Save name'}
   819→          </Button>
   820→        </div>
   821→      </form>
   822→    </SheetModal>
   823→  )
   824→}

   828→function AddMembersModal({
   829→  open,
   830→  onClose,
   831→  conversationId,
   832→  meId,
   833→  existingIds,
   834→}: {
   835→  open: boolean
   836→  onClose: () => void
   837→  conversationId: string
   838→  meId: string
   839→  existingIds: string[]
   840→}) {
   841→  const queryClient = useQueryClient()
   842→  const [search, setSearch] = useState('')
   843→  const [selected, setSelected] = useState<Set<string>>(new Set())
   844→
   845→  const usersQ = useQuery({
   846→    queryKey: ['users'],
   847→    queryFn: async (): Promise<AppUser[]> => {
   848→      const res = await apiJson<UsersResponse>('/api/users')
   849→      return res.users
   850→    },
   851→    enabled: open,
   852→  })
   853→
   854→  const existingSet = useMemo(() => new Set(existingIds), [existingIds])
   855→  const candidates = useMemo(() => {
   856→    const q = search.trim().toLowerCase()
   857→    return (usersQ.data ?? [])
   858→      .filter((u) => !existingSet.has(u.id))
   859→      .filter(
   860→        (u) =>
   861→          q.length === 0 ||
   862→          u.name.toLowerCase().includes(q) ||
   863→          (u.username ?? '').toLowerCase().includes(q),
   864→      )
   865→  }, [usersQ.data, existingSet, search])
   866→
   867→  const addMembers = useMutation({
   868→    mutationFn: async (userIds: string[]): Promise<AddedResponse> => {
   869→      return apiJson<AddedResponse>(
   870→        `/api/conversations/${encodeURIComponent(conversationId)}/members`,
   871→        jsonBody({ requesterId: meId, userIds }),
   872→      )
   873→    },
   874→    onMutate: async (userIds: string[]) => {
   875→      await queryClient.cancelQueries({ queryKey: ['conversation', conversationId] })
   876→      const prev = queryClient.getQueryData<ConversationDetail>(['conversation', conversationId])
   877→      const now = new Date().toISOString()
   878→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) => {
   879→        if (!old) return old
   880→        const known = new Set(old.members.map((m) => m.id))
   881→        const fresh = (usersQ.data ?? [])
   882→          .filter((u) => userIds.includes(u.id) && !known.has(u.id))
   883→          .map((u) => ({ ...u, lastReadAt: now, role: 'member' as GroupRole }))
   884→        return fresh.length > 0 ? { ...old, members: [...old.members, ...fresh] } : old
   885→      })
   886→      return { prev }
   887→    },
   888→    onSuccess: (data) => {
   889→      // route returns the detail built for the requester (me) — safe to cache directly
   890→      queryClient.setQueryData(['conversation', conversationId], data.conversation)
   891→      void queryClient.invalidateQueries({ queryKey: ['conversations', meId] })
   892→      toast.success(
   893→        data.added.length === 1 ? '1 member added' : `${data.added.length} members added`,
   894→      )
   895→      onClose()
   896→    },
   897→    onError: (error: Error, _vars, ctx) => {
   898→      if (ctx?.prev) queryClient.setQueryData(['conversation', conversationId], ctx.prev)
   899→      toast.error(error.message || 'Could not add members')
   900→    },
   901→  })
   902→
   903→  const toggle = (id: string) => {
   904→    setSelected((prev) => {
   905→      const next = new Set(prev)
   906→      if (next.has(id)) next.delete(id)
   907→      else next.add(id)
   908→      return next
   909→    })
   910→  }
   911→
   912→  return (
   913→    <SheetModal open={open} onClose={onClose} label="Add members">
   914→      <div className="flex flex-col">
   915→        <h3 className="text-base font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   916→          Add members
   917→        </h3>
   918→        <p className="mt-1 text-xs leading-relaxed text-zinc-500 dark:text-zinc-400">
   919→          Pick people already on this Pulse. They start with no unread backlog.
   920→        </p>
   921→
   922→        {/* search */}
   923→        <div className="relative mt-3">
   924→          <Search
   925→            className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-zinc-400"
   926→            aria-hidden
   927→          />
   928→          <Input
   929→            value={search}
   930→            onChange={(event) => setSearch(event.target.value.slice(0, 40))}
   931→            placeholder="Search people…"
   932→            aria-label="Search people to add"
   933→            autoComplete="off"
   934→            className="h-10 rounded-xl border-zinc-200 bg-zinc-50 pl-9 text-sm focus-visible:ring-emerald-500/60 dark:border-zinc-700 dark:bg-zinc-800"
   935→          />
   936→        </div>
   937→
   938→        {/* candidates */}
   939→        <div className="pulse-scroll mt-2 max-h-[38dvh] min-h-[120px] space-y-0.5 overflow-y-auto rounded-xl border border-zinc-200 bg-zinc-50/60 p-1.5 dark:border-zinc-800 dark:bg-zinc-800/40">
   940→          {usersQ.isPending ? (
   941→            <div className="space-y-2 p-2" aria-busy="true">
   942→              {Array.from({ length: 4 }).map((_, i) => (
   943→                <div key={i} className="flex items-center gap-3">
   944→                  <Skeleton className="size-9 rounded-full" />
   945→                  <Skeleton className="h-3.5 w-1/2 rounded-md" />
   946→                </div>
   947→              ))}
   948→            </div>
   949→          ) : usersQ.isError ? (
   950→            <p className="px-3 py-6 text-center text-xs text-zinc-500 dark:text-zinc-400">
   951→              Could not load people. Close and try again.
   952→            </p>

   955→              {search.trim().length > 0
   956→                ? 'No one matches that search.'
   957→                : 'Everyone on this Pulse is already here.'}
   958→            </p>
   959→          ) : (
   960→            candidates.map((user) => {
   961→              const checked = selected.has(user.id)
   962→              return (
   963→                <label
   964→                  key={user.id}
   965→                  className="flex min-h-[52px] cursor-pointer items-center gap-3 rounded-xl px-2 py-1.5 transition-colors hover:bg-white dark:hover:bg-zinc-800"
   966→                >
   967→                  <Checkbox

   975→                    <span className="block truncate text-sm font-semibold text-zinc-800 dark:text-zinc-100">
   976→                      {user.name}
   977→                    </span>
   978→                    {user.username ? (
   979→                      <span className="block truncate text-[11px] text-zinc-400 dark:text-zinc-500">
   980→                        @{user.username}
   981→                      </span>
   982→                    ) : null}

   992→
   993→        <div className="mt-3 flex gap-2">
   994→          <Button
   995→            type="button"
   996→            variant="outline"
   997→            onClick={onClose}
