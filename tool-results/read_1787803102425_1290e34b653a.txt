   359→   359→  const sendMessage = useMutation({

   435→   435→  const toggleReaction = useMutation({
   436→   436→    mutationFn: async ({ messageId, emoji }: { messageId: string; emoji: string }) => {

   438→   438→        method: 'POST',

   445→   445→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   446→   446→        old
   447→   447→          ? old.map((m) => {
   448→   448→              if (m.id !== messageId) return m
   449→   449→              const groups = m.reactions.map((g) => ({ ...g, userIds: [...g.userIds] }))
   450→   450→              const mineIdx = groups.findIndex((g) => g.emoji === emoji)
   451→   451→              if (mineIdx >= 0) {
   452→   452→                const group = groups[mineIdx]
   453→   453→                const had = group.userIds.includes(me.id)
   454→   454→                if (had) {
   455→   455→                  group.userIds = group.userIds.filter((id) => id !== me.id)
   456→   456→                } else {
   457→   457→                  group.userIds.push(me.id)
   458→   458→                }
   459→   459→                group.count = group.userIds.length
   460→   460→                if (group.count === 0) groups.splice(mineIdx, 1)
   461→   461→              } else {
   462→   462→                groups.push({ emoji, userIds: [me.id], count: 1 })
   463→   463→              }
   464→   464→              return { ...m, reactions: groups }
   465→   465→            })
   466→   466→          : old,
   467→   467→      )
   468→   468→    },
   469→   469→    onSuccess: (real) => {
   470→   470→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   471→   471→        old ? old.map((m) => (m.id === real.id ? { ...m, reactions: real.reactions } : m)) : old,
   472→   472→      )
   473→   473→    },
   474→   474→    onError: () => {
   475→   475→      toast.error('Reaction failed — try again')
   476→   476→      queryClient.invalidateQueries({ queryKey: ['messages', conversationId] })
   477→   477→    },
   478→   478→  })

   480→   480→  const handleToggleReaction = useCallback(
   481→   481→    (messageId: string, emoji: string) => {

   484→   484→    },
   485→   485→    [toggleReaction],
   486→   486→  )

   488→   488→  const deleteMessage = useMutation({
   489→   489→    mutationFn: async (messageId: string) => {
   490→   490→      return apiJson<DeleteResponse>(`/api/messages/${encodeURIComponent(messageId)}`, {
   491→   491→        method: 'DELETE',
   492→   492→        headers: { 'Content-Type': 'application/json' },

   494→   494→      })
   495→   495→    },

   498→   498→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   499→   499→        old
   500→   500→          ? old.map((m) => (m.id === messageId ? { ...m, deletedAt: new Date().toISOString() } : m))
   501→   501→          : old,
   502→   502→      )
   503→   503→      return { previous }
   504→   504→    },
   505→   505→    onSuccess: (data) => {
   506→   506→      const real = data.message
   507→   507→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   508→   508→        old ? old.map((m) => (m.id === real.id ? real : m)) : old,
   509→   509→      )
   510→   510→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   511→   511→      toast.success('Message deleted')
   512→   512→    },
   513→   513→    onError: (_error, _messageId, context) => {
   514→   514→      if (context?.previous) {
   515→   515→        queryClient.setQueryData(['messages', conversationId], context.previous)
   516→   516→      }
   517→   517→      toast.error('Could not delete the message')
   518→   518→    },
   519→   519→    onSettled: () => {
   520→   520→      setSelected(null)
   521→   521→      setConfirmingDelete(false)
   522→   522→    },
   523→   523→  })
