   115→function formatVoicems(ms: number): string {
   116→  const total = Math.max(0, Math.round(ms / 1000))
   117→  const m = Math.floor(total / 60)
   118→  const s = total % 60
   119→  return `${m}:${String(s).padStart(2, '0')}`
   120→}

   446→  const sendMessage = useMutation({

   480→        ? {

   492→        deletedAt: null,

   498→        audioPath: audioPath ?? null,
   499→        durationMs: durationMs ?? null,
   500→      }

   514→            !(m.id.startsWith('temp-') && m.content === real.content && m.senderId === real.senderId),

   528→  /** Toggle an emoji reaction (optimistic; server response is truth). */
   529→  const toggleReaction = useMutation({
   530→    mutationFn: async ({ messageId, emoji }: { messageId: string; emoji: string }) => {
   531→      const res = await apiJson<SendResponse>(`/api/messages/${encodeURIComponent(messageId)}/react`, {
   532→        method: 'POST',
   533→        headers: { 'Content-Type': 'application/json' },
   534→        body: JSON.stringify({ userId: me.id, emoji }),
   535→      })
   536→      return res.message
   537→    },
   538→    onMutate: async ({ messageId, emoji }) => {
   539→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   540→        old
   541→          ? old.map((m) => {
   542→              if (m.id !== messageId) return m
   543→              const groups = m.reactions.map((g) => ({ ...g, userIds: [...g.userIds] }))
   544→              const mineIdx = groups.findIndex((g) => g.emoji === emoji)
   545→              if (mineIdx >= 0) {
   546→                const group = groups[mineIdx]
   547→                const had = group.userIds.includes(me.id)
   548→                if (had) {
   549→                  group.userIds = group.userIds.filter((id) => id !== me.id)
   550→                } else {
   551→                  group.userIds.push(me.id)
   552→                }
   553→                group.count = group.userIds.length
   554→                if (group.count === 0) groups.splice(mineIdx, 1)
