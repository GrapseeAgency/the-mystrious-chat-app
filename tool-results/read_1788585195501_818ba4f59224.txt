    92→}) {
    93→  // remount the editor whenever the underlying identity changes,
    94→  // which re-initializes all local field state (no sync effects)
    95→  const identityKey = `${me.id}|${me.name}|${me.about}|${me.color}|${me.username}`

    99→function ProfileEditor({
   100→  me,
   101→  onOpenSavedMessage,
   102→}: {
   103→  me: AppUser
   104→  onOpenSavedMessage?: (conversationId: string, messageId: string) => void
   105→}) {
   106→  const queryClient = useQueryClient()
   107→  const setUser = usePulseSession((s) => s.setUser)
   108→  const reducedMotion = useReducedMotion()
   109→  const { onlineIds } = usePulseRealtime()
   110→  const mounted = useMounted()
   111→
   112→  // ── real sub-page routing — '/profile/edit' is owned here ──
   113→  // open = push the hash, close = history back, and reading the
   114→  // hash on mount makes #/profile/edit a deep-link.
   115→  const { path, navigate, back } = useHashNav()
   116→  const editing = path === '/profile/edit'
   117→
   118→  // edit form state (initialized from the committed identity;
   119→  // the identityKey remount re-syncs it after every save)
   120→  const [name, setName] = useState(me.name)
   121→  const [about, setAbout] = useState(me.about)
   122→  const [color, setColor] = useState<AvatarColor>((me.color as AvatarColor) ?? 'emerald')
   123→  const [statusEmoji, setStatusEmoji] = useState(me.statusEmoji ?? '')
   124→  const [statusText, setStatusText] = useState(me.statusText ?? '')
   125→
   126→  const [savedOpen, setSavedOpen] = useState(false)
   127→  const [handleOpen, setHandleOpen] = useState(false)
   128→  const [switchOpen, setSwitchOpen] = useState(false)
   129→  /** transient flash on the tap-to-copy handle chip */
   130→  const [handleCopied, setHandleCopied] = useState(false)
   131→  const copiedTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   132→  useEffect(() => {
   133→    return () => {
   134→      if (copiedTimer.current) clearTimeout(copiedTimer.current)
   135→    }
   136→  }, [])
   137→
   138→  // ── real data: profile stats + hub wallet + saved library ──
   139→  const statsQ = useQuery({
   140→    queryKey: ['user-stats', me.id],
   141→    queryFn: async (): Promise<UserStats> => {
   142→      const res = await apiJson<{ stats: UserStats }>(`/api/users/${encodeURIComponent(me.id)}/stats`)
   143→      return res.stats
   144→    },
   145→    staleTime: 30_000,
   146→  })
   147→
   148→  const walletQ = useQuery({
   149→    queryKey: ['hub-wallet', me.id],
   150→    queryFn: async () => {
   151→      return apiJson<{ wallet: { coins: number } }>(
   152→        `/api/hub/wallet?userId=${encodeURIComponent(me.id)}`,
   153→      )
   154→    },
   155→    select: (d) => d.wallet,
   156→    staleTime: 30_000,
   157→  })
   158→
   159→  const savedQuery = useQuery({
   160→    queryKey: ['saved', me.id],
   161→    queryFn: async (): Promise<SavedItem[]> => {
   162→      const res = await apiJson<{ items: SavedItem[] }>(`/api/users/${encodeURIComponent(me.id)}/saved`)
   163→      return res.items
   164→    },
   165→    enabled: savedOpen,
   166→  })
   167→
   168→  const iAmOnline = onlineIds.has(me.id)
   169→  const gradient = gradientFor(me.color)
   170→
   171→  const dirty =
   172→    name.trim() !== me.name.trim() ||
   173→    about.trim() !== me.about.trim() ||
   174→    color !== me.color ||
   175→    statusEmoji.trim() !== (me.statusEmoji ?? '') ||
   176→    statusText.trim() !== (me.statusText ?? '')
   177→
   178→  // real save: PATCH /api/users/[id] — optimistic update across the
   179→  // session store + query caches, toast, and rollback on failure
   180→  const saveProfile = useMutation({
   181→    mutationFn: async () => {
   182→      return apiJson<UsersResponse>(`/api/users/${encodeURIComponent(me.id)}`, {
   183→        method: 'PATCH',
   184→        body: JSON.stringify({
   185→          name: name.trim(),
   186→          about: about.trim(),
   187→          color,
   188→          statusEmoji: statusEmoji.trim(),
   189→          statusText: statusText.trim(),
   190→        }),
   191→      })
   192→    },
   193→    onMutate: async () => {
   194→      const optimistic: AppUser = {
   195→        ...me,
   196→        name: name.trim(),
   197→        about: about.trim(),
   198→        color,
   199→        statusEmoji: statusEmoji.trim() || null,
   200→        statusText: statusText.trim() || null,
   201→      }
   202→      setUser(optimistic)
   203→      queryClient.setQueryData(['me', me.id], optimistic)
   204→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   205→        old?.map((u) => (u.id === me.id ? optimistic : u)),
   206→      )
   207→      return null
   208→    },
   209→    onSuccess: (data) => {
   210→      setUser(data.user)
   211→      queryClient.setQueryData(['me', me.id], data.user)
   212→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   213→        old?.map((u) => (u.id === me.id ? data.user : u)),
   214→      )
   215→      toast.success('Profile updated')
   216→      // social-flow: pop the sub-page back to the profile root
   217→      back("/profile")
   218→    },
   219→    onError: (error: Error) => {
   220→      // rollback to the server-known snapshot
   221→      setUser(me)
   222→      queryClient.setQueryData(['me', me.id], me)
   223→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   224→        old?.map((u) => (u.id === me.id ? me : u)),
   225→      )
   226→      toast.error(error.message || 'Could not save your profile')
   227→    },
   228→  })
   229→
   230→  const canSave = dirty && name.trim().length > 0 && !saveProfile.isPending
   231→
   232→  const copyId = async () => {
   233→    haptic(8)
   234→    try {
   235→      await navigator.clipboard.writeText(me.id)
   236→      toast.success('Account ID copied')
   237→    } catch {
   238→      toast.error('Clipboard is unavailable here')
   239→    }
   240→  }
   241→
   242→  const copyHandle = async () => {
   243→    if (!me.username) return
   244→    haptic(8)
   245→    try {
   246→      await navigator.clipboard.writeText(`@${me.username}`)
   247→      setHandleCopied(true)
   248→      if (copiedTimer.current) clearTimeout(copiedTimer.current)
   249→      copiedTimer.current = setTimeout(() => setHandleCopied(false), 1600)
   250→      toast.success('Handle copied')
   251→    } catch {
   252→      toast.error('Clipboard is unavailable here')
   253→    }
   254→  }
   255→
   256→  /** real share: the OS share sheet when available, clipboard fallback */
   257→  const shareProfile = async () => {
   258→    haptic(10)
   259→    if (!me.username) {
   260→      toast.info('Claim a handle first — it is how people find you')
   261→      setHandleOpen(true)
   262→      return
   263→    }
   264→    const text = `Find me on Pulse — @${me.username}`
   265→    try {
   266→      if (typeof navigator.share === 'function') {
   267→        await navigator.share({ text })
   268→        return
   269→      }
   270→      await navigator.clipboard.writeText(text)
   271→      toast.success('Share text copied')
   272→    } catch (error) {
   273→      // user dismissed the share sheet — not an error
   274→      if (error instanceof DOMException && error.name === 'AbortError') return
   275→      toast.error('Could not share right now')
   276→    }
   277→  }
   278→
   279→  const signOut = () => {
   280→    usePulseSession.getState().clear()
   281→    window.location.reload()
   282→  }
   283→
   284→  const memberSinceShort = useMemo(() => {
   285→    const d = new Date(me.createdAt)
   286→    return Number.isNaN(d.getTime())
   287→      ? ''
   288→      : d.toLocaleDateString('en-US', { month: 'short', year: 'numeric' })
   289→  }, [me.createdAt])
   290→
   291→  // Escape closes the edit sub-page (only when no dialog sits on top)
   292→  useEffect(() => {
   293→    if (!editing || handleOpen || savedOpen || switchOpen) return
   294→    const onKey = (e: KeyboardEvent) => {
   295→      if (e.key === "Escape") back("/profile")
   296→    }
   297→    window.addEventListener('keydown', onKey)
   298→    return () => window.removeEventListener('keydown', onKey)
   299→  }, [editing, handleOpen, savedOpen, switchOpen, back])
   300→
   301→  return (
   302→    <div className="absolute inset-0 flex flex-col">
   303→      {/* ── ROOT PAGE ───────────────────────────────────────── */}
   304→      <div className="pulse-scroll relative min-h-0 flex-1 overflow-y-auto overscroll-contain pb-[calc(104px+env(safe-area-inset-bottom))]">
   305→        {/* ── HERO ── */}
   306→        <section aria-label="Profile" className="relative">
   307→          {/* gradient cover from the identity color + glass sheen */}
   308→          <div
   309→            className={cn(
   310→              'relative isolate h-40 overflow-hidden rounded-b-[28px] bg-gradient-to-br sm:h-48',
   311→              gradient,
   312→            )}
   313→          >
   314→            <span
   315→              aria-hidden
   316→              className="absolute inset-0 bg-[radial-gradient(80%_90%_at_18%_0%,rgba(255,255,255,0.32),transparent_55%),radial-gradient(70%_80%_at_88%_18%,rgba(255,255,255,0.16),transparent_50%)]"
   317→            />
   318→            <motion.span
   319→              aria-hidden
   320→              className="absolute -right-10 -top-14 size-44 rounded-full bg-white/15 blur-2xl"
   321→              animate={reducedMotion ? undefined : { scale: [1, 1.12, 1], opacity: [0.7, 1, 0.7] }}
   322→              transition={{ duration: 9, repeat: Infinity, ease: 'easeInOut' }}
   323→            />
   324→            <motion.span
   325→              aria-hidden
   326→              className="absolute -left-12 -bottom-8 size-36 rounded-full bg-black/10 blur-2xl"
   327→              animate={reducedMotion ? undefined : { scale: [1, 1.18, 1], opacity: [0.5, 0.8, 0.5] }}
   328→              transition={{ duration: 11, delay: 1.2, repeat: Infinity, ease: 'easeInOut' }}
   329→            />
   330→          </div>
   331→
   332→          <div className="relative px-4">
   333→            {/* overlapping avatar with specular gradient ring */}
   334→            <motion.div
   335→              initial={reducedMotion ? false : { scale: 0.8, opacity: 0, y: 10 }}
   336→              animate={{ scale: 1, opacity: 1, y: 0 }}
   337→              transition={spring.bouncy}
   338→              className={cn(
   339→                'relative z-10 -mt-12 w-fit rounded-full p-[3px] shadow-lg shadow-black/15',
   340→                'bg-gradient-to-br',
   341→                gradient,
   342→              )}
   343→            >
   344→              <div className="rounded-full bg-white p-[3px] dark:bg-zinc-900">
   345→                <UserAvatar name={me.name} color={me.color} size={92} showPresence online={iAmOnline} />
   346→              </div>
   347→            </motion.div>
   348→
   349→            {/* name + verified-style member badge */}
   350→            <motion.div
   351→              initial={reducedMotion ? false : { opacity: 0, y: 10 }}
   352→              animate={{ opacity: 1, y: 0 }}
   353→              transition={{ duration: 0.3, ease: ease.out, delay: 0.05 }}
   354→              className="mt-3 flex items-center gap-2"
   355→            >
   356→              <h1 className="truncate text-2xl font-black tracking-tight text-zinc-900 dark:text-zinc-50">
   357→                {me.name}
   358→              </h1>
   359→              <span
   360→                title="Registered member"
   361→                className="flex size-5 shrink-0 items-center justify-center rounded-full"
   362→                aria-label="Registered member"
   363→              >
   364→                <BadgeCheck className="size-5 fill-[var(--ui-accent,#10b981)] text-white dark:text-zinc-900" aria-hidden />
   365→              </span>
   366→            </motion.div>
   367→
   368→            {/* @handle — tap to copy (editing lives on the sub-page) */}
   369→            <motion.div
   370→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   371→              animate={{ opacity: 1, y: 0 }}
   372→              transition={{ duration: 0.3, ease: ease.out, delay: 0.09 }}
   373→              className="mt-1.5"
   374→            >
   375→              <motion.button
   376→                type="button"
   377→                whileTap={reducedMotion ? undefined : pressTap}
   378→                transition={pressSpring}
   379→                onClick={me.username ? copyHandle : () => setHandleOpen(true)}
   380→                aria-label={
   381→                  me.username ? `Copy handle @${me.username}` : 'Set your handle'
   382→                }
   383→                className="glass-pill flex min-h-[32px] items-center gap-1.5 rounded-full px-3 py-1 text-xs font-bold text-[var(--ui-accent,#10b981)] outline-none"
   384→              >
   385→                <AtSign className="size-3.5" aria-hidden />
   386→                {me.username ? (
   387→                  handleCopied ? (
   388→                    <>
   389→                      <Check className="size-3.5" strokeWidth={3} aria-hidden />
   390→                      Copied
   391→                    </>
   392→                  ) : (
   393→                    `@${me.username}`
   394→                  )
   395→                ) : (
   396→                  'Set your handle'
   397→                )}
   398→              </motion.button>
   399→            </motion.div>
   400→
   401→            {/* status glyph + text */}
   402→            {me.statusEmoji || me.statusText ? (
   403→              <motion.p
   404→                initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   405→                animate={{ opacity: 1, y: 0 }}
   406→                transition={{ duration: 0.3, ease: ease.out, delay: 0.12 }}
   407→                className="mt-2.5 flex items-center gap-1.5 text-[13px] font-semibold text-zinc-700 dark:text-zinc-200"
   408→              >
   409→                {me.statusEmoji ? (
   410→                  <StatusGlyph value={me.statusEmoji} className="size-4 text-[var(--ui-accent,#10b981)]" />
   411→                ) : null}
   412→                {me.statusText}
   413→              </motion.p>
   414→            ) : null}
   415→
   416→            {/* bio */}
   417→            <motion.p
   418→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   419→              animate={{ opacity: 1, y: 0 }}
   420→              transition={{ duration: 0.3, ease: ease.out, delay: 0.15 }}
   421→              className="mt-1.5 max-w-md text-[13px] leading-relaxed text-zinc-500 dark:text-zinc-400"
   422→            >
   423→              {me.about.trim() || 'No bio yet'}
   424→            </motion.p>
   425→
   426→            {/* quick actions — two designed pills, not a toolbox dump */}
   427→            <motion.div
   428→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   429→              animate={{ opacity: 1, y: 0 }}
   430→              transition={{ duration: 0.3, ease: ease.out, delay: 0.18 }}
   431→              className="mt-4 flex items-center gap-2"
   432→            >
   433→              <motion.button
   434→                type="button"
   435→                whileTap={reducedMotion ? undefined : pressTap}
   436→                transition={pressSpring}
   437→                onClick={() => {
   438→                  haptic(10)
   439→                  navigate('/profile/edit')
   440→                }}
   441→                aria-label="Edit profile"
   442→                className="glass-pill flex h-10 items-center gap-2 rounded-full px-4 text-[13px] font-bold text-[var(--ui-accent,#10b981)] outline-none"
   443→              >
   444→                <Pencil className="size-4" aria-hidden />
   445→                Edit profile
   446→              </motion.button>
   447→              <motion.button
   448→                type="button"
   449→                whileTap={reducedMotion ? undefined : pressTap}
   450→                transition={pressSpring}
   451→                onClick={shareProfile}
   452→                aria-label="Share your profile"
   453→                className="glass-pill flex h-10 items-center gap-2 rounded-full px-4 text-[13px] font-bold text-zinc-600 outline-none dark:text-zinc-300"
   454→              >
   455→                <Share2 className="size-4" aria-hidden />
   456→                Share
   457→              </motion.button>
   458→            </motion.div>
   459→          </div>
   460→        </section>
   461→
   462→        {/* ── STATS — real data only ── */}
   463→        <section aria-label="Your activity" className="mt-5 px-4">
   464→          <div className="glass-deep glass-sheen isolate grid grid-cols-4 gap-1.5 rounded-3xl p-2">
   465→            {statsQ.isPending ? (
   466→              Array.from({ length: 4 }).map((_, i) => (
   467→                <Skeleton key={i} className="h-[58px] rounded-2xl" />
   468→              ))
   469→            ) : (
   470→              <>
   471→                <StatTile delay={0} label="Messages" value={statsQ.isError ? '—' : <CountUp value={statsQ.data?.messages ?? 0} />} />
   472→                <StatTile delay={1} label="Rooms" value={statsQ.isError ? '—' : <CountUp value={statsQ.data?.chats ?? 0} />} />
   473→                <StatTile
   474→                  delay={2}
   475→                  accent
   476→                  label="Coins"
   477→                  value={
   478→                    walletQ.isPending ? (
   479→                      <Skeleton className="h-5 w-10 rounded-md" />
   480→                    ) : walletQ.isError ? (
   481→                      '—'
   482→                    ) : (
   483→                      <span className="inline-flex items-center gap-1">
   484→                        <Coins className="size-3.5" aria-hidden />
   485→                        <CountUp value={walletQ.data?.coins ?? 0} />
   486→                      </span>
   487→                    )
   488→                  }
   489→                />
   490→                <StatTile delay={3} label="Member since" value={memberSinceShort || '—'} />
   491→              </>
   492→            )}
   493→          </div>
   494→        </section>
   495→
   496→        {/* ── SAVED — real starred-message library ── */}
   497→        <div className="px-4">
   498→          <ProfileSection title="Saved" delay={0.02}>
   499→            <ChevronRow
   500→              icon={Star}
   501→              iconClassName="bg-amber-500/12 text-amber-500"
   502→              title="Saved messages"
   503→              description="Long-press any message in a chat, then Save"
   504→              onPress={() => setSavedOpen(true)}
   505→            />
   506→          </ProfileSection>
   507→        </div>
   508→
   509→        {/* ── ACCOUNT ── */}
   510→        <div className="px-4">
   511→          <ProfileSection title="Account" delay={0.06}>
   512→            <div className="p-1">
   513→              <ChevronRow
   514→                icon={Fingerprint}
   515→                title="Copy account ID"
   516→                description={me.id}
   517→                onPress={copyId}
   518→                ariaLabel={`Copy account ID ${me.id}`}
   519→              />
   520→              <ChevronRow
   521→                icon={LogOut}
   522→                destructive
   523→                title="Sign out"
   524→                description="Return to the welcome screen — nothing is deleted"
   525→                onPress={() => setSwitchOpen(true)}
