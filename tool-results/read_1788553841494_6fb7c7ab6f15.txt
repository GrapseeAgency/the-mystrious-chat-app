   852→   852→/**
   853→   853→ * CountUp-style spring counter — the motion value renders straight to the DOM
   854→   854→ * (no per-frame React state), spring-popping on mount and retargeting on change.
   855→   855→ */
   856→   856→function CountUp({ value, cap = false }: { value: number; cap?: boolean }) {
   857→   857→  const reducedMotion = useReducedMotion()
   858→   858→  const count = useMotionValue(0)
   859→   859→  const text = useTransform(count, (v) => {
   860→   860→    const n = Math.round(v)
   861→   861→    return cap && n > 99 ? '99+' : n.toLocaleString('en-US')
   862→   862→  })
   863→   863→
   864→   864→  useEffect(() => {
   865→   865→    if (reducedMotion) {
   866→   866→      count.set(value)
   867→   867→      return
   868→   868→    }
   869→   869→    const controls = animate(count, value, { type: 'spring', stiffness: 140, damping: 26 })
   870→   870→    return () => controls.stop()
   871→   871→  }, [value, reducedMotion, count])
   872→   872→
   873→   873→  return <motion.span>{text}</motion.span>
   874→   874→}

   994→   994→function HandleEditorBody({

  1004→  1004→  const [debounced, setDebounced] = useState(me.username ?? '')
  1005→  1005→  /** suggestion captured from a 409 username_taken response */
  1006→  1006→  const [clashSuggestion, setClashSuggestion] = useState<string | null>(null)
  1007→  1007→
  1008→  1008→  const trimmed = value.trim()
  1009→  1009→  const changed = trimmed !== (me.username ?? '')
  1010→  1010→  const valid = isValidHandle(trimmed)
  1011→  1011→
  1012→  1012→  // live availability — debounced, skipped while re-typing the current handle
  1013→  1013→  const checkEnabled = valid && debounced !== (me.username ?? '')

  1015→  1015→    queryKey: ['username-check', debounced],

  1017→  1017→      return apiJson<{ available: boolean; suggestion: string | null }>(

  1022→  1022→    staleTime: 5_000,
  1023→  1023→  })

  1026→  1026→  const checking = valid && changed && (checkStale || checkQ.isFetching)

  1055→  1055→        // re-ask the availability endpoint for the nearest free variant

  1058→  1058→            `/api/users/check-username?username=${encodeURIComponent(trimmed)}`,
  1059→  1059→          )
  1060→  1060→          setClashSuggestion(res.suggestion)
  1061→  1061→        } catch {
  1062→  1062→          setClashSuggestion(null)

  1064→  1064→      }
