   100→function ProfileEditor({
   101→  me,
   102→  onOpenSavedMessage,
   103→}: {
   104→  me: AppUser
   105→  onOpenSavedMessage?: (conversationId: string, messageId: string) => void
   106→}) {
   107→  const queryClient = useQueryClient()
   108→  const setUser = usePulseSession((s) => s.setUser)
   109→  const reducedMotion = useReducedMotion()
   110→  const { onlineIds } = usePulseRealtime()
   111→  const mounted = useMounted()
   112→
   113→  // ── real sub-page routing — '/profile/edit' is owned here ──
   114→  // open = push the hash, close = history back, and reading the
   115→  // hash on mount makes #/profile/edit a deep-link.
   116→  const { path, navigate, back } = useHashNav()
   117→  const editing = path === '/profile/edit'
   118→
   119→  // edit form state (initialized from the committed identity;
   120→  // the identityKey remount re-syncs it after every save)
   121→  const [name, setName] = useState(me.name)
   122→  const [about, setAbout] = useState(me.about)
   123→  const [color, setColor] = useState<AvatarColor>((me.color as AvatarColor) ?? 'emerald')
   124→  const [statusEmoji, setStatusEmoji] = useState(me.statusEmoji ?? '')
   125→  const [statusText, setStatusText] = useState(me.statusText ?? '')
   126→
   127→  const [savedOpen, setSavedOpen] = useState(false)
   128→  const [handleOpen, setHandleOpen] = useState(false)
   129→  const [switchOpen, setSwitchOpen] = useState(false)
   130→  /** transient flash on the tap-to-copy handle chip */
   131→  const [handleCopied, setHandleCopied] = useState(false)
   132→  const copiedTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   133→  useEffect(() => {
   134→    return () => {
   135→      if (copiedTimer.current) clearTimeout(copiedTimer.current)
   136→    }
   137→  }, [])
   138→
   139→  // ── real data: profile stats + hub wallet + saved library ──
   140→  const statsQ = useQuery({
   141→    queryKey: ['user-stats', me.id],
   142→    queryFn: async (): Promise<UserStats> => {
   143→      const res = await apiJson<{ stats: UserStats }>(`/api/users/${encodeURIComponent(me.id)}/stats`)
   144→      return res.stats
   145→    },
   146→    staleTime: 30_000,
   147→  })
   148→
   149→  const walletQ = useQuery({
   150→    queryKey: ['hub-wallet', me.id],
   151→    queryFn: async () => {
   152→      return apiJson<{ wallet: { coins: number } }>(
   153→        `/api/hub/wallet?userId=${encodeURIComponent(me.id)}`,
   154→      )
   155→    },
   156→    select: (d) => d.wallet,
   157→    staleTime: 30_000,
   158→  })
   159→
   160→  const savedQuery = useQuery({
   161→    queryKey: ['saved', me.id],
   162→    queryFn: async (): Promise<SavedItem[]> => {
   163→      const res = await apiJson<{ items: SavedItem[] }>(`/api/users/${encodeURIComponent(me.id)}/saved`)
   164→      return res.items
   165→    },
   166→    enabled: savedOpen,
   167→  })
   168→
   169→  const iAmOnline = onlineIds.has(me.id)
   170→  const gradient = gradientFor(me.color)
   171→
   172→  const dirty =
   173→    name.trim() !== me.name.trim() ||
   174→    about.trim() !== me.about.trim() ||
   175→    color !== me.color ||
   176→    statusEmoji.trim() !== (me.statusEmoji ?? '') ||
   177→    statusText.trim() !== (me.statusText ?? '')
   178→
   179→  // real save: PATCH /api/users/[id] — optimistic update across the
   180→  // session store + query caches, toast, and rollback on failure
   181→  const saveProfile = useMutation({
   182→    mutationFn: async () => {
   183→      return apiJson<UsersResponse>(`/api/users/${encodeURIComponent(me.id)}`, {
   184→        method: 'PATCH',
   185→        body: JSON.stringify({
   186→          name: name.trim(),
   187→          about: about.trim(),
   188→          color,
   189→          statusEmoji: statusEmoji.trim(),
   190→          statusText: statusText.trim(),
   191→        }),
   192→      })
   193→    },
   194→    onMutate: async () => {
   195→      const optimistic: AppUser = {
   196→        ...me,
   197→        name: name.trim(),
   198→        about: about.trim(),
   199→        color,
   200→        statusEmoji: statusEmoji.trim() || null,
   201→        statusText: statusText.trim() || null,
   202→      }
   203→      setUser(optimistic)
   204→      queryClient.setQueryData(['me', me.id], optimistic)
   205→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   206→        old?.map((u) => (u.id === me.id ? optimistic : u)),
   207→      )
   208→      return null
   209→    },
   210→    onSuccess: (data) => {
   211→      setUser(data.user)
   212→      queryClient.setQueryData(['me', me.id], data.user)
   213→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   214→        old?.map((u) => (u.id === me.id ? data.user : u)),
   215→      )
   216→      toast.success('Profile updated')
   217→      // social-flow: pop the sub-page back to the profile root
   218→      back("/profile")
   219→    },
   220→    onError: (error: Error) => {
   221→      // rollback to the server-known snapshot
   222→      setUser(me)
   223→      queryClient.setQueryData(['me', me.id], me)
   224→      queryClient.setQueryData<AppUser[]>(['users'], (old) =>
   225→        old?.map((u) => (u.id === me.id ? me : u)),
   226→      )
   227→      toast.error(error.message || 'Could not save your profile')
   228→    },
   229→  })
   230→
   231→  const canSave = dirty && name.trim().length > 0 && !saveProfile.isPending
   232→
   233→  const copyId = async () => {
   234→    haptic(8)
   235→    try {
   236→      await navigator.clipboard.writeText(me.id)
   237→      toast.success('Account ID copied')
   238→    } catch {
   239→      toast.error('Clipboard is unavailable here')
   240→    }
   241→  }
   242→
   243→  const copyHandle = async () => {
   244→    if (!me.username) return
   245→    haptic(8)
   246→    try {
   247→      await navigator.clipboard.writeText(`@${me.username}`)
   248→      setHandleCopied(true)
   249→      if (copiedTimer.current) clearTimeout(copiedTimer.current)
   250→      copiedTimer.current = setTimeout(() => setHandleCopied(false), 1600)
   251→      toast.success('Handle copied')
   252→    } catch {
   253→      toast.error('Clipboard is unavailable here')
   254→    }
   255→  }
   256→
   257→  /** real share: the OS share sheet when available, clipboard fallback */
   258→  const shareProfile = async () => {
   259→    haptic(10)
   260→    if (!me.username) {
   261→      toast.info('Claim a handle first — it is how people find you')
   262→      setHandleOpen(true)
   263→      return
   264→    }
   265→    const text = `Find me on Pulse — @${me.username}`
   266→    try {
   267→      if (typeof navigator.share === 'function') {
   268→        await navigator.share({ text })
   269→        return
   270→      }
   271→      await navigator.clipboard.writeText(text)
   272→      toast.success('Share text copied')
   273→    } catch (error) {
   274→      // user dismissed the share sheet — not an error
   275→      if (error instanceof DOMException && error.name === 'AbortError') return
   276→      toast.error('Could not share right now')
   277→    }
   278→  }
   279→
   280→  const signOut = () => {
   281→    usePulseSession.getState().clear()
   282→    window.location.reload()
   283→  }
   284→
   285→  const memberSinceShort = useMemo(() => {
   286→    const d = new Date(me.createdAt)
   287→    return Number.isNaN(d.getTime())
   288→      ? ''
   289→      : d.toLocaleDateString('en-US', { month: 'short', year: 'numeric' })
   290→  }, [me.createdAt])
   291→
   292→  // Escape closes the edit sub-page (only when no dialog sits on top)
   293→  useEffect(() => {
   294→    if (!editing || handleOpen || savedOpen || switchOpen) return
   295→    const onKey = (e: KeyboardEvent) => {
   296→      if (e.key === "Escape") back("/profile")
   297→    }
   298→    window.addEventListener('keydown', onKey)
   299→    return () => window.removeEventListener('keydown', onKey)
   300→  }, [editing, handleOpen, savedOpen, switchOpen, back])
   301→
   302→  return (
   303→    <div className="absolute inset-0 flex flex-col">
   304→      {/* ── ROOT PAGE ───────────────────────────────────────── */}
   305→      <div className="pulse-scroll relative min-h-0 flex-1 overflow-y-auto overscroll-contain pb-[calc(104px+env(safe-area-inset-bottom))]">
   306→        {/* ── HERO ── */}
   307→        <section aria-label="Profile" className="relative">
   308→          {/* gradient cover from the identity color + glass sheen */}
   309→          <div
   310→            className={cn(
   311→              'relative isolate h-40 overflow-hidden rounded-b-[28px] bg-gradient-to-br sm:h-48',
   312→              gradient,
   313→            )}
   314→          >
   315→            <span
   316→              aria-hidden
   317→              className="absolute inset-0 bg-[radial-gradient(80%_90%_at_18%_0%,rgba(255,255,255,0.32),transparent_55%),radial-gradient(70%_80%_at_88%_18%,rgba(255,255,255,0.16),transparent_50%)]"
   318→            />
   319→            <motion.span
   320→              aria-hidden
   321→              className="absolute -right-10 -top-14 size-44 rounded-full bg-white/15 blur-2xl"
   322→              animate={reducedMotion ? undefined : { scale: [1, 1.12, 1], opacity: [0.7, 1, 0.7] }}
   323→              transition={{ duration: 9, repeat: Infinity, ease: 'easeInOut' }}
   324→            />
   325→            <motion.span
   326→              aria-hidden
   327→              className="absolute -left-12 -bottom-8 size-36 rounded-full bg-black/10 blur-2xl"
   328→              animate={reducedMotion ? undefined : { scale: [1, 1.18, 1], opacity: [0.5, 0.8, 0.5] }}
   329→              transition={{ duration: 11, delay: 1.2, repeat: Infinity, ease: 'easeInOut' }}
   330→            />
   331→          </div>
   332→
   333→          <div className="relative px-4">
   334→            {/* overlapping avatar with specular gradient ring */}
   335→            <motion.div
   336→              initial={reducedMotion ? false : { scale: 0.8, opacity: 0, y: 10 }}
   337→              animate={{ scale: 1, opacity: 1, y: 0 }}
   338→              transition={spring.bouncy}
   339→              className={cn(
   340→                'relative z-10 -mt-12 w-fit rounded-full p-[3px] shadow-lg shadow-black/15',
   341→                'bg-gradient-to-br',
   342→                gradient,
   343→              )}
   344→            >
   345→              <div className="rounded-full bg-white p-[3px] dark:bg-zinc-900">
   346→                <UserAvatar name={me.name} color={me.color} avatar={me.avatar} size={92} showPresence online={iAmOnline} />
   347→              </div>
   348→            </motion.div>
   349→
   350→            {/* name + verified-style member badge */}
   351→            <motion.div
   352→              initial={reducedMotion ? false : { opacity: 0, y: 10 }}
   353→              animate={{ opacity: 1, y: 0 }}
   354→              transition={{ duration: 0.3, ease: ease.out, delay: 0.05 }}
   355→              className="mt-3 flex items-center gap-2"
   356→            >
   357→              <h1 className="truncate text-2xl font-black tracking-tight text-zinc-900 dark:text-zinc-50">
   358→                {me.name}
   359→              </h1>
   360→              <span
   361→                title="Registered member"
   362→                className="flex size-5 shrink-0 items-center justify-center rounded-full"
   363→                aria-label="Registered member"
   364→              >
   365→                <BadgeCheck className="size-5 fill-[var(--ui-accent,#10b981)] text-white dark:text-zinc-900" aria-hidden />
   366→              </span>
   367→            </motion.div>
   368→
   369→            {/* @handle — tap to copy (editing lives on the sub-page) */}
   370→            <motion.div
   371→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   372→              animate={{ opacity: 1, y: 0 }}
   373→              transition={{ duration: 0.3, ease: ease.out, delay: 0.09 }}
   374→              className="mt-1.5"
   375→            >
   376→              <motion.button
   377→                type="button"
   378→                whileTap={reducedMotion ? undefined : pressTap}
   379→                transition={pressSpring}
   380→                onClick={me.username ? copyHandle : () => setHandleOpen(true)}
   381→                aria-label={
   382→                  me.username ? `Copy handle @${me.username}` : 'Set your handle'
   383→                }
   384→                className="glass-pill flex min-h-[32px] items-center gap-1.5 rounded-full px-3 py-1 text-xs font-bold text-[var(--ui-accent,#10b981)] outline-none"
   385→              >
   386→                <AtSign className="size-3.5" aria-hidden />
   387→                {me.username ? (
   388→                  handleCopied ? (
   389→                    <>
   390→                      <Check className="size-3.5" strokeWidth={3} aria-hidden />
   391→                      Copied
   392→                    </>
   393→                  ) : (
   394→                    `@${me.username}`
   395→                  )
   396→                ) : (
   397→                  'Set your handle'
   398→                )}
   399→              </motion.button>
   400→            </motion.div>
   401→
   402→            {/* status glyph + text */}
   403→            {me.statusEmoji || me.statusText ? (
   404→              <motion.p
   405→                initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   406→                animate={{ opacity: 1, y: 0 }}
   407→                transition={{ duration: 0.3, ease: ease.out, delay: 0.12 }}
   408→                className="mt-2.5 flex items-center gap-1.5 text-[13px] font-semibold text-zinc-700 dark:text-zinc-200"
   409→              >
   410→                {me.statusEmoji ? (
   411→                  <StatusGlyph value={me.statusEmoji} className="size-4 text-[var(--ui-accent,#10b981)]" />
   412→                ) : null}
   413→                {me.statusText}
   414→              </motion.p>
   415→            ) : null}
   416→
   417→            {/* bio */}
   418→            <motion.p
   419→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   420→              animate={{ opacity: 1, y: 0 }}
   421→              transition={{ duration: 0.3, ease: ease.out, delay: 0.15 }}
   422→              className="mt-1.5 max-w-md text-[13px] leading-relaxed text-zinc-500 dark:text-zinc-400"
   423→            >
   424→              {me.about.trim() || 'No bio yet'}
   425→            </motion.p>
   426→
   427→            {/* quick actions — two designed pills, not a toolbox dump */}
   428→            <motion.div
   429→              initial={reducedMotion ? false : { opacity: 0, y: 6 }}
   430→              animate={{ opacity: 1, y: 0 }}
   431→              transition={{ duration: 0.3, ease: ease.out, delay: 0.18 }}
   432→              className="mt-4 flex items-center gap-2"
   433→            >
   434→              <motion.button
   435→                type="button"
   436→                whileTap={reducedMotion ? undefined : pressTap}
   437→                transition={pressSpring}
   438→                onClick={() => {
   439→                  haptic(10)
   440→                  navigate('/profile/edit')
   441→                }}
   442→                aria-label="Edit profile"
   443→                className="glass-pill flex h-10 items-center gap-2 rounded-full px-4 text-[13px] font-bold text-[var(--ui-accent,#10b981)] outline-none"
   444→              >
   445→                <Pencil className="size-4" aria-hidden />
   446→                Edit profile
   447→              </motion.button>
   448→              <motion.button
   449→                type="button"
   450→                whileTap={reducedMotion ? undefined : pressTap}
   451→                transition={pressSpring}
   452→                onClick={shareProfile}
   453→                aria-label="Share your profile"
   454→                className="glass-pill flex h-10 items-center gap-2 rounded-full px-4 text-[13px] font-bold text-zinc-600 outline-none dark:text-zinc-300"
   455→              >
   456→                <Share2 className="size-4" aria-hidden />
   457→                Share

   460→          </div>

   462→
   463→        {/* ── STATS — real data only ── */}
   464→        <section aria-label="Your activity" className="mt-5 px-4">
   465→          <div className="glass-deep glass-sheen isolate grid grid-cols-4 gap-1.5 rounded-3xl p-2">
   466→            {statsQ.isPending ? (
   467→              Array.from({ length: 4 }).map((_, i) => (
   468→                <Skeleton key={i} className="h-[58px] rounded-2xl" />
   469→              ))
   470→            ) : (
   471→              <>
   472→                <StatTile delay={0} label="Messages" value={statsQ.isError ? '—' : <CountUp value={statsQ.data?.messages ?? 0} />} />
   473→                <StatTile delay={1} label="Rooms" value={statsQ.isError ? '—' : <CountUp value={statsQ.data?.chats ?? 0} />} />
   474→                <StatTile
   475→                  delay={2}
   476→                  accent
   477→                  label="Coins"
   478→                  value={
   479→                    walletQ.isPending ? (
   480→                      <Skeleton className="h-5 w-10 rounded-md" />
   481→                    ) : walletQ.isError ? (
   482→                      '—'
   483→                    ) : (
   484→                      <span className="inline-flex items-center gap-1">
   485→                        <Coins className="size-3.5" aria-hidden />
   486→                        <CountUp value={walletQ.data?.coins ?? 0} />

   497→        {/* ── SAVED — real starred-message library ── */}

   500→            <ChevronRow

   503→              title="Saved messages"
   504→              description="Long-press any message in a chat, then Save"
   505→              onPress={() => setSavedOpen(true)}
   506→            />
   507→          </ProfileSection>
   508→        </div>
   509→
   510→        {/* ── ACCOUNT ── */}
   511→        <div className="px-4">
   512→          <ProfileSection title="Account" delay={0.06}>
   513→            <div className="p-1">
   514→              <ChevronRow
   515→                icon={Fingerprint}
   516→                title="Copy account ID"
   517→                description={me.id}
   518→                onPress={copyId}
   519→                ariaLabel={`Copy account ID ${me.id}`}
   520→              />
   521→              <ChevronRow
   522→                icon={LogOut}
   523→                destructive
