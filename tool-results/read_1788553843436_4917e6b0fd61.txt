   245→  value,
   246→  onChange,
   247→  options,
   248→  ariaLabel,

   250→  value: T

   259→        className="grid h-10 w-full auto-cols-fr grid-flow-col rounded-xl bg-zinc-100 p-1 dark:bg-zinc-800"
   260→      >
   261→        {options.map((o) => (
   262→          <TabsTrigger
   263→            key={o.value}
   264→            value={o.value}
   265→            className="h-8 gap-1.5 rounded-lg text-[12.5px] font-semibold text-zinc-500 data-[state=active]:bg-white data-[state=active]:text-zinc-900 data-[state=active]:shadow-sm dark:text-zinc-400 dark:data-[state=active]:bg-zinc-600/80 dark:data-[state=active]:text-white"
   266→          >
   267→            {o.Icon ? <o.Icon className="size-3.5" aria-hidden /> : null}
   268→            {o.label}

   272→    </Tabs>

   278→export function SettingsScreen({
   279→  open,
   280→  onClose,
   281→  me,
   282→  onOpenHub,
   283→}: {
   284→  open: boolean
   285→  onClose: () => void
   286→  /** session user (falls back to the pulse session store when omitted) */
   287→  me?: AppUser
   288→  /** closes settings and switches the shell to the Hub tab */
   289→  onOpenHub?: () => void
   290→}) {
   291→  const sessionUser = usePulseSession((s) => s.user)
   292→  const user = me ?? sessionUser
   293→  const prefs = usePrefsValues()
   294→  const save = usePrefs((s) => s.save)
   295→  const { theme, setTheme } = useTheme()
   296→  const [navStyle] = useNavStyle()
   297→
   298→  const [level, setLevel] = useState<SettingsLevel>('root')
   299→  const [dir, setDir] = useState(1)
   300→
   301→  const reduced = prefs.reducedMotion
   302→  const spring = reduced ? { duration: 0 } : { type: 'spring' as const, stiffness: 420, damping: 36 }
   303→
   304→  const go = (next: SettingsLevel) => {
   305→    setDir(LEVEL_DEPTH[next] >= LEVEL_DEPTH[level] ? 1 : -1)
   306→    setLevel(next)
   307→  }
   308→
   309→  // real counts from the live DB — fetched only while the panel is open
   310→  const stats = useQuery({
   311→    queryKey: ['user-stats', user?.id ?? '-'],
   312→    enabled: open && level === 'storage' && !!user,
   313→    staleTime: 15_000,
   314→    queryFn: async (): Promise<UserStats> => {
   315→      const res = await apiJson<{ stats: UserStats }>(`/api/users/${encodeURIComponent(user?.id ?? '')}/stats`)
   316→      return res.stats
   317→    },
   318→  })
   319→
   320→  const navMeta = useMemo(() => NAV_STYLE_META.find((m) => m.id === navStyle), [navStyle])
   321→
   322→  const themeValue = theme === 'light' || theme === 'dark' ? theme : 'system'
   323→
   324→  if (!open) return null
   325→
   326→  const statusLine =
   327→    user !== null && (user.statusEmoji !== null || user.statusText !== null)
   328→      ? `${user.statusEmoji ?? ''} ${user.statusText ?? ''}`.trim()
   329→      : (user?.about ?? '')
   330→
   331→  const notifCaption = `Sound ${prefs.notifSound ? 'on' : 'off'} · previews ${prefs.notifPreviews ? 'on' : 'off'}`
   332→  const privacyCaption = `Last seen ${prefs.lastSeenVisible ? 'visible' : 'hidden'} · receipts ${prefs.readReceipts ? 'on' : 'off'}`
   333→  const themeLabel = themeValue === 'system' ? 'System' : themeValue === 'dark' ? 'Dark' : 'Light'
   334→  const wallpaperLabel = WALLPAPERS.find((w) => w.id === prefs.wallpaper)?.label ?? 'None'
   335→  const appearanceCaption = `${themeLabel} · ${wallpaperLabel} wallpaper`
   336→
   337→  return (
   338→    <motion.div
   339→      initial={reduced ? { opacity: 0 } : { opacity: 0, y: 28, scale: 0.99 }}
   340→      animate={{ opacity: 1, y: 0, scale: 1 }}
   341→      exit={{ opacity: 0, y: 18, transition: { duration: 0.14 } }}
   342→      transition={spring}
   343→      className="absolute inset-0 z-[70] bg-background"
   344→      role="dialog"
   345→      aria-modal="true"
   346→      aria-label="Settings"
   347→    >
   348→      <div className="flex h-full flex-col">
   349→        {/* header */}
   350→        <div className="flex h-14 shrink-0 items-center gap-1 border-b border-zinc-200/70 px-2 dark:border-zinc-800/70">
   351→          {level === 'root' ? (
   352→            <>
   353→              <h2 className="pl-3 text-[17px] font-bold tracking-tight text-zinc-900 dark:text-zinc-50">Settings</h2>
   354→              <div className="flex-1" />
   355→              <Button
   356→                variant="ghost"
   357→                size="icon"
   358→                aria-label="Close settings"
   359→                onClick={onClose}
   360→                className="size-10 rounded-full text-zinc-500 hover:bg-zinc-100 active:scale-95 dark:hover:bg-zinc-800"
   361→              >
   362→                <X className="size-[18px]" aria-hidden />
   363→              </Button>
   364→            </>
   365→          ) : (
   366→            <>
   367→              <Button
   368→                variant="ghost"
   369→                size="icon"
   370→                aria-label="Back to settings"
   371→                onClick={() => go('root')}
   372→                className="size-10 rounded-full text-zinc-500 hover:bg-zinc-100 active:scale-95 dark:hover:bg-zinc-800"
   373→              >
   374→                <ChevronLeft className="size-5" aria-hidden />
   375→              </Button>
   376→              <h2 className="text-[16px] font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   377→                {LEVEL_TITLE[level]}
   378→              </h2>
   379→            </>
   380→          )}
   381→        </div>
   382→
   383→        {/* levels */}
   384→        <div className="pulse-scroll min-h-0 flex-1 overflow-y-auto overscroll-contain px-4 pb-12 pt-4">
   385→          <AnimatePresence mode="wait" custom={dir} initial={false}>
   386→            <motion.div
   387→              key={level}
   388→              custom={dir}
   389→              variants={levelVariants}
   390→              initial="enter"
   391→              animate="center"
   392→              exit="exit"
   393→              transition={spring}
   394→              className="min-h-full"
   395→            >
   396→              {level === 'root' ? (
   397→                <RootLevel
   398→                  user={user}
   399→                  statusLine={statusLine}
   400→                  appearanceCaption={appearanceCaption}
   401→                  notifCaption={notifCaption}
   402→                  privacyCaption={privacyCaption}
   403→                  onOpenLevel={go}
   404→                />
   405→              ) : null}
   406→
   407→              {level === 'appearance' ? (
   408→                <AppearanceLevel
   409→                  prefs={prefs}
   410→                  save={save}
   411→                  themeValue={themeValue}
   412→                  setTheme={setTheme}
   413→                  navMeta={navMeta}
   414→                />
   415→              ) : null}
   416→
   417→              {level === 'notifications' ? (
   418→                <div className={cn(GLASS, 'overflow-hidden')}>
   419→                  <ToggleRow
   420→                    Icon={Eye}
   421→                    title="Message previews"
   422→                    description="Show message text in notification banners."
   423→                    checked={prefs.notifPreviews}
   424→                    onCheckedChange={(v) => save({ notifPreviews: v })}
   425→                  />
   426→                  <ToggleRow
   427→                    Icon={Volume2}
   428→                    title="Notification sound"
   429→                    description="Play a soft pop for incoming messages."
   430→                    checked={prefs.notifSound}
   431→                    onCheckedChange={(v) => save({ notifSound: v })}
   432→                  />
   433→                  <ToggleRow
   434→                    Icon={Vibrate}
   435→                    title="Vibration"
   436→                    description="Buzz on incoming messages, where the device supports it."
   437→                    checked={prefs.notifVibrate}
   438→                    onCheckedChange={(v) => save({ notifVibrate: v })}
   439→                  />
   440→                </div>
   441→              ) : null}
   442→
   443→              {level === 'privacy' ? (
   444→                <div className={cn(GLASS, 'overflow-hidden')}>
   445→                  <ToggleRow
   446→                    Icon={Clock3}
   447→                    title="Last seen & online"
   448→                    description="Let people see when you were last active on Pulse."
   449→                    checked={prefs.lastSeenVisible}
   450→                    onCheckedChange={(v) => save({ lastSeenVisible: v })}
   451→                  />
   452→                  <ToggleRow
   453→                    Icon={CheckCheck}
   454→                    title="Read receipts"
   455→                    description="Show others when you've read their messages."
   456→                    checked={prefs.readReceipts}
   457→                    onCheckedChange={(v) => save({ readReceipts: v })}
   458→                  />
   459→                </div>
   460→              ) : null}
   461→
   462→              {level === 'storage' ? <StorageLevel user={user} stats={stats} /> : null}
   463→
   464→              {level === 'hub' ? <HubLevel onOpenHub={onOpenHub} /> : null}
   465→
   466→              {level === 'about' ? <AboutLevel /> : null}
   467→            </motion.div>
   468→          </AnimatePresence>
   469→        </div>
   470→      </div>
   471→    </motion.div>
   472→  )
   473→}

   694→  user,
   695→  stats,
   696→}: {
   697→  user: AppUser | null
   698→  stats: ReturnType<typeof useQuery<UserStats>>
   699→}) {
   700→  if (!user) {
   701→    return (
   702→      <div className={cn(GLASS, 'flex items-center gap-3 p-4')}>
   703→        <TriangleAlert className="size-5 text-amber-500" aria-hidden />
   704→        <p className="text-[13px] font-medium text-zinc-600 dark:text-zinc-300">Sign in to see your stats.</p>
   705→      </div>
   706→    )
   707→  }
   708→
   709→  if (stats.isPending) {
   710→    return (
   711→      <div className="grid grid-cols-3 gap-2" role="status" aria-label="Loading stats">
   712→        {Array.from({ length: 6 }).map((_, i) => (
   713→          <Skeleton key={i} className="h-[92px] rounded-2xl" />
   714→        ))}
   715→      </div>
   716→    )
   717→  }
   718→
   719→  if (stats.isError || !stats.data) {
   720→    return (
   721→      <div className={cn(GLASS, 'flex flex-col items-start gap-2 p-4')}>
   722→        <p className="flex items-center gap-2 text-[13px] font-medium text-zinc-600 dark:text-zinc-300">

   725→        </p>

   728→          Try again
   729→        </Button>
   730→      </div>
   731→    )
   732→  }
   733→
   734→  const s = stats.data

   736→    <div className="flex flex-col gap-5">
   737→      <div>
   738→        <SectionLabel>Your footprint</SectionLabel>
   739→        <div className="grid grid-cols-3 gap-2">
