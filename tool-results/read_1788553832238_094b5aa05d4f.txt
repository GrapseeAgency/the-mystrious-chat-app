    56→/** Shared unread counter from the real conversations API. */
    57→function useUnread(me: AppUser): number {
    58→  const { data } = useQuery({
    59→    queryKey: ['conversations', me.id],
    60→    queryFn: async (): Promise<ConversationSummary[]> => {
    61→      const res = await apiJson<{ conversations: ConversationSummary[] }>(
    62→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
    63→      )
    64→      return res.conversations
    65→    },
    66→    refetchInterval: 25_000,
    67→  })
    68→  return (data ?? []).reduce((sum, c) => sum + c.unreadCount, 0)
    69→}

    71→/** Unread counter — pops (spring) on every value CHANGE via key remount. */
    72→function UnreadBadge({ count, className }: { count: number; className?: string }) {
    73→  const reduced = useReducedMotion()
    74→  if (count <= 0) return null
    75→  return (
    76→    <motion.span
    77→      key={count}
    78→      initial={reduced ? false : { scale: 0.4, opacity: 0 }}
    79→      animate={{ scale: 1, opacity: 1 }}
    80→      transition={reduced ? { duration: 0 } : spring.bouncy}
    81→      className={cn(
    82→        'pointer-events-none absolute -right-2.5 -top-1.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-gradient-to-br from-emerald-500 to-teal-500 px-1.5 text-[10px] font-bold text-white shadow-[0_4px_12px_-2px_rgba(16,185,129,0.65)] ring-2 ring-white dark:ring-zinc-900',
    83→        className,
    84→      )}
    85→    >
    86→      {count > 99 ? '99+' : count}
    87→    </motion.span>
    88→  )
    89→}

    91→/** One capsule destination: press physics, wobble, active pill, staggered label. */
    92→function CapsuleTab({
    93→  id,
    94→  label,
    95→  Icon,
    96→  active,
    97→  unread,
    98→  onSelect,
    99→  layoutId,
   100→}: {
   101→  id: PulseTab
   102→  label: string
   103→  Icon: typeof MessageCircle
   104→  active: boolean
   105→  unread: number
   106→  onSelect: (tab: PulseTab) => void
   107→  layoutId: string
   108→}) {
   109→  const reduced = useReducedMotion()
   110→  const wobble = useAnimationControls()
   111→  const press = useCallback(() => {
   112→    haptic(12)
   113→    onSelect(id)
   114→    if (!reduced) void wobble.start({ rotate: [0, -8, 6, 0] }, { duration: 0.35, ease: ease.out })
   115→  }, [id, onSelect, reduced, wobble])
   116→  return (
   117→    <motion.button
   118→      type="button"
   119→      role="tab"
   120→      aria-selected={active}
   121→      aria-label={label}
   122→      onClick={press}
   123→      whileTap={reduced ? undefined : { scale: 0.88, y: 1 }}
   124→      transition={reduced ? { duration: 0 } : spring.bouncy}
   125→      className="relative flex min-h-[52px] flex-1 touch-manipulation select-none flex-col items-center justify-center gap-[3px] rounded-[22px] outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60"
   126→    >
   127→      {active ? (
   128→        <motion.span
   129→          layoutId={layoutId}
   130→          transition={reduced ? { duration: 0 } : spring.snappy}
   131→          className="absolute inset-0 rounded-[22px] bg-gradient-to-b from-emerald-500/20 to-emerald-500/[0.06] shadow-[0_6px_20px_-6px_rgba(16,185,129,0.55)] ring-1 ring-inset ring-emerald-500/30 dark:from-emerald-400/[0.16] dark:to-emerald-400/[0.05] dark:ring-emerald-400/25"
   132→          style={{ willChange: 'transform' }}
   133→        />
   134→      ) : null}
   135→      <motion.span animate={wobble} className="relative">
   136→        <motion.span
   137→          animate={{ scale: active ? 1.08 : 1, y: active ? -1 : 0 }}
   138→          transition={reduced ? { duration: 0 } : spring.bouncy}
   139→          className="relative block"
   140→        >
   141→          <Icon
   142→            className={cn(
   143→              'size-[22px] transition-colors duration-200',
   144→              active ? 'text-emerald-600 dark:text-emerald-400' : 'text-zinc-500 dark:text-zinc-400',
   145→            )}
   146→            strokeWidth={active ? 2.2 : 1.8}
   147→            aria-hidden
   148→          />
   149→          {id === 'chats' ? <UnreadBadge count={unread} /> : null}
   150→        </motion.span>
   151→      </motion.span>
   152→      <motion.span
   153→        animate={{ y: active ? -1 : 0 }}
   154→        transition={{ ...(reduced ? { duration: 0 } : spring.bouncy), delay: reduced ? 0 : stagger(1, 0.02) }}
   155→        className={cn(
   156→          'text-[10px] leading-none transition-colors duration-200',
   157→          active ? 'font-semibold text-emerald-600 dark:text-emerald-400' : 'font-medium text-zinc-500 dark:text-zinc-400',
   158→        )}
   159→      >
   160→        {label}
   161→      </motion.span>
   162→    </motion.button>
   163→  )
   164→}

   961→interface TabProps {

   967→export type NavContextAction = 'new-chat' | 'search' | 'new-group' | 'settings'

   969→/**
   970→ * One-call navigation: reads the persisted style and renders the matching
   971→ * architecture. `onContextAction` feeds the contextual dock's per-tab chip;
   972→ * `onSearch`/`onSettings` feed the command bar's inline triggers.
   973→ */
   974→export function PulseNavBar({
   975→  me,
   976→  active,
   977→  onChange,
   978→  onSearch,
   979→  onSettings,
   980→  onContextAction,
   981→}: {
   982→  me: AppUser
   983→  active: PulseTab
   984→  onChange: (tab: PulseTab) => void
   985→  onSearch: () => void
   986→  onSettings: () => void
   987→  onContextAction: (action: NavContextAction) => void
   988→}) {
   989→  const [style] = useNavStyle()
   990→  const unread = useUnread(me)
   991→  switch (style) {
   992→    case 'floating-top':
   993→      return <FloatingTopNav active={active} onChange={onChange} unread={unread} />
   994→    case 'floating-dock':
   995→      return <FloatingDock active={active} onChange={onChange} unread={unread} />
   996→    case 'pill':
   997→      return <PillNav active={active} onChange={onChange} unread={unread} />
   998→    case 'bottom-bar':
   999→      return <BottomBar active={active} onChange={onChange} unread={unread} />
  1000→    case 'tab-bar':
  1001→      return <TabBarNav active={active} onChange={onChange} unread={unread} />
  1002→    case 'floating-tab-bar':
  1003→      return <FloatingTabBar active={active} onChange={onChange} unread={unread} />
  1004→    case 'command-bar':
  1005→      return <CommandBarNav active={active} onChange={onChange} unread={unread} onSearch={onSearch} onSettings={onSettings} />
  1006→    case 'rail':
  1007→      return <RailNav active={active} onChange={onChange} unread={unread} />
  1008→    case 'island':
  1009→      return <IslandNav active={active} onChange={onChange} unread={unread} />
  1010→    case 'radial':
