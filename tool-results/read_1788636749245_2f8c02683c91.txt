    75→/** Disappearing-message TTL → short label (mirrors the room header menu). */
    76→function ttlLabel(ttlSeconds: number): string {
    77→  if (ttlSeconds <= 0) return 'Off'
    78→  if (ttlSeconds === 86_400) return '24h'
    79→  if (ttlSeconds === 604_800) return '7d'
    80→  if (ttlSeconds === 2_592_000) return '30d'
    81→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))}d`
    82→}
