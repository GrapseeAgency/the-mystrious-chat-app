   401→   401→/**
   402→   402→ * Snippet with the first match highlighted — clips a ≤64-char window
   403→   403→ * around the hit so long messages stay one tidy line.
   404→   404→ */
   405→   405→function SearchSnippet({ content, query }: { content: string; query: string }) {
   406→   406→  const lower = content.toLowerCase()
   407→   407→  const q = query.toLowerCase()
   408→   408→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   409→   409→  let from = 0
   410→   410→  let clippedHead = false
   411→   411→  if (idx > 28) {
   412→   412→    from = idx - 24
   413→   413→    clippedHead = true
   414→   414→  }
   415→   415→  const end = Math.min(content.length, idx + q.length + 28)
   416→   416→  const clippedTail = end < content.length
   417→   417→  const body = content.slice(from, end)
   418→   418→  const localIdx = idx - from
   419→   419→  return (
   420→   420→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   421→   421→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   422→   422→      {idx >= 0 ? (
   423→   423→        <>
   424→   424→          {body.slice(0, localIdx)}
   425→   425→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   426→   426→            {body.slice(localIdx, localIdx + q.length)}
   427→   427→          </mark>
   428→   428→          {body.slice(localIdx + q.length)}
   429→   429→        </>
   430→   430→      ) : (
   431→   431→        body
   432→   432→      )}
   433→   433→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   434→   434→    </p>
   435→   435→  )
