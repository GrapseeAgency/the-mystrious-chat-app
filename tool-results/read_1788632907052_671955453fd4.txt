   558→/**
   559→ * WeChat-style red packet riding a chat message — the sender's wallet is
   560→ * debited atomically at creation; `slices` holds a precomputed random split
   561→ * (JSON number[] summing exactly to `total`) and `grabbed` is the atomic
   562→ * cursor into it. Unclaimed slices refund to the sender after `expiresAt`.
   563→ */
   564→model RedPacket {
   565→  id        String          @id @default(cuid())
   566→  messageId String          @unique
   567→  senderId  String
   568→  total     Int // PC debited from the sender at creation
   569→  count     Int // number of grab slices
   570→  grabbed   Int             @default(0) // atomic counter — next slice index
   571→  slices    String // JSON number[] — random split summing exactly to total
   572→  note      String          @default("") // festive caption on the bubble
   573→  expiresAt DateTime // createdAt + 24h
   574→  createdAt DateTime        @default(now())
   575→  message   Message         @relation(fields: [messageId], references: [id], onDelete: Cascade)
   576→  grabs     RedPacketGrab[]
   577→}
   578→
   579→model RedPacketGrab {
   580→  id        String    @id @default(cuid())
   581→  packetId  String
   582→  userId    String
   583→  amount    Int
   584→  createdAt DateTime  @default(now())
   585→  packet    RedPacket @relation(fields: [packetId], references: [id], onDelete: Cascade)
   586→
   587→  @@unique([packetId, userId])
   588→  @@index([packetId, createdAt])
   589→}

   591→/**
   592→ * In-chat game match — tic-tac-toe first. `board` is 9 chars
   593→ * (' ' empty / 'X' / 'O'), `playerOId` null = open challenge any room
   594→ * member can claim via the join button on the card.
   595→ */
   596→model GameMatch {
   597→  id             String       @id @default(cuid())
   598→  conversationId String
   599→  game           String       @default("tictactoe")
   600→  playerXId      String // creator plays X
   601→  playerOId      String? // null = open challenge (join button on card)
   602→  board          String       @default("         ")
   603→  turn           String       @default("X") // side to move: "X" | "O"
   604→  status         String       @default("active") // active|x_won|o_won|draw|abandoned
   605→  winnerId       String?
   606→  winLine        String? // JSON number[] — winning triple for the highlight
   607→  moveCount      Int          @default(0)
   608→  createdAt      DateTime     @default(now())
   609→  updatedAt      DateTime     @updatedAt
   610→  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
   611→
   612→  @@index([conversationId, createdAt])
   613→}

   615→/**
   616→ * Group kanban card — three real columns per conversation, position-stable
   617→ * ordering, optional assignee resolved against room members client-side.
   618→ */
   619→model KanbanCard {
   620→  id              String       @id @default(cuid())
   621→  conversationId  String
   622→  title           String
   623→  column          String       @default("todo") // todo|doing|done
   624→  position        Int          @default(0) // stable sort within (conversation, column)
   625→  assigneeId      String?
   626→  createdById     String?
   627→  sourceMessageId String? // Chanty-style provenance: card born from a chat message
   628→  createdAt       DateTime     @default(now())
   629→  updatedAt       DateTime     @updatedAt
   630→  conversation    Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
   631→
   632→  @@index([conversationId, column, position])
   633→}

   635→/**
   636→ * Group calendar event with per-member RSVP.
   637→ */
   638→model GroupEvent {
   639→  id             String       @id @default(cuid())
   640→  conversationId String
   641→  title          String
   642→  description    String       @default("")
   643→  location       String       @default("")
   644→  startsAt       DateTime
   645→  createdById    String?
   646→  createdAt      DateTime     @default(now())
   647→  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
   648→  rsvps          EventRsvp[]
   649→
   650→  @@index([conversationId, startsAt])
   651→}
   652→
   653→model EventRsvp {
   654→  id          String     @id @default(cuid())
   655→  eventId     String
   656→  userId      String
   657→  status      String     @default("going") // going|maybe|no
   658→  checkedInAt DateTime? // BAND-style attendance stamp (set during the event window)
   659→  createdAt   DateTime   @default(now())
   660→  event       GroupEvent @relation(fields: [eventId], references: [id], onDelete: Cascade)
   661→
   662→  @@unique([eventId, userId])
   663→}
