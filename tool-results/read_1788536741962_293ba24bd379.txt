   792→const HANDLE_MIN = 3
   793→const HANDLE_MAX = 20
   794→const HANDLE_RE = /^[a-z0-9_]+$/
   795→const HANDLE_DEBOUNCE_MS = 350
   796→
   797→/** Mirrors normalizeUsername on the server: 3–20 chars, a-z0-9_ */
   798→function isValidHandle(value: string): boolean {
   799→  return value.length >= HANDLE_MIN && value.length <= HANDLE_MAX && HANDLE_RE.test(value)
   800→}
   801→
   802→/** Keep only characters the server would accept. */
   803→function sanitizeHandleInput(value: string): string {
   804→  return value
   805→    .toLowerCase()
   806→    .replace(/[^a-z0-9_]/g, '')
   807→    .slice(0, HANDLE_MAX)
   808→}

   815→}: {

   817→  onOpenChange: (v: boolean) => void

   819→  onSaved: (user: AppUser) => void
   820→}) {

   822→  if (!open) return null
   823→  return (

   830→function HandleEditorBody({
   831→  me,
   832→  onSaved,
   833→  onClose,
   834→}: {
   835→  me: AppUser
   836→  onSaved: (user: AppUser) => void
   837→  onClose: () => void
   838→}) {
   839→  const [value, setValue] = useState(me.username ?? '')
   840→  const [debounced, setDebounced] = useState(me.username ?? '')
   841→  /** suggestion captured from a 409 username_taken response */
   842→  const [clashSuggestion, setClashSuggestion] = useState<string | null>(null)
   843→
   844→  const trimmed = value.trim()
   845→  const changed = trimmed !== (me.username ?? '')
   846→  const valid = isValidHandle(trimmed)
   847→
   848→  // live availability — debounced, skipped while re-typing the current handle
   849→  const checkEnabled = valid && debounced !== (me.username ?? '')
   850→  const checkQ = useQuery({
   851→    queryKey: ['username-check', debounced],
   852→    queryFn: async (): Promise<{ available: boolean; suggestion: string | null }> => {
   853→      return apiJson<{ available: boolean; suggestion: string | null }>(
   854→        `/api/users/check-username?username=${encodeURIComponent(debounced)}`,
   855→      )
   856→    },
   857→    enabled: checkEnabled,
   858→    staleTime: 5_000,
   859→  })
   860→
   861→  const checkStale = !checkEnabled || debounced !== trimmed
   862→  const checking = valid && changed && (checkStale || checkQ.isFetching)
   863→  const available = valid && changed && !checkStale && checkQ.data?.available === true
   864→  const taken = valid && changed && !checkStale && checkQ.data?.available === false
   865→
   866→  useEffect(() => {
   867→    const timer = setTimeout(() => setDebounced(trimmed), HANDLE_DEBOUNCE_MS)
   868→    return () => clearTimeout(timer)
   869→  }, [trimmed])
   870→
   871→  const save = useMutation({
   872→    mutationFn: async (): Promise<AppUser> => {
   873→      const res = await apiJson<{ user: AppUser }>(
   874→        `/api/users/${encodeURIComponent(me.id)}`,
   875→        {
   876→          method: 'PATCH',
   877→          body: JSON.stringify({ username: trimmed }),
   878→        },
   879→      )
   880→      return res.user
   881→    },
   882→    onSuccess: (user) => {
   883→      onSaved(user)
   884→      toast.success(user.username ? `@${user.username} is yours now` : 'Handle saved')
   885→      onClose()
   886→    },
   887→    onError: async (error: Error) => {
   888→      toast.error(error.message || 'Could not save the handle')
   889→      if (error instanceof ApiError && error.status === 409) {
   890→        // the PATCH route's 409 carries a suggestion but ApiError drops it —
   891→        // re-ask the availability endpoint for the nearest free variant
   892→        try {
   893→          const res = await apiJson<{ available: boolean; suggestion: string | null }>(
   894→            `/api/users/check-username?username=${encodeURIComponent(trimmed)}`,
   895→          )
   896→          setClashSuggestion(res.suggestion)
   897→        } catch {
   898→          setClashSuggestion(null)
   899→        }
   900→      }
   901→    },
   902→  })
   903→
   904→  return (
   905→    <DialogContent className="max-w-[340px] gap-3 rounded-2xl border-zinc-200 bg-white p-5 dark:border-zinc-800 dark:bg-zinc-900">
   906→        <DialogHeader className="text-left">
   907→          <DialogTitle className="text-base tracking-tight">Your @handle</DialogTitle>
   908→          <DialogDescription className="text-xs leading-relaxed">
   909→            3–20 characters: lowercase letters, digits, underscore. Friends can find you by it.
   910→          </DialogDescription>
   911→        </DialogHeader>
   912→
   913→        <div className="space-y-1.5">
   914→          <div className="relative">
   915→            <span
   916→              aria-hidden
   917→              className="pointer-events-none absolute left-3.5 top-1/2 -translate-y-1/2 text-[15px] font-semibold text-zinc-400 dark:text-zinc-500"
   918→            >
   919→              @
   920→            </span>
   921→            <Input
   922→              value={value}
   923→              onChange={(e) => {
   924→                setClashSuggestion(null)
   925→                setValue(sanitizeHandleInput(e.target.value))
   926→              }}
   927→              placeholder="e.g. alice_chen"
   928→              maxLength={HANDLE_MAX}
   929→              autoComplete="off"
   930→              autoCorrect="off"
   931→              autoCapitalize="none"
   932→              spellCheck={false}
   933→              enterKeyHint="done"
   934→              aria-label="Your handle"
   935→              aria-invalid={(taken || clashSuggestion !== null) || undefined}
   936→              aria-describedby="handle-editor-availability"
   937→              className={cn(
   938→                'h-11 rounded-xl border-zinc-200 bg-zinc-50 pl-8 text-[15px] focus-visible:ring-emerald-500/60 dark:border-zinc-700 dark:bg-zinc-800',
   939→                (taken || clashSuggestion !== null) &&
   940→                  'border-amber-400 focus-visible:ring-amber-500/50 dark:border-amber-500/60',
   941→              )}
   942→            />
   943→          </div>
   944→
   945→          <p
   946→            id="handle-editor-availability"
   947→            role="status"
   948→            aria-live="polite"
   949→            className="flex min-h-[18px] flex-wrap items-center gap-1.5 text-xs font-medium"
   950→          >
   951→            {trimmed.length === 0 ? (
   952→              <span className="text-zinc-400 dark:text-zinc-500">Type a handle, or leave empty.</span>
   953→            ) : !valid ? (
   954→              <span className="text-zinc-500 dark:text-zinc-400">
   955→                {HANDLE_MIN}–{HANDLE_MAX} characters: a-z, 0-9, underscore.
   956→              </span>
   957→            ) : checking ? (
   958→              <span className="flex items-center gap-1.5 text-zinc-500 dark:text-zinc-400">
   959→                <LoaderCircle className="size-3.5 animate-spin" aria-hidden />
   960→                Checking @{trimmed}…
   961→              </span>
   962→            ) : available ? (
   963→              <span className="flex items-center gap-1.5 text-emerald-600 dark:text-emerald-400">
   964→                <Check className="size-3.5" strokeWidth={3} aria-hidden />
   965→                @{trimmed} is free!
   966→              </span>
   967→            ) : trimmed === (me.username ?? '') ? (
   968→              <span className="flex items-center gap-1.5 text-zinc-500 dark:text-zinc-400">
   969→                <Check className="size-3.5" strokeWidth={3} aria-hidden />
   970→                That&apos;s your current handle
   971→              </span>
   972→            ) : taken ? (
   973→              <span className="flex flex-wrap items-center gap-1.5 text-amber-600 dark:text-amber-400">
   974→                <X className="size-3.5" strokeWidth={3} aria-hidden />
   975→                @{trimmed} is taken
   976→                {checkQ.data?.suggestion ? (
   977→                  <button
