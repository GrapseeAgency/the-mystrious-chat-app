    71→/** Disappearing-message TTL → short label (mirrors the room header menu). */
    72→function ttlLabel(ttlSeconds: number): string {
    73→  if (ttlSeconds <= 0) return 'Off'
    74→  if (ttlSeconds === 86_400) return '24h'
    75→  if (ttlSeconds === 604_800) return '7d'
    76→  if (ttlSeconds === 2_592_000) return '30d'
    77→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))}d`
    78→}

   487→              Disappearing messages
