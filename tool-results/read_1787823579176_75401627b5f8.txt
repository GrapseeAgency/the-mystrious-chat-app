   243→/**
   244→ * Snippet with the first match highlighted — clips a ≤64-char window
   245→ * around the hit so long messages stay one tidy line.
   246→ */
   247→function SearchSnippet({ content, query }: { content: string; query: string }) {
   248→  const lower = content.toLowerCase()
   249→  const q = query.toLowerCase()
   250→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   251→  let from = 0
   252→  let clippedHead = false
   253→  if (idx > 28) {
   254→    from = idx - 24
   255→    clippedHead = true
   256→  }
   257→  const end = Math.min(content.length, idx + q.length + 28)
   258→  const clippedTail = end < content.length
   259→  const body = content.slice(from, end)
   260→  const localIdx = idx - from
   261→  return (
   262→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   263→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   264→      {idx >= 0 ? (
   265→        <>
   266→          {body.slice(0, localIdx)}
   267→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   268→            {body.slice(localIdx, localIdx + q.length)}
   269→          </mark>
   270→          {body.slice(localIdx + q.length)}
   271→        </>
   272→      ) : (
   273→        body
   274→      )}
   275→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   276→    </p>
   277→  )
   278→}

   356→export function ChatsTab({
   357→  me,
   358→  onOpenConversation,
   359→  onOpenContacts,
   360→  onRequestNewChat,
   361→  onGoProfile,
   362→}: {
   363→  me: AppUser
   364→  onOpenConversation: (conversationId: string, unreadAnchorMs: number | null, jumpMessageId?: string) => void
   365→  onOpenContacts: () => void
   366→  onRequestNewChat: () => void
   367→  onGoProfile: () => void
   368→}) {
   369→  const realtime = usePulseRealtime()
   370→  const typersIn = realtime.typersIn
   371→  const onlineIds = realtime.onlineIds
   372→  const [searching, setSearching] = useState(false)
   373→  const [searchQuery, setSearchQuery] = useState('')
   374→  /** debounced mirror of searchQuery feeding the server message search */
   375→  const [deferredQuery, setDeferredQuery] = useState('')
   376→  const deferredTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   377→
   378→  const conversations = useQuery({
   379→    queryKey: ['conversations', me.id],
   380→    queryFn: async (): Promise<ConversationSummary[]> => {
   381→      const res = await apiJson<ConversationsResponse>(
   382→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
   383→      )
   384→      return res.conversations
   385→    },
   386→    refetchInterval: 6_000,
   387→  })
   388→
   389→  // live drafts → "Draft: …" previews in the list (zustand external store)
   390→  const allDrafts = useStore(pulseDraftsStore, (s) => s.drafts)
   391→
   392→  const rows = useMemo(() => {
   393→    return (conversations.data ?? []).map((conv) => {
   394→      const previewInfo = conversationPreview(conv, me.id)
   395→      const other = conv.isGroup ? null : otherMemberOf(conv, me.id)
   396→      const groupName =
   397→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   398→      const displayName = conversationDisplayName(conv, me.id)
   399→      return {
   400→        conv,
   401→        props: {
   402→          id: conv.id,
   403→          isGroup: conv.isGroup,
   404→          name: displayName,
   405→          time: conv.lastMessage
   406→            ? formatListStamp(conv.lastMessage.createdAt)
   407→            : formatListStamp(conv.updatedAt),
   408→          preview: previewInfo.text,
   409→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   410→          previewDeleted: previewInfo.deleted,
   411→          draft: allDrafts[conv.id] ?? null,
   412→          unreadCount: conv.unreadCount,
   413→          dmName: other?.name ?? null,
   414→          dmColor: other?.color ?? 'emerald',
   415→          groupTitle: groupName,
   416→          online: !conv.isGroup && other !== null && onlineIds.has(other.id),
   417→          pinned: conv.pinnedAt !== null,
   418→          muted: conv.mutedUntil !== null && Date.parse(conv.mutedUntil) > Date.now(),
   419→          typing: typersIn(conv.id, me.id).length > 0,
   420→        },
   421→      }
   422→    })
   423→  }, [conversations.data, me.id, onlineIds, typersIn, allDrafts])
   424→
   425→  const filteredRows = useMemo(() => {
   426→    const q = searchQuery.trim().toLowerCase()
   427→    if (!q) return rows
   428→    return rows.filter(({ conv, props }) => {
   429→      if (props.name.toLowerCase().includes(q)) return true
   430→      if (props.draft?.toLowerCase().includes(q)) return true
   431→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   432→      return false
   433→    })
   434→  }, [rows, searchQuery])
   435→
   436→  /** main-list rows exclude the viewer's archived chats */
   437→  const activeRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt === null), [rows])
   438→  const archivedRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt !== null), [rows])
   439→  const archivedUnread = useMemo(
   440→    () => archivedRows.reduce((sum, { conv }) => sum + conv.unreadCount, 0),
   441→    [archivedRows],
   442→  )
   443→  /** archived-chats drawer (WhatsApp-style) */
   444→  const [archivedOpen, setArchivedOpen] = useState(false)
   445→
   446→  /** Freeze "where was I" from the list summary AT TAP TIME (pre-read watermark). */
   447→  const handlePress = useCallback(
   448→    (conv: ConversationSummary) => {
   449→      const mine = conv.members.find((m) => m.id === me.id) as
   450→        | (AppUser & { lastReadAt?: string })
   451→        | undefined
   452→      const wm = mine?.lastReadAt ? Date.parse(mine.lastReadAt) : Number.NaN
   453→      const anchor = conv.unreadCount > 0 && !Number.isNaN(wm) ? wm : null
   454→      onOpenConversation(conv.id, anchor)
   455→    },
   456→    [onOpenConversation, me.id],
   457→  )
   458→
   459→  const closeSearch = useCallback(() => {
   460→    if (deferredTimerRef.current !== null) {
   461→      clearTimeout(deferredTimerRef.current)
   462→      deferredTimerRef.current = null
   463→    }
   464→    setSearching(false)
   465→    setSearchQuery('')
   466→    setDeferredQuery('')
   467→  }, [])
   468→
   469→  /** typing in the search field: instant local filter + 250ms-debounced server search */
   470→  const handleSearchInput = useCallback((value: string) => {
   471→    setSearchQuery(value)
   472→    if (deferredTimerRef.current !== null) clearTimeout(deferredTimerRef.current)
   473→    const v = value.trim()
   474→    deferredTimerRef.current = setTimeout(() => {
   475→      deferredTimerRef.current = null
   476→      setDeferredQuery(v)
   477→    }, 250)
   478→  }, [])
   479→
   480→  const serverSearch = useQuery({
   481→    queryKey: ['global-search', me.id, deferredQuery],
   482→    enabled: searching && deferredQuery.length >= 2,
   483→    staleTime: 15_000,
   484→    queryFn: async (): Promise<SearchResponse> =>
   485→      apiJson<SearchResponse>(
   486→        `/api/search?userId=${encodeURIComponent(me.id)}&q=${encodeURIComponent(deferredQuery)}`,
   487→      ),
   488→  })
   489→  const serverHits = searching && deferredQuery.length >= 2 ? (serverSearch.data?.messages ?? []) : []
   490→
   491→  // ── row long-press action sheet (pin/unpin) ───────────────
   492→  const [sheetConv, setSheetConv] = useState<ConversationSummary | null>(null)
   493→  const queryClient = useQueryClient()
   494→
   495→  const togglePin = useMutation({
   496→    mutationFn: async (conv: ConversationSummary) => {
   497→      return apiJson<{ ok: boolean; pinned: boolean }>(
   498→        `/api/conversations/${encodeURIComponent(conv.id)}/pin`,
   499→        {
   500→          method: 'PATCH',
   501→          headers: { 'Content-Type': 'application/json' },
   502→          body: JSON.stringify({ userId: me.id }),
   503→        },
   504→      )
   505→    },
   506→    onSuccess: (data) => {
   507→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   508→      toast.success(data.pinned ? 'Pinned to top' : 'Unpinned')
   509→      setSheetConv(null)
   510→    },
   511→    onError: () => {
   512→      toast.error('Could not update the pin')
   513→    },
   514→  })
   515→
   516→  /** per-user notification mute — '8h' | '1w' | 'always' | null */
