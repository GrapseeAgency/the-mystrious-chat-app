   278→   278→export function SettingsScreen({
   279→   279→  open,
   280→   280→  onClose,
   281→   281→  me,
   282→   282→  onOpenHub,
   283→   283→}: {
   284→   284→  open: boolean
   285→   285→  onClose: () => void
   286→   286→  /** session user (falls back to the pulse session store when omitted) */
   287→   287→  me?: AppUser
   288→   288→  /** closes settings and switches the shell to the Hub tab */
   289→   289→  onOpenHub?: () => void
   290→   290→}) {
   291→   291→  const sessionUser = usePulseSession((s) => s.user)
   292→   292→  const user = me ?? sessionUser
   293→   293→  const prefs = usePrefsValues()
   294→   294→  const save = usePrefs((s) => s.save)
   295→   295→  const { theme, setTheme } = useTheme()
   296→   296→  const [navStyle] = useNavStyle()
   297→   297→
   298→   298→  const [level, setLevel] = useState<SettingsLevel>('root')
   299→   299→  const [dir, setDir] = useState(1)
   300→   300→
   301→   301→  const reduced = prefs.reducedMotion
   302→   302→  const spring = reduced ? { duration: 0 } : { type: 'spring' as const, stiffness: 420, damping: 36 }
   303→   303→
   304→   304→  const go = (next: SettingsLevel) => {
   305→   305→    setDir(LEVEL_DEPTH[next] >= LEVEL_DEPTH[level] ? 1 : -1)
   306→   306→    setLevel(next)
   307→   307→  }
   308→   308→
   309→   309→  // real counts from the live DB — fetched only while the panel is open
   310→   310→  const stats = useQuery({
   311→   311→    queryKey: ['user-stats', user?.id ?? '-'],
   312→   312→    enabled: open && level === 'storage' && !!user,
   313→   313→    staleTime: 15_000,
   314→   314→    queryFn: async (): Promise<UserStats> => {
   315→   315→      const res = await apiJson<{ stats: UserStats }>(`/api/users/${encodeURIComponent(user?.id ?? '')}/stats`)
   316→   316→      return res.stats
   317→   317→    },
   318→   318→  })
   319→   319→
   320→   320→  const navMeta = useMemo(() => NAV_STYLE_META.find((m) => m.id === navStyle), [navStyle])
   321→   321→
   322→   322→  const themeValue = theme === 'light' || theme === 'dark' ? theme : 'system'
   323→   323→
   324→   324→  if (!open) return null
   325→   325→
   326→   326→  const statusLine =
   327→   327→    user !== null && (user.statusEmoji !== null || user.statusText !== null)
   328→   328→      ? `${user.statusEmoji ?? ''} ${user.statusText ?? ''}`.trim()
   329→   329→      : (user?.about ?? '')
   330→   330→
   331→   331→  const notifCaption = `Sound ${prefs.notifSound ? 'on' : 'off'} · previews ${prefs.notifPreviews ? 'on' : 'off'}`
   332→   332→  const privacyCaption = `Last seen ${prefs.lastSeenVisible ? 'visible' : 'hidden'} · receipts ${prefs.readReceipts ? 'on' : 'off'}`
   333→   333→  const themeLabel = themeValue === 'system' ? 'System' : themeValue === 'dark' ? 'Dark' : 'Light'
   334→   334→  const wallpaperLabel = WALLPAPERS.find((w) => w.id === prefs.wallpaper)?.label ?? 'None'
   335→   335→  const appearanceCaption = `${themeLabel} · ${wallpaperLabel} wallpaper`
   336→   336→
   337→   337→  return (
   338→   338→    <motion.div
   339→   339→      initial={reduced ? { opacity: 0 } : { opacity: 0, y: 28, scale: 0.99 }}
   340→   340→      animate={{ opacity: 1, y: 0, scale: 1 }}
   341→   341→      exit={{ opacity: 0, y: 18, transition: { duration: 0.14 } }}
   342→   342→      transition={spring}
   343→   343→      className="absolute inset-0 z-[70] bg-background"
   344→   344→      role="dialog"
   345→   345→      aria-modal="true"
   346→   346→      aria-label="Settings"
   347→   347→    >
   348→   348→      <div className="flex h-full flex-col">
   349→   349→        {/* header */}
   350→   350→        <div className="flex h-14 shrink-0 items-center gap-1 border-b border-zinc-200/70 px-2 dark:border-zinc-800/70">
   351→   351→          {level === 'root' ? (
   352→   352→            <>
   353→   353→              <h2 className="pl-3 text-[17px] font-bold tracking-tight text-zinc-900 dark:text-zinc-50">Settings</h2>
   354→   354→              <div className="flex-1" />
   355→   355→              <Button
   356→   356→                variant="ghost"
   357→   357→                size="icon"
   358→   358→                aria-label="Close settings"
   359→   359→                onClick={onClose}
   360→   360→                className="size-10 rounded-full text-zinc-500 hover:bg-zinc-100 active:scale-95 dark:hover:bg-zinc-800"
   361→   361→              >
   362→   362→                <X className="size-[18px]" aria-hidden />
   363→   363→              </Button>
   364→   364→            </>
   365→   365→          ) : (
   366→   366→            <>
   367→   367→              <Button
   368→   368→                variant="ghost"
   369→   369→                size="icon"
   370→   370→                aria-label="Back to settings"
   371→   371→                onClick={() => go('root')}
   372→   372→                className="size-10 rounded-full text-zinc-500 hover:bg-zinc-100 active:scale-95 dark:hover:bg-zinc-800"
   373→   373→              >
   374→   374→                <ChevronLeft className="size-5" aria-hidden />
   375→   375→              </Button>
   376→   376→              <h2 className="text-[16px] font-bold tracking-tight text-zinc-900 dark:text-zinc-50">
   377→   377→                {LEVEL_TITLE[level]}
   378→   378→              </h2>
   379→   379→            </>
   380→   380→          )}
   381→   381→        </div>
   382→   382→
   383→   383→        {/* levels */}
   384→   384→        <div className="pulse-scroll min-h-0 flex-1 overflow-y-auto overscroll-contain px-4 pb-12 pt-4">
   385→   385→          <AnimatePresence mode="wait" custom={dir} initial={false}>
   386→   386→            <motion.div
   387→   387→              key={level}
   388→   388→              custom={dir}
   389→   389→              variants={levelVariants}
   390→   390→              initial="enter"
   391→   391→              animate="center"
   392→   392→              exit="exit"
   393→   393→              transition={spring}
   394→   394→              className="min-h-full"
   395→   395→            >
   396→   396→              {level === 'root' ? (
   397→   397→                <RootLevel
   398→   398→                  user={user}
   399→   399→                  statusLine={statusLine}
   400→   400→                  appearanceCaption={appearanceCaption}
   401→   401→                  notifCaption={notifCaption}
   402→   402→                  privacyCaption={privacyCaption}
   403→   403→                  onOpenLevel={go}
   404→   404→                />
   405→   405→              ) : null}
   406→   406→
   407→   407→              {level === 'appearance' ? (
   408→   408→                <AppearanceLevel
   409→   409→                  prefs={prefs}
   410→   410→                  save={save}
   411→   411→                  themeValue={themeValue}
   412→   412→                  setTheme={setTheme}
   413→   413→                  navMeta={navMeta}
   414→   414→                />
   415→   415→              ) : null}
   416→   416→
   417→   417→              {level === 'notifications' ? (
   418→   418→                <div className={cn(GLASS, 'overflow-hidden')}>
   419→   419→                  <ToggleRow
   420→   420→                    Icon={Eye}
   421→   421→                    title="Message previews"
   422→   422→                    description="Show message text in notification banners."
   423→   423→                    checked={prefs.notifPreviews}
   424→   424→                    onCheckedChange={(v) => save({ notifPreviews: v })}
   425→   425→                  />
   426→   426→                  <ToggleRow
   427→   427→                    Icon={Volume2}
   428→   428→                    title="Notification sound"
   429→   429→                    description="Play a soft pop for incoming messages."
   430→   430→                    checked={prefs.notifSound}
   431→   431→                    onCheckedChange={(v) => save({ notifSound: v })}
   432→   432→                  />
   433→   433→                  <ToggleRow
   434→   434→                    Icon={Vibrate}
   435→   435→                    title="Vibration"
   436→   436→                    description="Buzz on incoming messages, where the device supports it."
   437→   437→                    checked={prefs.notifVibrate}
   438→   438→                    onCheckedChange={(v) => save({ notifVibrate: v })}
   439→   439→                  />
   440→   440→                </div>
   441→   441→              ) : null}
   442→   442→
   443→   443→              {level === 'privacy' ? (
   444→   444→                <div className={cn(GLASS, 'overflow-hidden')}>
   445→   445→                  <ToggleRow
   446→   446→                    Icon={Clock3}
   447→   447→                    title="Last seen & online"
   448→   448→                    description="Let people see when you were last active on Pulse."
   449→   449→                    checked={prefs.lastSeenVisible}
   450→   450→                    onCheckedChange={(v) => save({ lastSeenVisible: v })}
   451→   451→                  />
   452→   452→                  <ToggleRow
   453→   453→                    Icon={CheckCheck}
   454→   454→                    title="Read receipts"
   455→   455→                    description="Show others when you've read their messages."
   456→   456→                    checked={prefs.readReceipts}
   457→   457→                    onCheckedChange={(v) => save({ readReceipts: v })}
   458→   458→                  />
   459→   459→                </div>
   460→   460→              ) : null}
   461→   461→
   462→   462→              {level === 'storage' ? <StorageLevel user={user} stats={stats} /> : null}
   463→   463→
   464→   464→              {level === 'hub' ? <HubLevel onOpenHub={onOpenHub} /> : null}
   465→   465→
