   232→   232→/**
   233→   233→ * Snippet with the first match highlighted — clips a ≤64-char window
   234→   234→ * around the hit so long messages stay one tidy line.
   235→   235→ */
   236→   236→function SearchSnippet({ content, query }: { content: string; query: string }) {
   237→   237→  const lower = content.toLowerCase()
   238→   238→  const q = query.toLowerCase()
   239→   239→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   240→   240→  let from = 0
   241→   241→  let clippedHead = false
   242→   242→  if (idx > 28) {
   243→   243→    from = idx - 24
   244→   244→    clippedHead = true
   245→   245→  }
   246→   246→  const end = Math.min(content.length, idx + q.length + 28)
   247→   247→  const clippedTail = end < content.length
   248→   248→  const body = content.slice(from, end)
   249→   249→  const localIdx = idx - from
   250→   250→  return (
   251→   251→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   252→   252→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   253→   253→      {idx >= 0 ? (
   254→   254→        <>
   255→   255→          {body.slice(0, localIdx)}
   256→   256→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   257→   257→            {body.slice(localIdx, localIdx + q.length)}
   258→   258→          </mark>
   259→   259→          {body.slice(localIdx + q.length)}
   260→   260→        </>
   261→   261→      ) : (
   262→   262→        body
   263→   263→      )}
   264→   264→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   265→   265→    </p>
   266→   266→  )
   267→   267→}

   345→   345→export function ChatsTab({
   346→   346→  me,

   352→   352→  me: AppUser
   353→   353→  onOpenConversation: (conversationId: string, unreadAnchorMs: number | null, jumpMessageId?: string) => void
   354→   354→  onOpenContacts: () => void
   355→   355→  onRequestNewChat: () => void
   356→   356→  onGoProfile: () => void
   357→   357→}) {
   358→   358→  const realtime = usePulseRealtime()
   359→   359→  const typersIn = realtime.typersIn
   360→   360→  const onlineIds = realtime.onlineIds
   361→   361→  const [searching, setSearching] = useState(false)
   362→   362→  const [searchQuery, setSearchQuery] = useState('')
   363→   363→  /** debounced mirror of searchQuery feeding the server message search */
   364→   364→  const [deferredQuery, setDeferredQuery] = useState('')
   365→   365→  const deferredTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   366→   366→
   367→   367→  const conversations = useQuery({
   368→   368→    queryKey: ['conversations', me.id],
   369→   369→    queryFn: async (): Promise<ConversationSummary[]> => {
   370→   370→      const res = await apiJson<ConversationsResponse>(
   371→   371→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
   372→   372→      )
   373→   373→      return res.conversations
   374→   374→    },
   375→   375→    refetchInterval: 6_000,
   376→   376→  })
   377→   377→
   378→   378→  const rows = useMemo(() => {
   379→   379→    return (conversations.data ?? []).map((conv) => {
   380→   380→      const previewInfo = conversationPreview(conv, me.id)
   381→   381→      const other = conv.isGroup ? null : otherMemberOf(conv, me.id)
   382→   382→      const groupName =
   383→   383→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   384→   384→      const displayName = conversationDisplayName(conv, me.id)
   385→   385→      return {
   386→   386→        conv,
   387→   387→        props: {
   388→   388→          id: conv.id,
   389→   389→          isGroup: conv.isGroup,
   390→   390→          name: displayName,
   391→   391→          time: conv.lastMessage
   392→   392→            ? formatListStamp(conv.lastMessage.createdAt)
   393→   393→            : formatListStamp(conv.updatedAt),
   394→   394→          preview: previewInfo.text,
   395→   395→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   396→   396→          previewDeleted: previewInfo.deleted,
   397→   397→          unreadCount: conv.unreadCount,
   398→   398→          dmName: other?.name ?? null,
   399→   399→          dmColor: other?.color ?? 'emerald',
   400→   400→          groupTitle: groupName,
   401→   401→          online: !conv.isGroup && other !== null && onlineIds.has(other.id),
   402→   402→          pinned: conv.pinnedAt !== null,
   403→   403→          muted: conv.mutedUntil !== null && Date.parse(conv.mutedUntil) > Date.now(),
   404→   404→          typing: typersIn(conv.id, me.id).length > 0,
   405→   405→        },
   406→   406→      }
   407→   407→    })
   408→   408→  }, [conversations.data, me.id, onlineIds, typersIn])
   409→   409→
   410→   410→  const filteredRows = useMemo(() => {
   411→   411→    const q = searchQuery.trim().toLowerCase()
   412→   412→    if (!q) return rows
   413→   413→    return rows.filter(({ conv, props }) => {
   414→   414→      if (props.name.toLowerCase().includes(q)) return true
   415→   415→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   416→   416→      return false
   417→   417→    })
   418→   418→  }, [rows, searchQuery])
   419→   419→
   420→   420→  /** Freeze "where was I" from the list summary AT TAP TIME (pre-read watermark). */
   421→   421→  const handlePress = useCallback(
   422→   422→    (conv: ConversationSummary) => {
   423→   423→      const mine = conv.members.find((m) => m.id === me.id) as
   424→   424→        | (AppUser & { lastReadAt?: string })
   425→   425→        | undefined
   426→   426→      const wm = mine?.lastReadAt ? Date.parse(mine.lastReadAt) : Number.NaN
   427→   427→      const anchor = conv.unreadCount > 0 && !Number.isNaN(wm) ? wm : null
   428→   428→      onOpenConversation(conv.id, anchor)
   429→   429→    },
   430→   430→    [onOpenConversation, me.id],
   431→   431→  )
   432→   432→
   433→   433→  const closeSearch = useCallback(() => {
   434→   434→    if (deferredTimerRef.current !== null) {
   435→   435→      clearTimeout(deferredTimerRef.current)
   436→   436→      deferredTimerRef.current = null
   437→   437→    }
   438→   438→    setSearching(false)
   439→   439→    setSearchQuery('')
   440→   440→    setDeferredQuery('')
   441→   441→  }, [])
