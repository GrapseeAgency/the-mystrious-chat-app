   402→   402→/**
   403→   403→ * Snippet with the first match highlighted — clips a ≤64-char window
   404→   404→ * around the hit so long messages stay one tidy line.
   405→   405→ */
   406→   406→function SearchSnippet({ content, query }: { content: string; query: string }) {
   407→   407→  const lower = content.toLowerCase()
   408→   408→  const q = query.toLowerCase()
   409→   409→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   410→   410→  let from = 0
   411→   411→  let clippedHead = false
   412→   412→  if (idx > 28) {
   413→   413→    from = idx - 24
   414→   414→    clippedHead = true
   415→   415→  }
   416→   416→  const end = Math.min(content.length, idx + q.length + 28)
   417→   417→  const clippedTail = end < content.length
   418→   418→  const body = content.slice(from, end)
   419→   419→  const localIdx = idx - from
   420→   420→  return (
   421→   421→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   422→   422→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   423→   423→      {idx >= 0 ? (
   424→   424→        <>
   425→   425→          {body.slice(0, localIdx)}
   426→   426→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   427→   427→            {body.slice(localIdx, localIdx + q.length)}
   428→   428→          </mark>
   429→   429→          {body.slice(localIdx + q.length)}
   430→   430→        </>
   431→   431→      ) : (
   432→   432→        body
   433→   433→      )}
