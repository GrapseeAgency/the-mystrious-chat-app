    97→const CLUSTER_WINDOW_MS = 5 * 60 * 1000

   105→export function ChatRoom({
   106→  me,
   107→  conversationId,
   108→  unreadAnchorMs = null,
   109→  onClose,
   110→}: {
   111→  me: AppUser
   112→  conversationId: string
   113→  /** pre-open read watermark frozen by the chats list at tap time (unread divider) */
   114→  unreadAnchorMs?: number | null
   115→  onClose: () => void
   116→}) {
   117→  const queryClient = useQueryClient()
   118→  const realtime = usePulseRealtime()
   119→  const { resolvedTheme } = useTheme()
   120→  const themeMounted = useMounted()
   121→
   122→  // restore persisted draft once per opened conversation
   123→  const [input, setInput] = useState(() => pulseDraftsStore.getState().drafts[conversationId] ?? '')
   124→  const draftTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   125→  const fileInputRef = useRef<HTMLInputElement>(null)
   126→  const [menuOpen, setMenuOpen] = useState(false)
   127→  const [infoOpen, setInfoOpen] = useState(false)
   128→  const [selected, setSelected] = useState<ChatMessage | null>(null)
   129→  const [confirmingDelete, setConfirmingDelete] = useState(false)
   130→  const [showJump, setShowJump] = useState(false)
   131→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null)
   132→  const [lightboxSrc, setLightboxSrc] = useState<string | null>(null)
   133→  /** {message, emoji} → who-reacted sheet */
   134→  const [reactionInfo, setReactionInfo] = useState<{ message: ChatMessage; emoji: string } | null>(null)
   135→  const [sendingImage, setSendingImage] = useState(false)
   136→
   137→  const viewportRef = useRef<HTMLDivElement>(null)
   138→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   139→  const nearBottomRef = useRef(true)
   140→  const longPressRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   141→  /** flipped (inside rAF) after the first history fetch so render stays ref-free */
   142→  const [historyLoaded, setHistoryLoaded] = useState(false)
   143→
   144→  // ── data ───────────────────────────────────────────────────
   145→
   146→  const detail = useQuery({
   147→    queryKey: ['conversation', conversationId],
   148→    queryFn: async (): Promise<ConversationDetail> => {
   149→      const res = await apiJson<DetailResponse>(
   150→        `/api/conversations/${encodeURIComponent(conversationId)}?userId=${encodeURIComponent(me.id)}`,
   151→      )
   152→      return res.conversation
   153→    },
   154→    staleTime: 15_000,
   155→    // safety net: keeps read-ticks/membership fresh even if the socket path degrades
   156→    refetchInterval: 6_000,
   157→  })
   158→
   159→  const messages = useQuery({
   160→    queryKey: ['messages', conversationId],
   161→    queryFn: async (): Promise<ChatMessage[]> => {
   162→      const res = await apiJson<MessagesResponse>(
   163→        `/api/conversations/${encodeURIComponent(conversationId)}/messages?limit=200`,
   164→      )
   165→      return res.messages
   166→    },
   167→    staleTime: 15_000,
   168→    // safety net: messages still arrive within seconds even without websocket realtime
   169→    refetchInterval: 3_500,
   170→  })
   171→
   172→  // ── realtime registration + read receipts ──────────────────
   173→
   174→  useEffect(() => {
   175→    realtime.setActiveConversation(conversationId)
   176→    return () => realtime.setActiveConversation(null)
   177→  }, [conversationId, realtime])
   178→
   179→  /** POST read marker (visible rooms only) then patch local caches. */
   180→  const markVisibleRead = useCallback(() => {
   181→    if (typeof document !== 'undefined' && document.visibilityState !== 'visible') return
   182→    if ((messages.data ?? []).length === 0) return
   183→    fetch(`/api/conversations/${encodeURIComponent(conversationId)}/read`, {
   184→      method: 'POST',
   185→      headers: { 'Content-Type': 'application/json' },
   186→      body: JSON.stringify({ userId: me.id }),
   187→    })
   188→      .then(() => {
   189→        const nowIso = new Date().toISOString()
   190→        queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   191→          old
   192→            ? {
   193→                ...old,
   194→                members: old.members.map((m) =>
   195→                  m.id === me.id ? { ...m, lastReadAt: nowIso } : m,
   196→                ),
   197→              }
   198→            : old,
   199→        )
   200→        // refresh list badges (my unread for this room is now zero)
   201→        queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   202→      })
   203→      .catch(() => undefined)
   204→  }, [conversationId, me.id, messages.data, queryClient])
   205→
   206→  // opening the room + every new batch of messages ⇒ (debounced) read
   207→  useEffect(() => {
   208→    const timer = setTimeout(markVisibleRead, 600)
   209→    return () => clearTimeout(timer)
   210→  }, [markVisibleRead])
   211→
   212→  useEffect(() => {
   213→    const onVisibility = () => {
   214→      if (document.visibilityState === 'visible') markVisibleRead()
   215→    }
   216→    document.addEventListener('visibilitychange', onVisibility)
   217→    return () => document.removeEventListener('visibilitychange', onVisibility)
   218→  }, [markVisibleRead])
   219→
   220→  // ── derived ────────────────────────────────────────────────
   221→
   222→  const detailData = detail.data
   223→  const other = useMemo(
   224→    () => (detailData ? otherMemberOf(detailData, me.id) : null),
   225→    [detailData, me.id],
   226→  )
   227→  const isGroup = detailData?.isGroup ?? false
   228→  const displayName = detailData ? conversationDisplayName(detailData, me.id) : ''
   229→
   230→  const recipients = useMemo(
   231→    () => (detailData ? detailData.members.filter((m) => m.id !== me.id).map((m) => m.id) : []),
   232→    [detailData, me.id],
   233→  )
   234→
   235→  const typers = realtime.typersIn(conversationId, me.id)
   236→  const typerLabel = useMemo(() => {
   237→    if (typers.length === 0) return ''
   238→    if (!isGroup) return 'typing…'
   239→    const names = typers.map((t) => t.userName)
   240→    if (names.length === 1) return `${names[0]} is typing…`
   241→    if (names.length === 2) return `${names[0]} and ${names[1]} are typing…`
   242→    return `${names.length} people are typing…`
   243→  }, [typers, isGroup])
   244→
   245→  const onlineOthers = useMemo(
   246→    () =>
   247→      detailData
   248→        ? detailData.members.filter((m) => m.id !== me.id && realtime.onlineIds.has(m.id)).length
   249→        : 0,
   250→    [detailData, me.id, realtime.onlineIds],
   251→  )
   252→
   253→  /** highest read watermark among OTHER members → double ticks */
   254→  const othersMaxReadMs = useMemo(() => {
   255→    if (!detailData) return Number.NEGATIVE_INFINITY
   256→    let max = Number.NEGATIVE_INFINITY
   257→    for (const m of detailData.members) {
   258→      if (m.id === me.id) continue
   259→      const t = Date.parse(m.lastReadAt)
   260→      if (!Number.isNaN(t) && t > max) max = t
   261→    }
   262→    return max
   263→  }, [detailData, me.id])
   264→
   265→  const items = useMemo<ClusterItem[]>(() => {
   266→    const list = messages.data ?? []
   267→    const now = new Date()
   268→
   269→    interface Entry {
   270→      message: ChatMessage
   271→      head: boolean
   272→    }
   273→    const built: Entry[] = []
   274→    let prev: ChatMessage | null = null
   275→    for (const message of list) {
   276→      const clusterBreak =
   277→        prev === null ||
   278→        !isSameDayIso(prev.createdAt, message.createdAt) ||
   279→        prev.senderId !== message.senderId ||
   280→        Date.parse(message.createdAt) - Date.parse(prev.createdAt) > CLUSTER_WINDOW_MS
   281→      built.push({ message, head: clusterBreak })
   282→      prev = message
   283→    }
   284→
   285→    const out: ClusterItem[] = []
   286→    let dayAnchor: ChatMessage | null = null
   287→    let dividerPlaced = unreadAnchorMs === null
   288→    for (let i = 0; i < built.length; i += 1) {
   289→      const { message, head } = built[i]
   290→      if (dayAnchor === null || !isSameDayIso(dayAnchor.createdAt, message.createdAt)) {
   291→        out.push({ kind: 'day', key: `day-${message.id}`, label: formatDayChip(message.createdAt, now) })
   292→        dayAnchor = message
   293→      }
   294→      if (
   295→        !dividerPlaced &&
   296→        message.senderId !== me.id &&
   297→        message.deletedAt === null &&
   298→        unreadAnchorMs !== null &&
   299→        Date.parse(message.createdAt) > unreadAnchorMs
   300→      ) {
   301→        out.push({ kind: 'unread', key: 'unread-divider' })
   302→        dividerPlaced = true
   303→      }
   304→      out.push({
   305→        kind: 'msg',
   306→        key: message.id,
   307→        message,
   308→        head,
   309→        tail: i === built.length - 1 || built[i + 1].head,
   310→      })
   311→    }
   312→    return out
   313→  }, [messages.data, unreadAnchorMs, me.id])
   314→
   315→  const lastMessageId =
   316→    messages.data && messages.data.length > 0 ? messages.data[messages.data.length - 1].id : null
   317→
   318→  // ── scrolling ──────────────────────────────────────────────
   319→
   320→  const scrollToBottom = useCallback((smooth: boolean) => {
   321→    const el = viewportRef.current
   322→    if (!el) return
   323→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   324→  }, [])
   325→
   326→  useEffect(() => {
   327→    if (lastMessageId === null || !historyLoaded) return
   328→    requestAnimationFrame(() => {
   329→      if (nearBottomRef.current) {
   330→        scrollToBottom(true)
   331→      } else {
   332→        setShowJump(true)
   333→        haptic(10)
   334→      }
   335→    })
   336→  }, [lastMessageId, historyLoaded, scrollToBottom])
   337→
   338→  useEffect(() => {
   339→    if (messages.isSuccess && !historyLoaded) {
   340→      const frame = requestAnimationFrame(() => {
   341→        scrollToBottom(false)
   342→        setHistoryLoaded(true)
   343→      })
   344→      return () => cancelAnimationFrame(frame)
   345→    }
   346→    return undefined
   347→  }, [messages.isSuccess, historyLoaded, scrollToBottom])
   348→
   349→  const handleScroll = useCallback(() => {
   350→    const el = viewportRef.current
   351→    if (!el) return
   352→    const distance = el.scrollHeight - el.scrollTop - el.clientHeight
   353→    nearBottomRef.current = distance < NEAR_BOTTOM_PX
   354→    if (nearBottomRef.current) setShowJump(false)
   355→  }, [])
   356→
   357→  // ── sending ────────────────────────────────────────────────
   358→
   359→  const sendMessage = useMutation({
   360→    mutationFn: async ({
   361→      clientId,
   362→      content,
   363→      replyToId,
   364→      imagePath,
   365→    }: {
   366→      clientId: string
   367→      content: string
   368→      replyToId?: string
   369→      imagePath?: string
   370→    }) => {
   371→      const res = await apiJson<SendResponse>(
   372→        `/api/conversations/${encodeURIComponent(conversationId)}/messages`,
   373→        {
   374→          method: 'POST',
   375→          headers: { 'Content-Type': 'application/json' },
   376→          body: JSON.stringify({
   377→            senderId: me.id,
   378→            content,
   379→            ...(replyToId ? { replyToId } : {}),
   380→            ...(imagePath ? { imagePath } : {}),
   381→          }),
   382→        },
   383→      )
   384→      return { res, clientId }
   385→    },
   386→    onMutate: async ({ clientId, content, replyToId, imagePath }) => {
   387→      const parentSnapshot = replyToId && replyTo && replyTo.id === replyToId
   388→        ? {
   389→            id: replyTo.id,
   390→            content: replyTo.content,
   391→            senderName: replyTo.sender.name,
   392→            deleted: replyTo.deletedAt !== null,
   393→          }
   394→        : null
   395→      const temp: ChatMessage = {
   396→        id: `temp-${clientId}`,
   397→        conversationId,
   398→        senderId: me.id,
   399→        content,
   400→        deletedAt: null,
   401→        createdAt: new Date().toISOString(),
   402→        sender: { id: me.id, name: me.name, color: me.color },
   403→        reactions: [],
   404→        replyTo: parentSnapshot,
   405→        imagePath: imagePath ?? null,
   406→      }
   407→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   408→        old ? [...old, temp] : [temp],
   409→      )
   410→    },
   411→    onSuccess: ({ res, clientId }) => {
   412→      const real = res.message
   413→      setReplyTo(null)
   414→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) => {
   415→        if (!old) return [real]
   416→        const hadReal = old.some((m) => m.id === real.id)
   417→        const cleaned = old.filter(
   418→          (m) =>
   419→            m.id !== `temp-${clientId}` &&
   420→            !(m.id.startsWith('temp-') && m.content === real.content && m.senderId === real.senderId),
   421→        )
   422→        return hadReal ? cleaned : [...cleaned, real]
   423→      })
   424→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   425→    },
   426→    onError: (_error, { clientId }) => {
   427→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   428→        old ? old.filter((m) => m.id !== `temp-${clientId}`) : old,
   429→      )
   430→      toast.error('Message failed to send')
   431→    },
   432→  })
   433→
   434→  /** Toggle an emoji reaction (optimistic; server response is truth). */
   435→  const toggleReaction = useMutation({
   436→    mutationFn: async ({ messageId, emoji }: { messageId: string; emoji: string }) => {
   437→      const res = await apiJson<SendResponse>(`/api/messages/${encodeURIComponent(messageId)}/react`, {
   438→        method: 'POST',
   439→        headers: { 'Content-Type': 'application/json' },
   440→        body: JSON.stringify({ userId: me.id, emoji }),
   441→      })
   442→      return res.message
   443→    },
   444→    onMutate: async ({ messageId, emoji }) => {
   445→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   446→        old
   447→          ? old.map((m) => {
   448→              if (m.id !== messageId) return m
   449→              const groups = m.reactions.map((g) => ({ ...g, userIds: [...g.userIds] }))
   450→              const mineIdx = groups.findIndex((g) => g.emoji === emoji)
   451→              if (mineIdx >= 0) {
   452→                const group = groups[mineIdx]
   453→                const had = group.userIds.includes(me.id)
   454→                if (had) {
   455→                  group.userIds = group.userIds.filter((id) => id !== me.id)
   456→                } else {
   457→                  group.userIds.push(me.id)
   458→                }
   459→                group.count = group.userIds.length
   460→                if (group.count === 0) groups.splice(mineIdx, 1)
   461→              } else {
   462→                groups.push({ emoji, userIds: [me.id], count: 1 })
   463→              }
   464→              return { ...m, reactions: groups }
   465→            })
   466→          : old,
   467→      )
   468→    },
   469→    onSuccess: (real) => {
   470→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   471→        old ? old.map((m) => (m.id === real.id ? { ...m, reactions: real.reactions } : m)) : old,
   472→      )
   473→    },
   474→    onError: () => {
   475→      toast.error('Reaction failed — try again')
   476→      queryClient.invalidateQueries({ queryKey: ['messages', conversationId] })
   477→    },
   478→  })
   479→
   480→  const handleToggleReaction = useCallback(
   481→    (messageId: string, emoji: string) => {
   482→      haptic(12)
   483→      toggleReaction.mutate({ messageId, emoji })
   484→    },
   485→    [toggleReaction],
   486→  )
   487→
   488→  const deleteMessage = useMutation({
   489→    mutationFn: async (messageId: string) => {
   490→      return apiJson<DeleteResponse>(`/api/messages/${encodeURIComponent(messageId)}`, {
   491→        method: 'DELETE',
   492→        headers: { 'Content-Type': 'application/json' },
   493→        body: JSON.stringify({ requesterId: me.id }),
   494→      })
   495→    },
   496→    onMutate: async (messageId) => {
   497→      const previous = queryClient.getQueryData<ChatMessage[]>(['messages', conversationId])
   498→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   499→        old
   500→          ? old.map((m) => (m.id === messageId ? { ...m, deletedAt: new Date().toISOString() } : m))
   501→          : old,
   502→      )
   503→      return { previous }
   504→    },
   505→    onSuccess: (data) => {
   506→      const real = data.message
   507→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   508→        old ? old.map((m) => (m.id === real.id ? real : m)) : old,
   509→      )
   510→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   511→      toast.success('Message deleted')
   512→    },
   513→    onError: (_error, _messageId, context) => {
   514→      if (context?.previous) {
   515→        queryClient.setQueryData(['messages', conversationId], context.previous)
   516→      }
   517→      toast.error('Could not delete the message')
   518→    },
   519→    onSettled: () => {
   520→      setSelected(null)
   521→      setConfirmingDelete(false)
   522→    },
   523→  })
   524→
   525→  // ── composer behaviour ─────────────────────────────────────
   526→
   527→  const autosize = useCallback(() => {
   528→    const el = textareaRef.current
   529→    if (!el) return
   530→    el.style.height = 'auto'
   531→    el.style.height = `${Math.min(el.scrollHeight, 120)}px`
   532→  }, [])
   533→
   534→  const stopTyping = useCallback(() => {
   535→    if (recipients.length > 0) {
   536→      realtime.cancelTyping(conversationId, { viewerId: me.id, recipients })
   537→    }
   538→  }, [realtime, conversationId, me.id, recipients])
   539→
   540→  useEffect(() => () => stopTyping(), [stopTyping])
   541→
   542→  const submit = useCallback(() => {
   543→    const content = input.trim()
   544→    if (content.length === 0 || sendMessage.isPending) return
   545→    stopTyping()
   546→    setInput('')
   547→    pulseDraftsStore.getState().clearDraft(conversationId)
   548→    requestAnimationFrame(autosize)
   549→    sendMessage.mutate({
   550→      clientId: uid(),
   551→      content,
   552→      ...(replyTo && !replyTo.deletedAt ? { replyToId: replyTo.id } : {}),
   553→    })
   554→  }, [input, sendMessage, stopTyping, autosize, replyTo, conversationId])
   555→
   556→  const handleInputChange = (value: string) => {
   557→    setInput(value)
   558→    autosize()
   559→    if (draftTimerRef.current !== null) clearTimeout(draftTimerRef.current)
   560→    draftTimerRef.current = setTimeout(() => {
   561→      draftTimerRef.current = null
   562→      pulseDraftsStore.getState().setDraft(conversationId, value)
   563→    }, 300)
   564→    if (value.trim().length > 0 && recipients.length > 0) {
   565→      realtime.signalTyping(conversationId, {
   566→        viewerId: me.id,
   567→        userName: me.name,
   568→        recipients,
   569→      })
   570→    } else {
   571→      stopTyping()
   572→    }
   573→  }
