   576→/**
   577→ * WeChat-style red packet riding a chat message — the sender's wallet is
   578→ * debited atomically at creation; `slices` holds a precomputed random split
   579→ * (JSON number[] summing exactly to `total`) and `grabbed` is the atomic
   580→ * cursor into it. Unclaimed slices refund to the sender after `expiresAt`.
   581→ */
   582→model RedPacket {
   583→  id        String          @id @default(cuid())
   584→  messageId String          @unique
   585→  senderId  String
   586→  total     Int // PC debited from the sender at creation
   587→  count     Int // number of grab slices
   588→  grabbed   Int             @default(0) // atomic counter — next slice index
   589→  slices    String // JSON number[] — random split summing exactly to total
   590→  note      String          @default("") // festive caption on the bubble
   591→  expiresAt DateTime // createdAt + 24h
   592→  createdAt DateTime        @default(now())
   593→  message   Message         @relation(fields: [messageId], references: [id], onDelete: Cascade)
   594→  grabs     RedPacketGrab[]
   595→}
