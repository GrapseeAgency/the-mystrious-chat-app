   251→/**
   252→ * Snippet with the first match highlighted — clips a ≤64-char window
   253→ * around the hit so long messages stay one tidy line.
   254→ */
   255→function SearchSnippet({ content, query }: { content: string; query: string }) {
   256→  const lower = content.toLowerCase()
   257→  const q = query.toLowerCase()
   258→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   259→  let from = 0
   260→  let clippedHead = false
   261→  if (idx > 28) {
   262→    from = idx - 24
   263→    clippedHead = true
   264→  }
   265→  const end = Math.min(content.length, idx + q.length + 28)
   266→  const clippedTail = end < content.length
   267→  const body = content.slice(from, end)
   268→  const localIdx = idx - from
   269→  return (
   270→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   271→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   272→      {idx >= 0 ? (
   273→        <>
   274→          {body.slice(0, localIdx)}
   275→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   276→            {body.slice(localIdx, localIdx + q.length)}
   277→          </mark>
   278→          {body.slice(localIdx + q.length)}
   279→        </>
   280→      ) : (
   281→        body
   282→      )}
   283→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   284→    </p>
   285→  )
   286→}

   435→export function ChatsTab({
   436→  me,
   437→  onOpenConversation,
   438→  onOpenContacts,
   439→  onRequestNewChat,
   440→  onGoProfile,
   441→}: {
   442→  me: AppUser
   443→  onOpenConversation: (conversationId: string, unreadAnchorMs: number | null, jumpMessageId?: string) => void
   444→  onOpenContacts: () => void
   445→  onRequestNewChat: () => void
   446→  onGoProfile: () => void
   447→}) {
   448→  const realtime = usePulseRealtime()
   449→  const typersIn = realtime.typersIn
   450→  const onlineIds = realtime.onlineIds
   451→  const [searching, setSearching] = useState(false)
   452→  const [searchQuery, setSearchQuery] = useState('')
   453→  /** debounced mirror of searchQuery feeding the server message search */
   454→  const [deferredQuery, setDeferredQuery] = useState('')
   455→  const deferredTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   456→
   457→  const conversations = useQuery({
   458→    queryKey: ['conversations', me.id],
   459→    queryFn: async (): Promise<ConversationSummary[]> => {
   460→      const res = await apiJson<ConversationsResponse>(
   461→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
   462→      )
   463→      return res.conversations
   464→    },
   465→    refetchInterval: 6_000,
   466→  })
   467→
   468→  // live drafts → "Draft: …" previews in the list (zustand external store)
   469→  const allDrafts = useStore(pulseDraftsStore, (s) => s.drafts)
   470→
   471→  const rows = useMemo(() => {
   472→    return (conversations.data ?? []).map((conv) => {
   473→      const previewInfo = conversationPreview(conv, me.id)
   474→      const other = conv.isGroup ? null : otherMemberOf(conv, me.id)
   475→      const groupName =
   476→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   477→      const displayName = conversationDisplayName(conv, me.id)
   478→      return {
   479→        conv,
   480→        props: {
   481→          id: conv.id,
   482→          isGroup: conv.isGroup,
   483→          name: displayName,
   484→          time: conv.lastMessage
   485→            ? formatListStamp(conv.lastMessage.createdAt)
   486→            : formatListStamp(conv.updatedAt),
   487→          preview: previewInfo.text,
   488→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   489→          previewDeleted: previewInfo.deleted,
   490→          draft: allDrafts[conv.id] ?? null,
   491→          unreadCount: conv.unreadCount,
   492→          dmName: other?.name ?? null,
   493→          dmColor: other?.color ?? 'emerald',
   494→          groupTitle: groupName,
   495→          online: !conv.isGroup && other !== null && onlineIds.has(other.id),
   496→          pinned: conv.pinnedAt !== null,
   497→          muted: conv.mutedUntil !== null && Date.parse(conv.mutedUntil) > Date.now(),
   498→          typing: typersIn(conv.id, me.id).length > 0,
   499→        },
   500→      }
   501→    })
   502→  }, [conversations.data, me.id, onlineIds, typersIn, allDrafts])
   503→
   504→  const filteredRows = useMemo(() => {
   505→    const q = searchQuery.trim().toLowerCase()
   506→    if (!q) return rows
   507→    return rows.filter(({ conv, props }) => {
   508→      if (props.name.toLowerCase().includes(q)) return true
   509→      if (props.draft?.toLowerCase().includes(q)) return true
   510→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   511→      return false
   512→    })
   513→  }, [rows, searchQuery])
   514→
   515→  /** main-list rows exclude the viewer's archived chats */
   516→  const activeRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt === null), [rows])
   517→  const archivedRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt !== null), [rows])
