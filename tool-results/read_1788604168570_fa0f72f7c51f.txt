   127→   127→function tallyOf(rsvps: EventRsvpWire[]): EventCounts {
   128→   128→  const counts: EventCounts = { going: 0, maybe: 0, no: 0 }
   129→   129→  for (const r of rsvps) counts[r.status] += 1
   130→   130→  return counts

   133→   133→/** 'in 2h 15m' · 'Starting now' under a minute · future only. */
   134→   134→function countdownLabel(startsAtMs: number, nowMs: number): string {
   135→   135→  const diff = startsAtMs - nowMs
   136→   136→  if (diff < 60_000) return 'Starting now'
   137→   137→  const mins = Math.floor(diff / 60_000)
   138→   138→  if (mins < 60) return `in ${mins}m`
   139→   139→  const hours = Math.floor(mins / 60)
   140→   140→  if (hours < 24) return mins % 60 > 0 ? `in ${hours}h ${mins % 60}m` : `in ${hours}h`
   141→   141→  const days = Math.floor(hours / 24)
   142→   142→  return hours % 24 > 0 ? `in ${days}d ${hours % 24}h` : `in ${days}d`
   143→   143→}

   145→   145→/** '2d ago' relative chip for the collapsed Past section. */
   146→   146→function pastLabel(startsAtMs: number, nowMs: number): string {
   147→   147→  const mins = Math.max(0, Math.floor((nowMs - startsAtMs) / 60_000))
   148→   148→  if (mins < 1) return 'just now'
   149→   149→  if (mins < 60) return `${mins}m ago`
   150→   150→  const hours = Math.floor(mins / 60)
   151→   151→  if (hours < 24) return `${hours}h ago`
   152→   152→  return `${Math.floor(hours / 24)}d ago`
   153→   153→}

   156→   156→function defaultStartLocal(): string {

   237→   237→export function EventsSheet({
   238→   238→  open,
   239→   239→  onClose,
   240→   240→  conversationId,
   241→   241→  me,
   242→   242→  members,
   243→   243→  myRole = 'member',
   244→   244→}: {
   245→   245→  open: boolean
   246→   246→  onClose: () => void
   247→   247→  conversationId: string
   248→   248→  me: { id: string }
   249→   249→  members: EventMember[]
   250→   250→  myRole?: GroupRole
   251→   251→}) {
   252→   252→  const reduce = useReducedMotion()
   253→   253→  const queryClient = useQueryClient()
   254→   254→
   255→   255→  // form state
   256→   256→  const [title, setTitle] = useState('')
   257→   257→  const [startsAtLocal, setStartsAtLocal] = useState(defaultStartLocal)
   258→   258→  const [location, setLocation] = useState('')
   259→   259→  // two-tap delete confirm (row id currently armed)
   260→   260→  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null)
   261→   261→  const deleteTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   262→   262→  const [pastOpen, setPastOpen] = useState(false)
   263→   263→
   264→   264→  // live clock for countdown chips — recomputed every 30s while open
   265→   265→  const [nowMs, setNowMs] = useState(() => Date.now())
   266→   266→  useEffect(() => {
   267→   267→    if (!open) return
   268→   268→    // async first tick (fresh clock on open) then steady 30s cadence
   269→   269→    let timer: ReturnType<typeof setTimeout>
   270→   270→    const tick = () => {
   271→   271→      setNowMs(Date.now())
   272→   272→      timer = setTimeout(tick, TICK_MS)
   273→   273→    }
   274→   274→    timer = setTimeout(tick, 0)
   275→   275→    return () => clearTimeout(timer)
   276→   276→  }, [open])
   277→   277→
   278→   278→  useEffect(() => {
   279→   279→    return () => {
   280→   280→      if (deleteTimerRef.current) clearTimeout(deleteTimerRef.current)
   281→   281→    }
   282→   282→  }, [])
   283→   283→
   284→   284→  const memberById = useMemo(() => new Map(members.map((m) => [m.id, m])), [members])
   285→   285→
   286→   286→  // ── data (5s poll, paused when document.hidden) ──────────────
   287→   287→
   288→   288→  const eventsQuery = useQuery({
   289→   289→    queryKey: ['events', conversationId],
   290→   290→    enabled: open && conversationId.length > 0 && me.id.length > 0,
   291→   291→    refetchInterval: POLL_MS,
   292→   292→    refetchIntervalInBackground: false, // tab hidden → poll pauses
   293→   293→    staleTime: 0,
   294→   294→    retry: 1,
   295→   295→    queryFn: async (): Promise<EventsGetResponse> =>
   296→   296→      apiJson<EventsGetResponse>(
   297→   297→        `/api/conversations/${encodeURIComponent(conversationId)}/events?userId=${encodeURIComponent(me.id)}`,
   298→   298→      ),
   299→   299→  })
   300→   300→
   301→   301→  const rsvpMutation = useMutation({
   302→   302→    mutationFn: ({ eventId, status }: { eventId: string; status: EventRsvpStatus }) =>
   303→   303→      apiJson<RsvpResponse>(`/api/events/${encodeURIComponent(eventId)}/rsvp`, {
   304→   304→        method: 'POST',
   305→   305→        body: JSON.stringify({ userId: me.id, status }),
   306→   306→      }),
   307→   307→    // optimistic: move my pill + tally immediately, the poll confirms
   308→   308→    onMutate: async ({ eventId, status }) => {
   309→   309→      await queryClient.cancelQueries({ queryKey: ['events', conversationId] })
   310→   310→      const prev = queryClient.getQueryData<EventsGetResponse>(['events', conversationId])
   311→   311→      queryClient.setQueryData<EventsGetResponse>(['events', conversationId], (old) => {
   312→   312→        if (!old) return old
   313→   313→        return {
   314→   314→          events: old.events.map((e) => {
   315→   315→            if (e.id !== eventId) return e
   316→   316→            const others = e.rsvps.filter((r) => r.userId !== me.id)
   317→   317→            const rsvps: EventRsvpWire[] = [
   318→   318→              ...others,
   319→   319→              { userId: me.id, name: memberById.get(me.id)?.name ?? 'You', status },
   320→   320→            ]
   321→   321→            return { ...e, rsvps, counts: tallyOf(rsvps), myStatus: status }
   322→   322→          }),
   323→   323→        }
   324→   324→      })
   325→   325→      return { prev }
   326→   326→    },
   327→   327→    onError: (error, _vars, ctx) => {
   328→   328→      if (ctx?.prev) queryClient.setQueryData(['events', conversationId], ctx.prev)
   329→   329→      toast.error(error instanceof Error ? error.message : 'RSVP failed — try again.')
   330→   330→    },
   331→   331→    onSettled: () => {
   332→   332→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   333→   333→    },
   334→   334→  })
   335→   335→
   336→   336→  const createMutation = useMutation({
   337→   337→    mutationFn: (payload: { title: string; startsAt: string; location: string }) =>
   338→   338→      apiJson<EventsPostResponse>(
   339→   339→        `/api/conversations/${encodeURIComponent(conversationId)}/events`,
   340→   340→        {
   341→   341→          method: 'POST',
   342→   342→          body: JSON.stringify({ userId: me.id, ...payload }),
   343→   343→        },
   344→   344→      ),
   345→   345→    onSuccess: (res) => {
   346→   346→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   347→   347→      setTitle('')
   348→   348→      setLocation('')
   349→   349→      setStartsAtLocal(defaultStartLocal())
   350→   350→      haptic(14)
   351→   351→      toast.success(`“${res.event.title}” is on the calendar.`)
   352→   352→    },
   353→   353→    onError: (error) => {
   354→   354→      toast.error(error instanceof Error ? error.message : 'Could not schedule the event.')
   355→   355→    },
   356→   356→  })
   357→   357→
   358→   358→  const deleteMutation = useMutation({
   359→   359→    mutationFn: (eventId: string) =>
   360→   360→      apiJson<DeleteResponse>(`/api/events/${encodeURIComponent(eventId)}?userId=${encodeURIComponent(me.id)}`, {
   361→   361→        method: 'DELETE',
   362→   362→      }),
   363→   363→    onSuccess: () => {
   364→   364→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   365→   365→      haptic(14)
   366→   366→      toast.success('Event deleted.')
   367→   367→    },
   368→   368→    onError: (error) => {
   369→   369→      toast.error(error instanceof Error ? error.message : 'Could not delete the event.')
   370→   370→      void queryClient.invalidateQueries({ queryKey: ['events', conversationId] })
   371→   371→    },
   372→   372→  })
   373→   373→
   374→   374→  // ── derived lists (server pre-sorts: upcoming asc, past desc) ─
   375→   375→
   376→   376→  const events = eventsQuery.data?.events ?? []
   377→   377→  const upcoming = useMemo(
   378→   378→    () => events.filter((e) => new Date(e.startsAt).getTime() >= nowMs),
   379→   379→    [events, nowMs],
   380→   380→  )
   381→   381→  const past = useMemo(
   382→   382→    () => events.filter((e) => new Date(e.startsAt).getTime() < nowMs),
   383→   383→    [events, nowMs],
   384→   384→  )
   385→   385→
   386→   386→  const offline = eventsQuery.isError
   387→   387→
   388→   388→  const submitCreate = (e: React.FormEvent<HTMLFormElement>) => {
   389→   389→    e.preventDefault()
   390→   390→    const trimmed = title.trim()
   391→   391→    const ms = startsAtLocal.length > 0 ? Date.parse(startsAtLocal) : Number.NaN
   392→   392→    if (trimmed.length === 0 || Number.isNaN(ms)) return
   393→   393→    createMutation.mutate({
   394→   394→      title: trimmed,
   395→   395→      startsAt: new Date(ms).toISOString(),
   396→   396→      location: location.trim(),
   397→   397→    })
   398→   398→  }
   399→   399→
   400→   400→  const handleDeleteTap = (eventId: string) => {
   401→   401→    if (confirmDeleteId !== eventId) {
   402→   402→      setConfirmDeleteId(eventId)
   403→   403→      haptic(12)
   404→   404→      if (deleteTimerRef.current) clearTimeout(deleteTimerRef.current)
   405→   405→      deleteTimerRef.current = setTimeout(() => setConfirmDeleteId(null), DELETE_CONFIRM_MS)
   406→   406→      return
   407→   407→    }
   408→   408→    if (deleteTimerRef.current) {
   409→   409→      clearTimeout(deleteTimerRef.current)
   410→   410→      deleteTimerRef.current = null
   411→   411→    }
   412→   412→    setConfirmDeleteId(null)
   413→   413→    deleteMutation.mutate(eventId)
   414→   414→  }
   415→   415→
   416→   416→  const statusLine = offline ? 'Reconnecting…' : `${upcoming.length} upcoming · live`
   417→   417→
   418→   418→  // ── row renderer (shared by upcoming + past) ─────────────────
   419→   419→
   420→   420→  const renderRow = (e: GroupEventWire, kind: 'upcoming' | 'past') => {
   421→   421→    const startsAtMs = new Date(e.startsAt).getTime()
   422→   422→    const canDelete = me.id === e.createdById || myRole === 'admin'
   423→   423→    const creatorName = (e.createdById ? memberById.get(e.createdById)?.name : undefined) ?? e.createdByName
   424→   424→    return (
   425→   425→      <motion.div
   426→   426→        key={e.id}
   427→   427→        layout={!reduce}
   428→   428→        variants={rowVariants}
   429→   429→        initial={reduce ? false : 'hidden'}
   430→   430→        animate="show"
   431→   431→        exit={reduce ? { opacity: 0 } : { opacity: 0, scale: 0.96, y: 6 }}
   432→   432→        transition={spring.soft}
   433→   433→        style={{ willChange: 'transform' }}
   434→   434→        className="rounded-2xl border border-white/10 bg-white/[0.04] p-3.5"
   435→   435→      >
   436→   436→        <div className="flex gap-3">
   437→   437→          <DateTile startsAtMs={startsAtMs} />
   438→   438→          <div className="min-w-0 flex-1">
   439→   439→            <div className="flex items-start justify-between gap-2">
   440→   440→              <p className="min-w-0 truncate text-[14px] font-bold leading-tight text-zinc-50">
   441→   441→                {e.title}
   442→   442→              </p>
   443→   443→              {canDelete ? (
   444→   444→                <motion.button
   445→   445→                  type="button"
   446→   446→                  whileTap={reduce ? undefined : pressTap}
   447→   447→                  disabled={deleteMutation.isPending}
   448→   448→                  aria-label={
   449→   449→                    confirmDeleteId === e.id
   450→   450→                      ? `Tap again to delete event ${e.title}`
   451→   451→                      : `Delete event ${e.title}`
   452→   452→                  }
   453→   453→                  onClick={() => handleDeleteTap(e.id)}
   454→   454→                  className={cn(
   455→   455→                    'flex size-7 shrink-0 items-center justify-center rounded-full outline-none ring-emerald-400/60 transition-colors focus-visible:ring-2 disabled:opacity-50',
   456→   456→                    confirmDeleteId === e.id
   457→   457→                      ? 'bg-rose-500 text-white'
   458→   458→                      : 'text-zinc-500 hover:bg-rose-500/10 hover:text-rose-400',
   459→   459→                  )}
   460→   460→                >
   461→   461→                  <Trash2 className="size-3.5" aria-hidden />
   462→   462→                </motion.button>

   743→   743→/**
   744→   744→ * Drop-in wiring for chat-room (lead, 2 lines):
   745→   745→ *   const events = useEventsSheet(conversationId, me.id, members, myRole?)
   746→   746→ *   ... {events.node}
   747→   747→ * Listens for the `pulse:open-events` CustomEvent dispatched by
   748→   748→ * the /events slash-palette entry.
   749→   749→ */
   750→   750→export function useEventsSheet(
   751→   751→  conversationId: string,
   752→   752→  meId: string,
   753→   753→  members: EventMember[],
   754→   754→  myRole: GroupRole = 'member',
   755→   755→) {
   756→   756→  const [open, setOpen] = useState(false)
   757→   757→
   758→   758→  useEffect(() => {
   759→   759→    const handler = () => setOpen(true)
   760→   760→    window.addEventListener(EVENTS_OPEN_EVENT, handler)
   761→   761→    return () => window.removeEventListener(EVENTS_OPEN_EVENT, handler)
   762→   762→  }, [])
   763→   763→
   764→   764→  const node = (
   765→   765→    <EventsSheet
   766→   766→      open={open}
   767→   767→      onClose={() => setOpen(false)}
   768→   768→      conversationId={conversationId}
   769→   769→      me={{ id: meId }}
   770→   770→      members={members}
   771→   771→      myRole={myRole}
   772→   772→    />
   773→   773→  )
   774→   774→
   775→   775→  return { open, setOpen, node }
   776→   776→}
