   245→/**
   246→ * Snippet with the first match highlighted — clips a ≤64-char window
   247→ * around the hit so long messages stay one tidy line.
   248→ */
   249→function SearchSnippet({ content, query }: { content: string; query: string }) {
   250→  const lower = content.toLowerCase()
   251→  const q = query.toLowerCase()
   252→  const idx = q.length > 0 ? lower.indexOf(q) : -1
   253→  let from = 0
   254→  let clippedHead = false
   255→  if (idx > 28) {
   256→    from = idx - 24
   257→    clippedHead = true
   258→  }
   259→  const end = Math.min(content.length, idx + q.length + 28)
   260→  const clippedTail = end < content.length
   261→  const body = content.slice(from, end)
   262→  const localIdx = idx - from
   263→  return (
   264→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
   265→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   266→      {idx >= 0 ? (
   267→        <>
   268→          {body.slice(0, localIdx)}
   269→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
   270→            {body.slice(localIdx, localIdx + q.length)}
   271→          </mark>
   272→          {body.slice(localIdx + q.length)}
   273→        </>
   274→      ) : (
   275→        body
   276→      )}
   277→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
   278→    </p>
   279→  )
   280→}

   358→export function ChatsTab({
   359→  me,
   360→  onOpenConversation,
   361→  onOpenContacts,
   362→  onRequestNewChat,
   363→  onGoProfile,
   364→}: {
   365→  me: AppUser
   366→  onOpenConversation: (conversationId: string, unreadAnchorMs: number | null, jumpMessageId?: string) => void
   367→  onOpenContacts: () => void
   368→  onRequestNewChat: () => void
   369→  onGoProfile: () => void
   370→}) {
   371→  const realtime = usePulseRealtime()
   372→  const typersIn = realtime.typersIn
   373→  const onlineIds = realtime.onlineIds
   374→  const [searching, setSearching] = useState(false)
   375→  const [searchQuery, setSearchQuery] = useState('')
   376→  /** debounced mirror of searchQuery feeding the server message search */
   377→  const [deferredQuery, setDeferredQuery] = useState('')
   378→  const deferredTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   379→
   380→  const conversations = useQuery({
   381→    queryKey: ['conversations', me.id],
   382→    queryFn: async (): Promise<ConversationSummary[]> => {
   383→      const res = await apiJson<ConversationsResponse>(
   384→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
   385→      )
   386→      return res.conversations
   387→    },
   388→    refetchInterval: 6_000,
   389→  })
   390→
   391→  // live drafts → "Draft: …" previews in the list (zustand external store)
   392→  const allDrafts = useStore(pulseDraftsStore, (s) => s.drafts)
   393→
   394→  const rows = useMemo(() => {
   395→    return (conversations.data ?? []).map((conv) => {
   396→      const previewInfo = conversationPreview(conv, me.id)
   397→      const other = conv.isGroup ? null : otherMemberOf(conv, me.id)
   398→      const groupName =
   399→        conv.name?.trim() || conv.members.filter((m) => m.id !== me.id).map((m) => m.name).join(', ') || 'Group'
   400→      const displayName = conversationDisplayName(conv, me.id)
   401→      return {
   402→        conv,
   403→        props: {
   404→          id: conv.id,
   405→          isGroup: conv.isGroup,
   406→          name: displayName,
   407→          time: conv.lastMessage
   408→            ? formatListStamp(conv.lastMessage.createdAt)
   409→            : formatListStamp(conv.updatedAt),
   410→          preview: previewInfo.text,
   411→          previewPrefix: conversationPreviewPrefix(previewInfo, conv.isGroup),
   412→          previewDeleted: previewInfo.deleted,
   413→          draft: allDrafts[conv.id] ?? null,
   414→          unreadCount: conv.unreadCount,
   415→          dmName: other?.name ?? null,
   416→          dmColor: other?.color ?? 'emerald',
   417→          groupTitle: groupName,
   418→          online: !conv.isGroup && other !== null && onlineIds.has(other.id),
   419→          pinned: conv.pinnedAt !== null,
   420→          muted: conv.mutedUntil !== null && Date.parse(conv.mutedUntil) > Date.now(),
   421→          typing: typersIn(conv.id, me.id).length > 0,
   422→        },
   423→      }
   424→    })
   425→  }, [conversations.data, me.id, onlineIds, typersIn, allDrafts])
   426→
   427→  const filteredRows = useMemo(() => {
   428→    const q = searchQuery.trim().toLowerCase()
   429→    if (!q) return rows
   430→    return rows.filter(({ conv, props }) => {
   431→      if (props.name.toLowerCase().includes(q)) return true
   432→      if (props.draft?.toLowerCase().includes(q)) return true
   433→      if (conv.lastMessage && !conv.lastMessage.deletedAt && conv.lastMessage.content.toLowerCase().includes(q)) return true
   434→      return false
   435→    })
   436→  }, [rows, searchQuery])
   437→
   438→  /** main-list rows exclude the viewer's archived chats */
   439→  const activeRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt === null), [rows])
   440→  const archivedRows = useMemo(() => rows.filter(({ conv }) => conv.archivedAt !== null), [rows])
   441→
   442→  // Telegram-style folder filter (persisted preference)
   443→  const listFilter = useStore(pulseSettingsStore, (s) => s.listFilter)
   444→  const setListFilter = useStore(pulseSettingsStore, (s) => s.setListFilter)
   445→  const unreadTotal = useMemo(
   446→    () => activeRows.reduce((sum, { conv }) => sum + conv.unreadCount, 0),
   447→    [activeRows],
   448→  )
   449→  const folderFiltered = useMemo(() => {
   450→    if (listFilter === 'unread') return activeRows.filter(({ conv }) => conv.unreadCount > 0)
   451→    if (listFilter === 'groups') return activeRows.filter(({ conv }) => conv.isGroup)
   452→    return activeRows
   453→  }, [activeRows, listFilter])
   454→  const archivedUnread = useMemo(
   455→    () => archivedRows.reduce((sum, { conv }) => sum + conv.unreadCount, 0),
   456→    [archivedRows],
   457→  )
   458→  /** archived-chats drawer (WhatsApp-style) */
   459→  const [archivedOpen, setArchivedOpen] = useState(false)
   460→
   461→  /** Freeze "where was I" from the list summary AT TAP TIME (pre-read watermark). */
   462→  const handlePress = useCallback(
   463→    (conv: ConversationSummary) => {
   464→      const mine = conv.members.find((m) => m.id === me.id) as
   465→        | (AppUser & { lastReadAt?: string })
   466→        | undefined
   467→      const wm = mine?.lastReadAt ? Date.parse(mine.lastReadAt) : Number.NaN
   468→      const anchor = conv.unreadCount > 0 && !Number.isNaN(wm) ? wm : null
   469→      onOpenConversation(conv.id, anchor)
   470→    },
   471→    [onOpenConversation, me.id],
   472→  )
   473→
   474→  const closeSearch = useCallback(() => {
   475→    if (deferredTimerRef.current !== null) {
   476→      clearTimeout(deferredTimerRef.current)
   477→      deferredTimerRef.current = null
   478→    }
   479→    setSearching(false)
   480→    setSearchQuery('')
   481→    setDeferredQuery('')
   482→  }, [])
   483→
   484→  /** typing in the search field: instant local filter + 250ms-debounced server search */
   485→  const handleSearchInput = useCallback((value: string) => {
   486→    setSearchQuery(value)
   487→    if (deferredTimerRef.current !== null) clearTimeout(deferredTimerRef.current)
   488→    const v = value.trim()
   489→    deferredTimerRef.current = setTimeout(() => {
   490→      deferredTimerRef.current = null
   491→      setDeferredQuery(v)
   492→    }, 250)
   493→  }, [])
   494→
   495→  const serverSearch = useQuery({
   496→    queryKey: ['global-search', me.id, deferredQuery],
   497→    enabled: searching && deferredQuery.length >= 2,
   498→    staleTime: 15_000,
   499→    queryFn: async (): Promise<SearchResponse> =>
   500→      apiJson<SearchResponse>(
   501→        `/api/search?userId=${encodeURIComponent(me.id)}&q=${encodeURIComponent(deferredQuery)}`,
   502→      ),
   503→  })
   504→  const serverHits = searching && deferredQuery.length >= 2 ? (serverSearch.data?.messages ?? []) : []
   505→
   506→  // ── row long-press action sheet (pin/unpin) ───────────────
   507→  const [sheetConv, setSheetConv] = useState<ConversationSummary | null>(null)
   508→  const queryClient = useQueryClient()
   509→
   510→  const togglePin = useMutation({
   511→    mutationFn: async (conv: ConversationSummary) => {
   512→      return apiJson<{ ok: boolean; pinned: boolean }>(
   513→        `/api/conversations/${encodeURIComponent(conv.id)}/pin`,
