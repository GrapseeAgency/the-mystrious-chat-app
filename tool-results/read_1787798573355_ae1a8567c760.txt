    71→interface DetailResponse {
    72→  conversation: ConversationDetail
    73→}

    75→  messages: ChatMessage[]
    76→}
    77→interface SendResponse {
    78→  message: ChatMessage
    79→}
    80→interface DeleteResponse {
    81→  message: ChatMessage

    83→
    84→const CLUSTER_WINDOW_MS = 5 * 60 * 1000
    85→const NEAR_BOTTOM_PX = 160
    86→
    87→type ClusterItem =
    88→  | { kind: 'day'; key: string; label: string }
    89→  | { kind: 'msg'; key: string; message: ChatMessage; head: boolean; tail: boolean }

    91→export function ChatRoom({
    92→  me,
    93→  conversationId,
    94→  onClose,
    95→}: {
    96→  me: AppUser
    97→  conversationId: string
    98→  onClose: () => void
    99→}) {
   100→  const queryClient = useQueryClient()
   101→  const realtime = usePulseRealtime()
   102→  const { resolvedTheme } = useTheme()
   103→  const themeMounted = useMounted()
   104→
   105→  const [input, setInput] = useState('')
   106→  const [menuOpen, setMenuOpen] = useState(false)
   107→  const [infoOpen, setInfoOpen] = useState(false)
   108→  const [selected, setSelected] = useState<ChatMessage | null>(null)
   109→  const [confirmingDelete, setConfirmingDelete] = useState(false)
   110→  const [showJump, setShowJump] = useState(false)
   111→
   112→  const viewportRef = useRef<HTMLDivElement>(null)
   113→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   114→  const nearBottomRef = useRef(true)
   115→  const longPressRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   116→  /** flipped (inside rAF) after the first history fetch so render stays ref-free */
   117→  const [historyLoaded, setHistoryLoaded] = useState(false)
   118→
   119→  // ── data ───────────────────────────────────────────────────
   120→
   121→  const detail = useQuery({
   122→    queryKey: ['conversation', conversationId],
   123→    queryFn: async (): Promise<ConversationDetail> => {
   124→      const res = await apiJson<DetailResponse>(
   125→        `/api/conversations/${encodeURIComponent(conversationId)}?userId=${encodeURIComponent(me.id)}`,
   126→      )
   127→      return res.conversation
   128→    },
   129→    staleTime: 15_000,
   130→    // safety net: keeps read-ticks/membership fresh even if the socket path degrades
   131→    refetchInterval: 6_000,
   132→  })
   133→
   134→  const messages = useQuery({
   135→    queryKey: ['messages', conversationId],
   136→    queryFn: async (): Promise<ChatMessage[]> => {
   137→      const res = await apiJson<MessagesResponse>(
   138→        `/api/conversations/${encodeURIComponent(conversationId)}/messages?limit=200`,
   139→      )
   140→      return res.messages
   141→    },
   142→    staleTime: 15_000,
   143→    // safety net: messages still arrive within seconds even without websocket realtime
   144→    refetchInterval: 3_500,
   145→  })
   146→
   147→  // ── realtime registration + read receipts ──────────────────
   148→
   149→  useEffect(() => {
   150→    realtime.setActiveConversation(conversationId)
   151→    return () => realtime.setActiveConversation(null)
   152→  }, [conversationId, realtime])
   153→
   154→  /** POST read marker (visible rooms only) then patch local caches. */
   155→  const markVisibleRead = useCallback(() => {
   156→    if (typeof document !== 'undefined' && document.visibilityState !== 'visible') return
   157→    if ((messages.data ?? []).length === 0) return
   158→    fetch(`/api/conversations/${encodeURIComponent(conversationId)}/read`, {
   159→      method: 'POST',
   160→      headers: { 'Content-Type': 'application/json' },
   161→      body: JSON.stringify({ userId: me.id }),
   162→    })
   163→      .then(() => {
   164→        const nowIso = new Date().toISOString()
   165→        queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   166→          old
   167→            ? {
   168→                ...old,
   169→                members: old.members.map((m) =>
   170→                  m.id === me.id ? { ...m, lastReadAt: nowIso } : m,
   171→                ),
   172→              }
   173→            : old,
   174→        )
   175→        // refresh list badges (my unread for this room is now zero)
   176→        queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   177→      })
   178→      .catch(() => undefined)
   179→  }, [conversationId, me.id, messages.data, queryClient])
   180→
   181→  // opening the room + every new batch of messages ⇒ (debounced) read
   182→  useEffect(() => {
   183→    const timer = setTimeout(markVisibleRead, 600)
   184→    return () => clearTimeout(timer)
   185→  }, [markVisibleRead])
   186→
   187→  useEffect(() => {
   188→    const onVisibility = () => {
   189→      if (document.visibilityState === 'visible') markVisibleRead()
   190→    }
   191→    document.addEventListener('visibilitychange', onVisibility)
   192→    return () => document.removeEventListener('visibilitychange', onVisibility)
   193→  }, [markVisibleRead])
   194→
   195→  // ── derived ────────────────────────────────────────────────
   196→
   197→  const detailData = detail.data
   198→  const other = useMemo(
   199→    () => (detailData ? otherMemberOf(detailData, me.id) : null),
   200→    [detailData, me.id],
   201→  )
   202→  const isGroup = detailData?.isGroup ?? false
   203→  const displayName = detailData ? conversationDisplayName(detailData, me.id) : ''
   204→
   205→  const recipients = useMemo(
   206→    () => (detailData ? detailData.members.filter((m) => m.id !== me.id).map((m) => m.id) : []),
   207→    [detailData, me.id],
   208→  )
   209→
   210→  const typers = realtime.typersIn(conversationId, me.id)
   211→  const typerLabel = useMemo(() => {
   212→    if (typers.length === 0) return ''
   213→    if (!isGroup) return 'typing…'
   214→    const names = typers.map((t) => t.userName)
   215→    if (names.length === 1) return `${names[0]} is typing…`
   216→    if (names.length === 2) return `${names[0]} and ${names[1]} are typing…`
   217→    return `${names.length} people are typing…`
   218→  }, [typers, isGroup])
   219→
   220→  const onlineOthers = useMemo(
   221→    () =>
   222→      detailData
   223→        ? detailData.members.filter((m) => m.id !== me.id && realtime.onlineIds.has(m.id)).length
   224→        : 0,
   225→    [detailData, me.id, realtime.onlineIds],
   226→  )
   227→
   228→  /** highest read watermark among OTHER members → double ticks */
   229→  const othersMaxReadMs = useMemo(() => {
   230→    if (!detailData) return Number.NEGATIVE_INFINITY
   231→    let max = Number.NEGATIVE_INFINITY
   232→    for (const m of detailData.members) {
   233→      if (m.id === me.id) continue
   234→      const t = Date.parse(m.lastReadAt)
   235→      if (!Number.isNaN(t) && t > max) max = t
   236→    }
   237→    return max
   238→  }, [detailData, me.id])
   239→
   240→  const items = useMemo<ClusterItem[]>(() => {
   241→    const list = messages.data ?? []
   242→    const now = new Date()
   243→    interface Entry {
   244→      message: ChatMessage
   245→      head: boolean
   246→    }
   247→    const built: Entry[] = []
   248→    let prev: ChatMessage | null = null
   249→    for (const message of list) {
   250→      const clusterBreak =
   251→        prev === null ||
   252→        !isSameDayIso(prev.createdAt, message.createdAt) ||
   253→        prev.senderId !== message.senderId ||
   254→        Date.parse(message.createdAt) - Date.parse(prev.createdAt) > CLUSTER_WINDOW_MS
   255→      built.push({ message, head: clusterBreak })
   256→      prev = message
   257→    }
   258→
   259→    const out: ClusterItem[] = []
   260→    let dayAnchor: ChatMessage | null = null
   261→    for (let i = 0; i < built.length; i += 1) {
   262→      const { message, head } = built[i]
   263→      if (dayAnchor === null || !isSameDayIso(dayAnchor.createdAt, message.createdAt)) {
   264→        out.push({ kind: 'day', key: `day-${message.id}`, label: formatDayChip(message.createdAt, now) })
   265→        dayAnchor = message
   266→      }
   267→      out.push({
   268→        kind: 'msg',
   269→        key: message.id,
   270→        message,
   271→        head,
   272→        tail: i === built.length - 1 || built[i + 1].head,
   273→      })
   274→    }
   275→    return out
   276→  }, [messages.data])
   277→
   278→  const lastMessageId =
   279→    messages.data && messages.data.length > 0 ? messages.data[messages.data.length - 1].id : null
   280→
   281→  // ── scrolling ──────────────────────────────────────────────
   282→
   283→  const scrollToBottom = useCallback((smooth: boolean) => {
   284→    const el = viewportRef.current
   285→    if (!el) return
   286→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   287→  }, [])
   288→
   289→  useEffect(() => {
   290→    if (lastMessageId === null || !historyLoaded) return
   291→    requestAnimationFrame(() => {
   292→      if (nearBottomRef.current) {
   293→        scrollToBottom(true)
   294→      } else {
   295→        setShowJump(true)
   296→        buzz(10)
   297→      }
   298→    })
   299→  }, [lastMessageId, historyLoaded, scrollToBottom])
   300→
   301→  useEffect(() => {
   302→    if (messages.isSuccess && !historyLoaded) {
   303→      const frame = requestAnimationFrame(() => {
   304→        scrollToBottom(false)
   305→        setHistoryLoaded(true)
   306→      })
   307→      return () => cancelAnimationFrame(frame)
   308→    }
   309→    return undefined
   310→  }, [messages.isSuccess, historyLoaded, scrollToBottom])
   311→
   312→  const handleScroll = useCallback(() => {
   313→    const el = viewportRef.current
   314→    if (!el) return
   315→    const distance = el.scrollHeight - el.scrollTop - el.clientHeight
   316→    nearBottomRef.current = distance < NEAR_BOTTOM_PX
   317→    if (nearBottomRef.current) setShowJump(false)
   318→  }, [])
   319→
   320→  // ── sending ────────────────────────────────────────────────
   321→
   322→  const sendMessage = useMutation({
   323→    mutationFn: async ({ clientId, content }: { clientId: string; content: string }) => {
   324→      const res = await apiJson<SendResponse>(
   325→        `/api/conversations/${encodeURIComponent(conversationId)}/messages`,
   326→        {
   327→          method: 'POST',
   328→          headers: { 'Content-Type': 'application/json' },
   329→          body: JSON.stringify({ senderId: me.id, content }),
   330→        },
   331→      )
   332→      return { res, clientId }
   333→    },
   334→    onMutate: async ({ clientId, content }) => {
   335→      const temp: ChatMessage = {
   336→        id: `temp-${clientId}`,
   337→        conversationId,
   338→        senderId: me.id,
   339→        content,
   340→        deletedAt: null,
   341→        createdAt: new Date().toISOString(),
   342→        sender: { id: me.id, name: me.name, color: me.color },
   343→      }
   344→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   345→        old ? [...old, temp] : [temp],
   346→      )
   347→    },
   348→    onSuccess: ({ res, clientId }) => {
   349→      const real = res.message
   350→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) => {
   351→        if (!old) return [real]
   352→        const hadReal = old.some((m) => m.id === real.id)
   353→        const cleaned = old.filter(
   354→          (m) =>
   355→            m.id !== `temp-${clientId}` &&
   356→            !(m.id.startsWith('temp-') && m.content === real.content && m.senderId === real.senderId),
   357→        )
   358→        return hadReal ? cleaned : [...cleaned, real]
   359→      })
   360→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   361→    },
   362→    onError: (_error, { clientId }) => {
   363→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   364→        old ? old.filter((m) => m.id !== `temp-${clientId}`) : old,
   365→      )
   366→      toast.error('Message failed to send')
   367→    },
   368→  })
   369→
   370→  const deleteMessage = useMutation({
   371→    mutationFn: async (messageId: string) => {
   372→      return apiJson<DeleteResponse>(`/api/messages/${encodeURIComponent(messageId)}`, {
   373→        method: 'DELETE',
   374→        headers: { 'Content-Type': 'application/json' },
   375→        body: JSON.stringify({ requesterId: me.id }),
   376→      })
   377→    },
   378→    onMutate: async (messageId) => {
   379→      const previous = queryClient.getQueryData<ChatMessage[]>(['messages', conversationId])
   380→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   381→        old
   382→          ? old.map((m) => (m.id === messageId ? { ...m, deletedAt: new Date().toISOString() } : m))
   383→          : old,
   384→      )
   385→      return { previous }
   386→    },
   387→    onSuccess: (data) => {
   388→      const real = data.message
   389→      queryClient.setQueryData<ChatMessage[]>(['messages', conversationId], (old) =>
   390→        old ? old.map((m) => (m.id === real.id ? real : m)) : old,
   391→      )
   392→      queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   393→      toast.success('Message deleted')
   394→    },
   395→    onError: (_error, _messageId, context) => {
   396→      if (context?.previous) {
   397→        queryClient.setQueryData(['messages', conversationId], context.previous)
   398→      }
   399→      toast.error('Could not delete the message')
   400→    },
   401→    onSettled: () => {
   402→      setSelected(null)
   403→      setConfirmingDelete(false)
   404→    },
   405→  })
   406→
   407→  // ── composer behaviour ─────────────────────────────────────
   408→
   409→  const autosize = useCallback(() => {
   410→    const el = textareaRef.current
   411→    if (!el) return
   412→    el.style.height = 'auto'
   413→    el.style.height = `${Math.min(el.scrollHeight, 120)}px`
   414→  }, [])
   415→
   416→  const stopTyping = useCallback(() => {
   417→    if (recipients.length > 0) {
   418→      realtime.cancelTyping(conversationId, { viewerId: me.id, recipients })
   419→    }
   420→  }, [realtime, conversationId, me.id, recipients])
   421→
   422→  useEffect(() => () => stopTyping(), [stopTyping])
   423→
   424→  const submit = useCallback(() => {
   425→    const content = input.trim()
   426→    if (content.length === 0 || sendMessage.isPending) return
   427→    stopTyping()
   428→    setInput('')
   429→    requestAnimationFrame(autosize)
   430→    sendMessage.mutate({ clientId: uid(), content })
   431→  }, [input, sendMessage, stopTyping, autosize])
   432→
   433→  const handleInputChange = (value: string) => {
   434→    setInput(value)
   435→    autosize()
   436→    if (value.trim().length > 0 && recipients.length > 0) {
   437→      realtime.signalTyping(conversationId, {
   438→        viewerId: me.id,
   439→        userName: me.name,
   440→        recipients,
   441→      })
   442→    } else {
   443→      stopTyping()
   444→    }
   445→  }
   446→
   447→  const handleKeyDown = (event: React.KeyboardEvent<HTMLTextAreaElement>) => {
   448→    if (
   449→      event.key === 'Enter' &&
   450→      !event.shiftKey &&
   451→      typeof window !== 'undefined' &&
   452→      window.matchMedia('(min-width: 640px)').matches
   453→    ) {
   454→      event.preventDefault()
   455→      submit()
   456→    }
   457→  }
   458→
   459→  // ── long-press helpers ─────────────────────────────────────
   460→
   461→  const clearLongPress = useCallback(() => {
   462→    if (longPressRef.current !== null) {
   463→      clearTimeout(longPressRef.current)
   464→      longPressRef.current = null
   465→    }
   466→  }, [])
   467→
   468→  const startLongPress = useCallback(
   469→    (message: ChatMessage) => {
   470→      clearLongPress()
   471→      longPressRef.current = setTimeout(() => {
   472→        setSelected(message)
   473→        longPressRef.current = null
   474→      }, 450)
   475→    },
   476→    [clearLongPress],
   477→  )
   478→
   479→  const copySelected = async () => {
   480→    if (!selected) return
   481→    try {
   482→      await navigator.clipboard.writeText(selected.content)
   483→      toast.success('Copied to clipboard')
   484→    } catch {
   485→      toast.error('Clipboard is unavailable here')
   486→    }
   487→    setSelected(null)
   488→  }
   489→
   490→  const canDeleteSelected = selected !== null && selected.senderId === me.id && !selected.deletedAt
   491→
   492→  const headerTitle = !isGroup && other ? other.name : displayName || 'Conversation'
   493→
   494→  const subtitle = typerLabel.length > 0
   495→    ? typerLabel
   496→    : isGroup
   497→      ? `${detailData?.members.length ?? 0} members · ${onlineOthers} online`
   498→      : !other
   499→        ? ''
   500→        : realtime.onlineIds.has(other.id)
   501→          ? 'online'
   502→          : 'offline'
   503→
   504→  const dotColor = themeMounted && resolvedTheme === 'dark' ? 'rgba(255,255,255,0.055)' : 'rgba(0,0,0,0.05)'
   505→
   506→  return (
   507→    <motion.div
   508→      initial={{ y: '100%' }}
   509→      animate={{ y: 0 }}
   510→      exit={{ y: '100%' }}
   511→      transition={{ type: 'spring', stiffness: 320, damping: 34 }}
   512→      className="absolute inset-0 z-40 flex flex-col bg-white dark:bg-zinc-900"
   513→      role="dialog"
   514→      aria-label={`Conversation with ${headerTitle}`}
   515→    >
   516→      {/* header */}
   517→      <header className="relative z-20 flex min-h-14 shrink-0 items-center gap-1.5 border-b border-zinc-200 bg-white px-2 pt-[env(safe-area-inset-top)] dark:border-zinc-800 dark:bg-zinc-900">
   518→        <Button
   519→          variant="ghost"
   520→          size="icon"
   521→          aria-label="Back to chats"
   522→          onClick={() => {
   523→            stopTyping()
   524→            onClose()
   525→          }}
   526→          className="size-10 shrink-0 rounded-full text-zinc-600 hover:bg-transparent hover:text-zinc-900 active:scale-95 dark:text-zinc-300 dark:hover:text-white"
   527→        >
   528→          <ChevronLeft className="size-6" aria-hidden />
   529→        </Button>
   530→        {isGroup ? (
   531→          <GroupAvatar title={displayName} id={conversationId} size={36} />
   532→        ) : (
   533→          <UserAvatar
   534→            name={other?.name ?? displayName}
   535→            color={other?.color}
   536→            size={36}
   537→            showPresence
   538→            online={other ? realtime.onlineIds.has(other.id) : false}
   539→          />
   540→        )}
   541→        <button
   542→          type="button"
   543→          onClick={() => setInfoOpen(true)}
   544→          aria-label="Show info"
   545→          className="ml-1.5 min-w-0 flex-1 text-left outline-none"
   546→        >
   547→          <p className="truncate text-sm font-semibold tracking-tight text-zinc-900 dark:text-zinc-50">
   548→            {headerTitle}
   549→          </p>
   550→          <AnimatePresence mode="wait" initial={false}>
   551→            <motion.p
   552→              key={subtitle}
   553→              initial={{ opacity: 0, y: 3 }}
   554→              animate={{ opacity: 1, y: 0 }}
   555→              exit={{ opacity: 0 }}
   556→              transition={{ duration: 0.18 }}
   557→              className={cn(
   558→                'truncate text-[11px]',
   559→                typerLabel.length > 0
   560→                  ? 'font-medium text-emerald-600 italic dark:text-emerald-400'
   561→                  : 'text-zinc-500 dark:text-zinc-400',
   562→              )}
   563→            >
   564→              {subtitle}
   565→            </motion.p>
   566→          </AnimatePresence>
   567→        </button>
   568→
   569→        <Button
   570→          variant="ghost"
   571→          size="icon"
   572→          aria-haspopup="menu"
   573→          aria-expanded={menuOpen}
   574→          aria-label="Conversation menu"
   575→          onClick={() => setMenuOpen((v) => !v)}
   576→          className="size-10 shrink-0 rounded-full text-zinc-500 hover:text-zinc-700 active:scale-95 dark:hover:text-zinc-300"
   577→        >
   578→          <EllipsisVertical className="size-5" aria-hidden />
   579→        </Button>
   580→
   581→        <AnimatePresence>
   582→          {menuOpen ? (
   583→            <>
   584→              <button
   585→                type="button"
   586→                aria-hidden
   587→                tabIndex={-1}
   588→                onClick={() => setMenuOpen(false)}
