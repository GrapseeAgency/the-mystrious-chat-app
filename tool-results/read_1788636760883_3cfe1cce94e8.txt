    89→/** Shared unread counter from the real conversations API. */
    90→function useUnread(me: AppUser): number {
    91→  const { data } = useQuery({
    92→    queryKey: ['conversations', me.id],
    93→    queryFn: async (): Promise<ConversationSummary[]> => {
    94→      const res = await apiJson<{ conversations: ConversationSummary[] }>(
    95→        `/api/conversations?userId=${encodeURIComponent(me.id)}`,
    96→      )
    97→      return res.conversations
    98→    },
    99→    refetchInterval: 25_000,
   100→  })
   101→  return (data ?? []).reduce((sum, c) => sum + c.unreadCount, 0)
   102→}

   104→/** Unread counter — pops (spring) on every value CHANGE via key remount. */
   105→function UnreadBadge({ count, className }: { count: number; className?: string }) {
   106→  const reduced = useReducedMotion()
   107→  if (count <= 0) return null
   108→  return (
   109→    <motion.span
   110→      key={count}
   111→      initial={reduced ? false : { scale: 0.4, opacity: 0 }}
   112→      animate={{ scale: 1, opacity: 1 }}
   113→      transition={reduced ? { duration: 0 } : spring.bouncy}
   114→      className={cn(
   115→        'pointer-events-none absolute -right-2.5 -top-1.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-gradient-to-br from-emerald-500 to-teal-500 px-1.5 text-[10px] font-bold text-white shadow-[0_4px_12px_-2px_rgba(16,185,129,0.65)] ring-2 ring-white dark:ring-zinc-900',
   116→        className,
   117→      )}
   118→    >
   119→      {count > 99 ? '99+' : count}
   120→    </motion.span>
   121→  )
   122→}

   125→ * Capsule destination: press physics, wobble, active pill, staggered

   128→function CapsuleTab({
   129→  id,
   130→  label,
   131→  Icon,
   132→  badge,
   133→  active,
   134→  unread,
   135→  onSelect,
   136→  layoutId,
   137→}: {
   138→  id: PulseTab
   139→  label: string
   140→  Icon: LucideIcon
   141→  badge?: NavItemDef['badge']
   142→  active: boolean
   143→  unread: number
   144→  onSelect: (tab: PulseTab) => void
   145→  layoutId: string
   146→}) {
   147→  const reduced = useReducedMotion()
   148→  const wobble = useAnimationControls()
   149→  const press = useCallback(() => {
   150→    haptic(12)
   151→    onSelect(id)
   152→    if (!reduced) void wobble.start({ rotate: [0, -8, 6, 0] }, { duration: 0.35, ease: ease.out })
   153→  }, [id, onSelect, reduced, wobble])
   154→  return (
   155→    <motion.button
   156→      type="button"
   157→      role="tab"
   158→      aria-selected={active}
   159→      aria-label={label}
   160→      onClick={press}
   161→      whileTap={reduced ? undefined : { scale: 0.88, y: 1 }}
   162→      transition={reduced ? { duration: 0 } : spring.bouncy}
   163→      className="relative flex min-h-[52px] flex-1 touch-manipulation select-none flex-col items-center justify-center gap-[3px] rounded-[22px] outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60"
   164→    >
   165→      {active ? (
   166→        <motion.span
   167→          layoutId={layoutId}
   168→          transition={reduced ? { duration: 0 } : spring.snappy}
   169→          className="absolute inset-0 rounded-[22px] bg-gradient-to-b from-emerald-500/20 to-emerald-500/[0.06] shadow-[0_6px_20px_-6px_rgba(16,185,129,0.55)] ring-1 ring-inset ring-emerald-500/30 dark:from-emerald-400/[0.16] dark:to-emerald-400/[0.05] dark:ring-emerald-400/25"
   170→          style={{ willChange: 'transform' }}
   171→        />
   172→      ) : null}
   173→      <motion.span animate={wobble} className="relative">
   174→        <motion.span
   175→          animate={{ scale: active ? 1.08 : 1, y: active ? -1 : 0 }}
   176→          transition={reduced ? { duration: 0 } : spring.bouncy}
   177→          className="relative block"
   178→        >
   179→          <Icon
   180→            className={cn(
   181→              'size-[22px] transition-colors duration-200',
   182→              active ? 'text-emerald-600 dark:text-emerald-400' : 'text-zinc-500 dark:text-zinc-400',
   183→            )}
   184→            strokeWidth={active ? 2.2 : 1.8}
   185→            aria-hidden
   186→          />
   187→          {badge === 'chats-unread' ? <UnreadBadge count={unread} /> : null}
   188→        </motion.span>
   189→      </motion.span>
   190→      <motion.span
   191→        animate={{ y: active ? -1 : 0 }}
   192→        transition={{ ...(reduced ? { duration: 0 } : spring.bouncy), delay: reduced ? 0 : stagger(1, 0.02) }}
   193→        className={cn(
   194→          'text-[10px] leading-none transition-colors duration-200',
   195→          active ? 'font-semibold text-emerald-600 dark:text-emerald-400' : 'font-medium text-zinc-500 dark:text-zinc-400',
   196→        )}
   197→      >
   198→        {label}
   199→      </motion.span>
   200→    </motion.button>
   201→  )
   202→}

   210→/**
   211→ * R26-e overflow — "More" trigger (Ellipsis) + compact GlassMenu
   212→ * portal (Settings / Search / Saved / Stories). The menu floats via
   213→ * a document.body portal so nav ancestors with backdrop-filter or
   214→ * overflow-hidden can never clip it; position is measured from the
   215→ * trigger rect at open time (above or below, right-aligned).
   216→ * Springs: glassMenuMotion entrance + pressTap taps. Escape, veil
   217→ * tap, scroll or resize dismiss it.
   218→ */
   219→function NavOverflowButton({
   220→  placement,
   221→  onAction,
   222→  buttonClassName,
   223→  showLabel = false,
   224→  onOpenChange,
   225→}: {
   226→  placement: 'above' | 'below'
   227→  onAction: (action: NavContextAction) => void
   228→  buttonClassName?: string
   229→  showLabel?: boolean
   230→  /** reports open-state so hosts (island auto-collapse) can pause timers */
   231→  onOpenChange?: (open: boolean) => void
   232→}) {
   233→  const reduced = useReducedMotion()
   234→  const [open, setOpen] = useState(false)
   235→  const [anchor, setAnchor] = useState<{ x: number; y: number } | null>(null)
   236→  const btnRef = useRef<HTMLButtonElement | null>(null)
   237→
   238→  const openMenu = useCallback(() => {
   239→    const rect = btnRef.current?.getBoundingClientRect()
   240→    if (rect) setAnchor({ x: rect.right, y: placement === 'above' ? rect.top : rect.bottom })
   241→    setOpen(true)
   242→    onOpenChange?.(true)
   243→  }, [placement, onOpenChange])
   244→
   245→  const close = useCallback(() => {
   246→    setOpen(false)
   247→    onOpenChange?.(false)
   248→  }, [onOpenChange])
   249→
   250→  useEffect(() => {
   251→    if (!open) return
   252→    const onKey = (e: KeyboardEvent) => {
   253→      if (e.key === 'Escape') close()
   254→    }
   255→    const onReposition = () => close()
   256→    window.addEventListener('keydown', onKey)
   257→    window.addEventListener('resize', onReposition)
   258→    window.addEventListener('scroll', onReposition, true)
   259→    return () => {
   260→      window.removeEventListener('keydown', onKey)
   261→      window.removeEventListener('resize', onReposition)
   262→      window.removeEventListener('scroll', onReposition, true)
   263→    }
   264→  }, [open, close])
   265→
   266→  const runAction = (action: NavContextAction) => {
   267→    close()
   268→    onAction(action)
   269→  }
   270→
   271→  const menuMotionProps = reduced
   272→    ? { initial: { opacity: 0 }, animate: { opacity: 1 }, exit: { opacity: 0 } }
   273→    : {}
   274→
   275→  return (
   276→    <>
   277→      <motion.button
   278→        ref={btnRef}
   279→        type="button"
   280→        aria-label="More options"
   281→        aria-haspopup="menu"
   282→        aria-expanded={open}
   283→        onClick={() => {
   284→          haptic(10)
   285→          if (open) close()
   286→          else openMenu()
   287→        }}
   288→        whileTap={reduced ? undefined : pressTap}
   289→        transition={pressSpring}
   290→        className={cn(
   291→          'flex shrink-0 touch-manipulation select-none items-center justify-center rounded-full text-zinc-500 outline-none transition-colors',
   292→          'hover:bg-zinc-900/[0.05] focus-visible:ring-2 focus-visible:ring-emerald-500/60',
   293→          'dark:text-zinc-400 dark:hover:bg-white/[0.07]',
   294→          showLabel
   295→            ? 'min-h-[46px] flex-1 flex-col gap-0.5 rounded-3xl text-[9px] font-semibold'
   296→            : buttonClassName ?? 'size-10 self-center',
   297→        )}
   298→      >
   299→        <Ellipsis className="size-5" aria-hidden />
   300→        {showLabel ? <span className="relative text-zinc-600 dark:text-zinc-300">More</span> : null}
   301→      </motion.button>
   302→      {typeof document === 'undefined'
   303→        ? null
   304→        : createPortal(
   305→            <AnimatePresence>
   306→              {open && anchor ? (
   307→                <>
   308→                  <motion.button
   309→                    key="nav-more-veil"
   310→                    type="button"
   311→                    aria-label="Close menu"
   312→                    initial={{ opacity: 0 }}
   313→                    animate={{ opacity: 1 }}
   314→                    exit={{ opacity: 0 }}
   315→                    transition={{ duration: reduced ? 0 : 0.16 }}
   316→                    onClick={close}
   317→                    className="fixed inset-0 z-[80] cursor-default bg-zinc-950/25 backdrop-blur-[2px]"
   318→                  />
   319→                  <motion.div
   320→                    key="nav-more-anchor"
   321→                    initial={{ opacity: 0 }}
   322→                    animate={{ opacity: 1 }}
   323→                    exit={{ opacity: 0 }}
   324→                    transition={{ duration: reduced ? 0 : 0.12 }}
   325→                    className="fixed z-[81]"
   326→                    style={{
   327→                      left: anchor.x,
   328→                      top: anchor.y,
   329→                      transform: `translate(-100%, ${placement === 'above' ? '-100%' : '0'}) translateY(${placement === 'above' ? '-14px' : '14px'})`,
   330→                    }}
   331→                  >
   332→                    <GlassMenu aria-label="More options" className="w-[232px]" {...menuMotionProps}>
   333→                      <TapMenuItem
   334→                        icon={SettingsIcon}
   335→                        label="Settings"
   336→                        whileTap={reduced ? undefined : pressTap}
   337→                        transition={pressSpring}
   338→                        onClick={() => {
   339→                          haptic(10)
   340→                          runAction('settings')
   341→                        }}
   342→                      />
   343→                      <TapMenuItem
   344→                        icon={Search}
   345→                        label="Search"
   346→                        trailing={<Command className="size-3 text-zinc-400" aria-hidden />}
   347→                        whileTap={reduced ? undefined : pressTap}
   348→                        transition={pressSpring}
   349→                        onClick={() => {
   350→                          haptic(10)
   351→                          runAction('search')
   352→                        }}
   353→                      />
   354→                      <GlassMenuSeparator />
   355→                      <TapMenuItem
   356→                        icon={Bookmark}
   357→                        label="Saved"
   358→                        whileTap={reduced ? undefined : pressTap}
   359→                        transition={pressSpring}
   360→                        onClick={() => {
   361→                          haptic(10)
   362→                          runAction('saved')
   363→                        }}
   364→                      />
   365→                      <TapMenuItem
   366→                        icon={Aperture}
   367→                        label="Stories"
   368→                        whileTap={reduced ? undefined : pressTap}
   369→                        transition={pressSpring}
   370→                        onClick={() => {
   371→                          haptic(10)
   372→                          runAction('stories')
   373→                        }}
   374→                      />
   375→                    </GlassMenu>
   376→                  </motion.div>
   377→                </>
   378→              ) : null}
   379→            </AnimatePresence>,
   380→            document.body,
   381→          )}
   382→    </>
   383→  )
   384→}

   409→function CapsuleNav({
   410→  active,
   411→  onChange,
   412→  unread,
   413→  onContextAction,
   414→}: TabProps & { onContextAction: (action: NavContextAction) => void }) {
   415→  const reduced = useReducedMotion()
   416→  // registry-driven split: first half · center compose · second half · overflow
   417→  const mid = Math.ceil(NAV_ITEMS.length / 2)
   418→  const lead = NAV_ITEMS.slice(0, mid)
   419→  const tail = NAV_ITEMS.slice(mid)
   420→  return (
   421→    <div className="pointer-events-none absolute inset-x-3 bottom-0 z-[45] mb-[calc(env(safe-area-inset-bottom,0px)_+_10px)]">
   422→      <motion.nav
   423→        role="tablist"
   424→        aria-label="Main navigation (floating capsule)"
   425→        initial={reduced ? false : { y: 64, opacity: 0 }}
   426→        animate={{ y: 0, opacity: 1 }}
   427→        exit={reduced ? undefined : { y: 64, opacity: 0 }}
   428→        transition={reduced ? { duration: 0 } : spring.soft}
   429→        className={cn('pointer-events-auto flex items-stretch gap-1 rounded-[28px] p-1.5', GLASS_PANEL)}
   430→      >
   431→        {lead.map((item) => (
   432→          <CapsuleTab
   433→            key={item.id}
   434→            id={item.id}
   435→            label={item.label}
   436→            Icon={item.Icon}
   437→            badge={item.badge}
   438→            active={active === item.id}
   439→            unread={item.badge === 'chats-unread' ? unread : 0}
   440→            onSelect={onChange}
   441→            layoutId="nav-capsule-pill"
   442→          />
   443→        ))}
   444→        <CapsuleComposeButton onAction={onContextAction} />
   445→        {tail.map((item) => (
   446→          <CapsuleTab
   447→            key={item.id}
   448→            id={item.id}
   449→            label={item.label}
   450→            Icon={item.Icon}
   451→            badge={item.badge}
   452→            active={active === item.id}
   453→            unread={item.badge === 'chats-unread' ? unread : 0}
   454→            onSelect={onChange}
   455→            layoutId="nav-capsule-pill"
   456→          />
   457→        ))}
   458→        <NavOverflowButton placement="above" onAction={onContextAction} />
   459→      </motion.nav>
   460→    </div>
   461→  )
   462→}

   505→function FloatingDock({ active, onChange, unread }: TabProps) {
   506→  const reduced = useReducedMotion()
   507→  return (
   508→    <div className="pointer-events-none absolute inset-x-0 bottom-0 z-[45] mb-[calc(env(safe-area-inset-bottom,0px)_+_12px)] flex justify-center">

   513→        animate={{ y: 0, opacity: 1 }}
   514→        exit={reduced ? undefined : { y: 64, opacity: 0 }}
   515→        transition={reduced ? { duration: 0 } : spring.soft}
   516→        className={cn('pointer-events-auto flex items-end gap-1 rounded-[24px] px-2.5 py-2', GLASS_PANEL)}
   517→      >
   518→        {NAV_ITEMS.map((item) => (
   519→          <div key={item.id} className="flex items-end gap-1">
   520→            {item.id !== NAV_ITEMS[0].id ? <span aria-hidden className="mb-3 size-1 rounded-full bg-zinc-300 dark:bg-zinc-700" /> : null}
   521→            <motion.button
   522→              type="button"
   523→              role="tab"
   524→              aria-selected={active === item.id}
   525→              aria-label={item.label}
   526→              onClick={() => {
   527→                haptic(12)
   528→                onChange(item.id)
   529→              }}

   923→// ── 10 · island — dynamic-island expanding pill ──────────────

   939→    const t = window.setTimeout(() => setExpanded(false), 4200)

  1067→                <motion.button
  1068→                  key={item.id}

  1256→// ── Router ───────────────────────────────────────────────────

  1258→interface TabProps {
  1259→  active: PulseTab
  1260→  onChange: (tab: PulseTab) => void
  1261→  unread: number
  1262→}
  1263→
  1264→/**
  1265→ * Cross-file nav actions. R26-e extends the R25 union:
  1266→ *   new-chat / new-group  · existing compose flows (main-shell handles)
  1267→ *   search / settings     · existing inline triggers (main-shell handles)
  1268→ *   saved                 · NEW — jump to the profile Saved library
  1269→ *   stories               · NEW — open the chats-tab stories viewer
  1270→ * 'saved' and 'stories' are emitted by the nav overflow GlassMenu;
  1271→ * main-shell (lead-owned) must extend handleContextAction to wire them
  1272→ * (today its catch-all else-branch opens Settings — documented in the
  1273→ * R26-e worklog entry).
  1274→ */
  1275→export type NavContextAction = 'new-chat' | 'new-group' | 'search' | 'settings' | 'saved' | 'stories'
  1276→
  1277→/**
  1278→ * One-call navigation: reads the persisted style and renders the matching
  1279→ * architecture. `onContextAction` feeds the contextual dock's per-tab chip
  1280→ * AND the R26-e overflow GlassMenu (capsule / floating-top / island /
  1281→ * command-bar); `onSearch`/`onSettings` feed the command bar's inline
  1282→ * triggers.
  1283→ */
  1284→export function PulseNavBar({
  1285→  me,
  1286→  active,
  1287→  onChange,
  1288→  onSearch,
  1289→  onSettings,
  1290→  onContextAction,
  1291→}: {
  1292→  me: AppUser
  1293→  active: PulseTab
  1294→  onChange: (tab: PulseTab) => void
  1295→  onSearch: () => void
  1296→  onSettings: () => void
  1297→  onContextAction: (action: NavContextAction) => void
  1298→}) {
  1299→  const [style] = useNavStyle()
  1300→  const unread = useUnread(me)
  1301→  switch (style) {
  1302→    case 'floating-top':
  1303→      return <FloatingTopNav active={active} onChange={onChange} unread={unread} onContextAction={onContextAction} />
  1304→    case 'floating-dock':
  1305→      return <FloatingDock active={active} onChange={onChange} unread={unread} />
  1306→    case 'pill':
  1307→      return <PillNav active={active} onChange={onChange} unread={unread} />
  1308→    case 'bottom-bar':
  1309→      return <BottomBar active={active} onChange={onChange} unread={unread} />
  1310→    case 'tab-bar':
  1311→      return <TabBarNav active={active} onChange={onChange} unread={unread} />
  1312→    case 'floating-tab-bar':
  1313→      return <FloatingTabBar active={active} onChange={onChange} unread={unread} />
  1314→    case 'command-bar':
  1315→      return (
  1316→        <CommandBarNav
  1317→          active={active}
  1318→          onChange={onChange}
  1319→          unread={unread}
  1320→          onSearch={onSearch}
  1321→          onSettings={onSettings}
  1322→          onContextAction={onContextAction}
  1323→        />
  1324→      )
  1325→    case 'rail':
  1326→      return <RailNav active={active} onChange={onChange} unread={unread} />
  1327→    case 'island':
  1328→      return <IslandNav active={active} onChange={onChange} unread={unread} onContextAction={onContextAction} />
  1329→    case 'radial':
  1330→      return <RadialNav active={active} onChange={onChange} unread={unread} />
  1331→    case 'gesture':
  1332→      return <GestureNav active={active} onChange={onChange} unread={unread} />
  1333→    case 'contextual-dock':
  1334→      return <ContextualDock active={active} onChange={onChange} unread={unread} onContextAction={onContextAction} />
  1335→    case 'capsule':
  1336→    default:
  1337→      return <CapsuleNav active={active} onChange={onChange} unread={unread} onContextAction={onContextAction} />
  1338→  }
  1339→}
