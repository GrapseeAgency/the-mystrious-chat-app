    66→    66→ * around the hit so long messages stay one tidy line.

    68→    68→function SearchSnippet({ content, query }: { content: string; query: string }) {
    69→    69→  const lower = content.toLowerCase()
    70→    70→  const q = query.toLowerCase()
    71→    71→  const idx = q.length > 0 ? lower.indexOf(q) : -1
    72→    72→  let from = 0
    73→    73→  let clippedHead = false
    74→    74→  if (idx > 28) {
    75→    75→    from = idx - 24
    76→    76→    clippedHead = true
    77→    77→  }
    78→    78→  const end = Math.min(content.length, idx + q.length + 28)
    79→    79→  const clippedTail = end < content.length
    80→    80→  const body = content.slice(from, end)
    81→    81→  const localIdx = idx - from
    82→    82→  return (
    83→    83→    <p className="truncate text-[13px] text-zinc-500 dark:text-zinc-400">
    84→    84→      {clippedHead ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
    85→    85→      {idx >= 0 ? (
    86→    86→        <>
    87→    87→          {body.slice(0, localIdx)}
    88→    88→          <mark className="rounded bg-emerald-500/20 px-0.5 font-semibold text-emerald-700 dark:bg-emerald-500/25 dark:text-emerald-300">
    89→    89→            {body.slice(localIdx, localIdx + q.length)}
    90→    90→          </mark>
    91→    91→          {body.slice(localIdx + q.length)}
    92→    92→        </>
    93→    93→      ) : (
    94→    94→        body
    95→    95→      )}
    96→    96→      {clippedTail ? <span className="text-zinc-300 dark:text-zinc-600">…</span> : null}
    97→    97→    </p>
    98→    98→  )
    99→    99→}
