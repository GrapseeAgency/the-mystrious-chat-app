    67→/**
    68→ * Client-known set of connected appIds. Hydrated by the 100-app
    69→ * Promise.all on "My apps" chip activation (cached 60s), kept
    70→ * fresh by connect/disconnect mutations + per-app detail fetches.
    71→ */
    72→export async function fetchInstalledSet(meId: string): Promise<string[]> {
    73→  const results = await Promise.all(
    74→    MATRIX.map(async (app) => {
    75→      const res = await apiJson<{ installed: boolean }>(
    76→        `/api/hub/apps/${app.n}/install?userId=${encodeURIComponent(meId)}`,
    77→      )
    78→      return res.installed ? String(app.n) : null
    79→    }),
    80→  )
    81→  return results.filter((id): id is string => id !== null)
    82→}

    84→/** Fetch-on-demand hydration — dedupes, respects a 60s freshness window. */
    85→export function hydrateInstalledSet(qc: QueryClient, meId: string): Promise<string[]> {
    86→  return qc.fetchQuery({
    87→    queryKey: installedSetKey(meId),
    88→    queryFn: () => fetchInstalledSet(meId),
    89→    staleTime: 60_000,
    90→  })
    91→}

    93→/** Merge one app's server truth into the client-known set. */
    94→export function patchInstalledSet(qc: QueryClient, meId: string, appId: string, installed: boolean): void {
    95→  qc.setQueryData<string[]>(installedSetKey(meId), (prev) => {
    96→    if (!installed && prev === undefined) return undefined // never materialize a false-empty set
    97→    const base = prev ?? []
    98→    const has = base.includes(appId)
    99→    if (installed && !has) return [...base, appId]
   100→    if (!installed && has) return base.filter((id) => id !== appId)
   101→    return base
   102→  })
   103→}

   115→/** Per-app install status — GET truth; also patches the client-known set. */
   116→export function useAppInstallStatus(appId: string, meId: string) {
   117→  const qc = useQueryClient()
   118→  return useQuery({
   119→    queryKey: perAppKey(appId, meId),
   120→    queryFn: async () => {
   121→      const res = await apiJson<InstallStatus>(
   122→        `/api/hub/apps/${appId}/install?userId=${encodeURIComponent(meId)}`,
   123→      )
   124→      patchInstalledSet(qc, meId, appId, res.installed)
   125→      return res
   126→    },
   127→    staleTime: 30_000,
   128→  })

   131→/**
   132→ * Connect (POST) / disconnect (DELETE) with optimistic flip +
   133→ * rollback + error toast. Success shows as the UI flip itself
   134→ * (check pulse on tiles/detail), plus a short confirmation toast.
   135→ */
   136→export function useInstallToggle(app: { n: number; name: string }, meId: string) {
   137→  const qc = useQueryClient()
   138→  const appId = String(app.n)
   139→  return useMutation({
   140→    mutationFn: async (next: boolean) => {
   141→      const res = await apiJson<{ installed: boolean; installs: number }>(
   142→        `/api/hub/apps/${appId}/install`,
   143→        { method: next ? 'POST' : 'DELETE', body: JSON.stringify({ userId: meId }) },
   144→      )
   145→      return res
   146→    },
   147→    onMutate: async (next) => {
   148→      await qc.cancelQueries({ queryKey: perAppKey(appId, meId) })
   149→      const prevStatus = qc.getQueryData<InstallStatus>(perAppKey(appId, meId))
   150→      const prevSet = qc.getQueryData<string[]>(installedSetKey(meId))
   151→      if (prevStatus) {
   152→        qc.setQueryData<InstallStatus>(perAppKey(appId, meId), {
   153→          ...prevStatus,
   154→          installed: next,
   155→          status: next ? 'connected' : null,
   156→          installedAt: next ? (prevStatus.installedAt ?? new Date().toISOString()) : null,
   157→          installs: Math.max(0, prevStatus.installs + (next ? 1 : -1)),
   158→        })
   159→      } else if (next) {
   160→        qc.setQueryData<InstallStatus>(perAppKey(appId, meId), {
   161→          installed: true,
   162→          status: 'connected',
   163→          installedAt: new Date().toISOString(),
   164→          installs: 1,
   165→          installers: [],
   166→        })
   167→      }
   168→      patchInstalledSet(qc, meId, appId, next)
   169→      return { prevStatus, prevSet }
   170→    },
   171→    onError: (err: Error, _next, ctx) => {
   172→      if (ctx?.prevStatus !== undefined) qc.setQueryData(perAppKey(appId, meId), ctx.prevStatus)
   173→      else qc.removeQueries({ queryKey: perAppKey(appId, meId) })
   174→      if (ctx?.prevSet !== undefined) qc.setQueryData(installedSetKey(meId), ctx.prevSet)
   175→      else qc.removeQueries({ queryKey: installedSetKey(meId) })
   176→      toast.error(err instanceof Error ? err.message : 'Could not update the connection')
   177→    },
   178→    onSuccess: (res) => {
   179→      toast.success(res.installed ? `Connected to ${app.name}` : `Disconnected from ${app.name}`)
   180→      qc.setQueryData<InstallStatus>(perAppKey(appId, meId), (old) =>
   181→        old ? { ...old, installed: res.installed, installs: res.installs } : old,
   182→      )
   183→    },
   184→    onSettled: () => {
   185→      // marks stale even without observers → next detail mount refetches truth
   186→      void qc.invalidateQueries({ queryKey: perAppKey(appId, meId) })
   187→    },
   188→  })
   189→}

   223→/**
   224→ * Join (or found) the community — POST is idempotent on the server.
   225→ * onSuccess reconciles the query cache with server truth, teaches
   226→ * the chat list about the room, toasts, then hands the conversation
   227→ * id to `onJoined` (→ auto-open in the main chat surface).
   228→ * Hook-level callbacks keep firing even if the sheet unmounts on
   229→ * navigation, so the open-chat handoff is never dropped.
   230→ */
   231→export function useJoinCommunity(
   232→  app: { n: number; name: string },
   233→  meId: string,
   234→  opts?: { onJoined?: (conversationId: string) => void },
   235→) {
   236→  const qc = useQueryClient()
   237→  const appId = String(app.n)
   238→  return useMutation({
   239→    mutationFn: async () => {
   240→      const prior = qc.getQueryData<AppCommunityState>(communityKey(appId, meId))
   241→      const res = await apiJson<{ conversation: ConversationDetail; joined: boolean }>(
   242→        `/api/hub/apps/${appId}/community`,
   243→        { method: 'POST', body: JSON.stringify({ userId: meId }) },
   244→      )
   245→      return { res, founding: (prior?.conversation ?? null) === null }
   246→    },
   247→    onSuccess: ({ res, founding }) => {
   248→      const name =
   249→        res.conversation.name ?? `#${String(app.n).padStart(3, '0')} · ${app.name} community`
   250→      toast.success(`${founding ? 'Founded' : 'Joined'} ${name}`)
   251→      qc.setQueryData<AppCommunityState>(communityKey(appId, meId), {
   252→        conversation: res.conversation,
   253→        memberCount: res.conversation.members.length,
   254→        joined: true,
   255→      })
   256→      void qc.invalidateQueries({ queryKey: communityKey(appId, meId) })
   257→      void qc.invalidateQueries({ queryKey: ['conversations', meId] }) // chat list learns the room
   258→      opts?.onJoined?.(res.conversation.id)
   259→    },
   260→    onError: (err: Error) => {
   261→      toast.error(err instanceof Error ? err.message : 'Could not join the community')
   262→    },
   263→  })
   264→}

   268→/** 300ms count-up (framer motion values — no re-render churn). */
   269→export function CountUp({ value, className }: { value: number; className?: string }) {
   270→  const mv = useMotionValue(0)
   271→  const text = useTransform(mv, (v) => Math.round(v).toLocaleString())
   272→  useEffect(() => {
   273→    const controls = animate(mv, value, { duration: 0.3, ease: 'easeOut' })
   274→    return () => controls.stop()
   275→  }, [value, mv])
   276→  return <motion.span className={className}>{text}</motion.span>
   277→}

   285→      initial={{ scale: 1.4, opacity: 0.35 }}

   287→      transition={{ type: 'spring', stiffness: 520, damping: 15 }}
   288→    >
   289→      <CountUp value={value} />
   290→    </motion.span>
   291→  )
   292→}

   362→function formatDay(iso: string): string {
   363→  return new Date(iso).toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' })
   364→}

   366→/** Relative stamp for the viewer's own connector row ("3h ago"). */
   367→function formatRelative(iso: string, now: Date = new Date()): string {
   368→  const ms = now.getTime() - new Date(iso).getTime()
   369→  if (!Number.isFinite(ms)) return ''
   370→  const mins = Math.floor(ms / 60_000)
   371→  if (mins < 1) return 'just now'
   372→  if (mins < 60) return `${mins}m ago`
   373→  const hours = Math.floor(mins / 60)
   374→  if (hours < 24) return `${hours}h ago`
   375→  const days = Math.floor(hours / 24)
   376→  if (days < 7) return `${days}d ago`
   377→  return formatDay(iso)
   378→}
