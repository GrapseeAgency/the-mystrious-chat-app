    76→/** Disappearing-message TTL → short label (mirrors the room header menu). */
    77→function ttlLabel(ttlSeconds: number): string {
    78→  if (ttlSeconds <= 0) return 'Off'
    79→  if (ttlSeconds === 86_400) return '24h'
    80→  if (ttlSeconds === 604_800) return '7d'
    81→  if (ttlSeconds === 2_592_000) return '30d'
    82→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))}d`
    83→}
    84→
    85→/** TTL → spoken length for toasts/subtitles (mirrors the room header menu). */
    86→function ttlLong(ttlSeconds: number): string {
    87→  if (ttlSeconds === 86_400) return '24 hours'
    88→  if (ttlSeconds === 604_800) return '7 days'
    89→  if (ttlSeconds === 2_592_000) return '30 days'
    90→  return `${Math.max(1, Math.round(ttlSeconds / 86_400))} days`
    91→}

    94→/** The EXACT TTL presets the disappearing API accepts (seconds). 0 = off. */
    95→const TTL_STOPS = [0, 86_400, 604_800, 2_592_000] as const // off · 24h · 7d · 30d
    96→const TTL_STOP_LABELS = ['Off', '24h', '7d', '30d'] as const

    98→/**
    99→ * Horizontal drag-to-set timer track. Draggable springy thumb over the
   100→ * supported TTL stops, a live label riding under the thumb, and the old
   101→ * preset pills kept as tap-to-commit tick labels below. Pointer events
   102→ * only (touch + mouse), `touch-none` so the page never scrolls mid-drag.
   103→ */
   104→function DisappearingSlider({
   105→  ttlSeconds,
   106→  disabled,
   107→  reducedMotion,
   108→  onCommit,
   109→}: {
   110→  ttlSeconds: number
   111→  disabled: boolean
   112→  reducedMotion: boolean
   113→  onCommit: (ttlSeconds: number) => void
   114→}) {
   115→  const trackRef = useRef<HTMLDivElement | null>(null)
   116→  const [draftIdx, setDraftIdx] = useState(() =>
   117→    Math.max(0, TTL_STOPS.indexOf(ttlSeconds as (typeof TTL_STOPS)[number])),
   118→  )
   119→  const [dragging, setDragging] = useState(false)
   120→
   121→  // follow the committed server truth whenever a drag is not in flight
   122→  const committedIdx = Math.max(
   123→    0,
   124→    TTL_STOPS.indexOf(ttlSeconds as (typeof TTL_STOPS)[number]),
   125→  )
   126→  const shownIdx = dragging ? draftIdx : committedIdx
   127→
   128→  const idxFromClientX = (clientX: number): number => {
   129→    const el = trackRef.current
   130→    if (!el) return committedIdx
   131→    const rect = el.getBoundingClientRect()
   132→    if (rect.width <= 0) return committedIdx
   133→    const ratio = (clientX - rect.left) / rect.width
   134→    return Math.min(TTL_STOPS.length - 1, Math.max(0, Math.round(ratio * (TTL_STOPS.length - 1))))
   135→  }
   136→
   137→  const stopPct = (shownIdx / (TTL_STOPS.length - 1)) * 100
   138→  // keep the riding label inside the card at the extreme stops
   139→  const labelPct = Math.min(86, Math.max(14, stopPct))
   140→  const slideTransition = reducedMotion ? { duration: 0.12 } : spring.snappy
   141→
   142→  return (
   143→    <div className="mt-3 px-0.5 pb-0.5">
   144→      {/* live label riding under the thumb */}
   145→      <div className="relative mb-0.5 h-4" aria-hidden>
   146→        <motion.span
   147→          animate={{ left: `${labelPct}%` }}
   148→          transition={slideTransition}
   149→          style={{ x: '-50%' }}
   150→          className="absolute top-0 whitespace-nowrap rounded-full bg-emerald-500/15 px-1.5 py-px text-[9.5px] font-bold text-emerald-600 ring-1 ring-inset ring-emerald-500/25 dark:text-emerald-400"
   151→        >
   152→          {TTL_STOP_LABELS[shownIdx]}
   153→        </motion.span>
   154→      </div>
   155→
   156→      <div
   157→        ref={trackRef}
   158→        role="slider"
   159→        aria-label="Disappearing message timer"
   160→        aria-valuemin={0}
   161→        aria-valuemax={TTL_STOPS.length - 1}
   162→        aria-valuenow={shownIdx}
   163→        aria-valuetext={TTL_STOP_LABELS[shownIdx]}
   164→        aria-disabled={disabled}
   165→        onPointerDown={(event) => {
   166→          if (disabled) return
   167→          event.currentTarget.setPointerCapture(event.pointerId)
   168→          setDragging(true)
   169→          setDraftIdx(idxFromClientX(event.clientX))
   170→          haptic(6)
   171→        }}
   172→        onPointerMove={(event) => {
   173→          if (!dragging || disabled) return
   174→          const idx = idxFromClientX(event.clientX)
   175→          if (idx !== draftIdx) {
   176→            setDraftIdx(idx)
   177→            haptic(4)
   178→          }
   179→        }}
   180→        onPointerUp={(event) => {
   181→          if (!dragging) return
   182→          setDragging(false)
   183→          const idx = idxFromClientX(event.clientX)
   184→          setDraftIdx(idx)
   185→          haptic(10)
   186→          const next = TTL_STOPS[idx]
   187→          if (next !== ttlSeconds) onCommit(next)
   188→        }}
   189→        onPointerCancel={() => setDragging(false)}
   190→        className={cn('relative h-7 touch-none select-none', disabled && 'pointer-events-none opacity-50')}
   191→      >
   192→        {/* resting track + emerald fill up to the thumb */}
   193→        <div className="absolute top-1/2 h-1.5 w-full -translate-y-1/2 rounded-full bg-zinc-900/[0.07] dark:bg-white/10" />
   194→        <div
   195→          className="absolute top-1/2 h-1.5 -translate-y-1/2 rounded-full bg-gradient-to-r from-emerald-500/60 to-emerald-500 transition-[width] duration-150"
   196→          style={{ width: `${stopPct}%` }}
   197→        />
   198→        {/* tick stops */}
   199→        {TTL_STOPS.map((_, idx) => (
   200→          <span
   201→            key={idx}
   202→            className={cn(
   203→              'absolute top-1/2 size-2 -translate-x-1/2 -translate-y-1/2 rounded-full ring-2 ring-white dark:ring-zinc-900',
   204→              idx <= shownIdx ? 'bg-emerald-500' : 'bg-zinc-300 dark:bg-zinc-600',
   205→            )}
   206→            style={{ left: `${(idx / (TTL_STOPS.length - 1)) * 100}%` }}
   207→          />
   208→        ))}
   209→        {/* springy thumb — scales on grab for the haptic feel */}
   210→        <motion.span
   211→          aria-hidden
   212→          animate={{ left: `${stopPct}%`, scale: dragging ? 1.18 : 1 }}
   213→          transition={slideTransition}
   214→          style={{ x: '-50%', y: '-50%' }}
   215→          className="absolute top-1/2 z-10 flex size-5 items-center justify-center rounded-full bg-white shadow-[0_2px_10px_rgba(0,0,0,0.28)] ring-1 ring-black/10 dark:bg-zinc-100"
   216→        >
   217→          <span className="block size-2 rounded-full bg-emerald-500" />
   218→        </motion.span>
   219→      </div>
   220→
   221→      {/* the old preset pills — kept as tap-to-commit tick labels */}
   222→      <div className="mt-1.5 grid grid-cols-4 gap-1" role="group" aria-label="Timer presets">
   223→        {TTL_STOPS.map((t, idx) => (
   224→          <button
   225→            key={t}
   226→            type="button"
   227→            disabled={disabled}
   228→            onClick={() => onCommit(t)}
   229→            className={cn(
   230→              'h-7 rounded-full text-[11px] font-bold outline-none transition-colors active:scale-95 disabled:opacity-50',
   231→              idx === shownIdx
   232→                ? 'bg-emerald-500/15 text-emerald-700 ring-1 ring-emerald-400 dark:text-emerald-300'
   233→                : 'bg-zinc-900/[0.05] text-zinc-600 hover:bg-emerald-500/15 hover:text-emerald-700 dark:bg-white/[0.07] dark:text-zinc-300 dark:hover:text-emerald-400',
   234→            )}
   235→          >
   236→            {TTL_STOP_LABELS[idx]}
   237→          </button>
   238→        ))}
   239→      </div>
   240→    </div>
   241→  )
   242→}

   407→            }
   408→          : old,

   411→      void queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })

   429→  // two-tap confirm for destructive row actions (remove / dismiss-admin / leave)

   452→    }
   453→  }

   456→  const tapRemove = (userId: string) =>
   457→    tapConfirm('remove', userId, () => removeMutation.mutate(userId))

   459→    tapConfirm('demote', userId, () => roleMutation.mutate({ userId, action: 'demote' }))

   461→  // R28-b: full-screen add-members sub-view INSIDE the info page

   464→  // R33-b — channel/group photo: admins get an "Edit photo" overlay on the

   469→    mutationFn: async (photo: string) => {

   478→      queryClient.setQueryData<ConversationDetail>(['conversation', conversationId], (old) =>
   479→        old ? { ...old, photo } : old,

   481→      return { prev }

   483→    onSuccess: () => {
   484→      void queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
   485→      void queryClient.invalidateQueries({ queryKey: ['conversations', me.id] })
   486→      void queryClient.invalidateQueries({ queryKey: ['channels', me.id] })

   488→      haptic(12)
   489→    },
   490→    onError: (error, _photo, ctx) => {
   491→      if (ctx?.prev) queryClient.setQueryData(['conversation', conversationId], ctx.prev)
   492→      toast.error(error instanceof Error ? error.message : 'Could not update the photo')

   496→  const handlePhotoPicked = async (file: File) => {
   497→    if (photoBusy) return
